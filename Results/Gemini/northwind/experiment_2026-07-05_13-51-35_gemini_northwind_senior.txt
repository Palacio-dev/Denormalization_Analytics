-- ==============================================================================
-- Analytical Physical Model (Star Schema)
-- Target RDBMS: PostgreSQL (Analytics Optimized)
-- ==============================================================================

-- 1. DIMENSION: DATE
-- Role-playing dimension for Order, Required, and Shipped dates.
CREATE TABLE IF NOT EXISTS public.dim_date (
    date_key integer NOT NULL, -- Format: YYYYMMDD
    full_date date NOT NULL,
    day_of_week smallint NOT NULL,
    day_name character varying(10) NOT NULL,
    day_of_month smallint NOT NULL,
    day_of_year smallint NOT NULL,
    week_of_year smallint NOT NULL,
    month_number smallint NOT NULL,
    month_name character varying(10) NOT NULL,
    quarter smallint NOT NULL,
    year integer NOT NULL,
    is_weekend boolean NOT NULL,
    is_holiday boolean NOT NULL DEFAULT false,
    CONSTRAINT pk_dim_date PRIMARY KEY (date_key)
);

-- 2. DIMENSION: CUSTOMER
-- Flattens customers, customer_customer_demo, and customer_demographics.
CREATE TABLE IF NOT EXISTS public.dim_customer (
    customer_key serial NOT NULL, -- Surrogate Key
    customer_id character varying(5) NOT NULL, -- Natural Key
    company_name character varying(40) NOT NULL,
    contact_name character varying(30),
    contact_title character varying(30),
    address character varying(60),
    city character varying(15),
    region character varying(15),
    postal_code character varying(10),
    country character varying(15),
    customer_demographics text, -- Absorbed M:N relationship (comma-separated or JSON array)
    -- SCD Type 2 tracking columns
    valid_from timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    valid_to timestamp without time zone DEFAULT '9999-12-31',
    is_current boolean DEFAULT true,
    CONSTRAINT pk_dim_customer PRIMARY KEY (customer_key)
);

-- 3. DIMENSION: EMPLOYEE
-- Flattens employees, self-referencing hierarchy, employee_territories, territories, and region.
CREATE TABLE IF NOT EXISTS public.dim_employee (
    employee_key serial NOT NULL, -- Surrogate Key
    employee_id smallint NOT NULL, -- Natural Key
    full_name character varying(50) NOT NULL,
    title character varying(30),
    hire_date date,
    manager_name character varying(50), -- Flattened hierarchy
    city character varying(15),
    region character varying(15),
    country character varying(15),
    assigned_territories text, -- Absorbed M:N relationship
    assigned_regions text,     -- Absorbed M:N relationship
    -- SCD Type 2 tracking columns
    valid_from timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    valid_to timestamp without time zone DEFAULT '9999-12-31',
    is_current boolean DEFAULT true,
    CONSTRAINT pk_dim_employee PRIMARY KEY (employee_key)
);

-- 4. DIMENSION: PRODUCT
-- Flattens products, categories, and suppliers.
CREATE TABLE IF NOT EXISTS public.dim_product (
    product_key serial NOT NULL, -- Surrogate Key
    product_id smallint NOT NULL, -- Natural Key
    product_name character varying(40) NOT NULL,
    quantity_per_unit character varying(20),
    unit_price numeric(10,2), -- Casted from REAL for financial accuracy
    is_discontinued boolean NOT NULL,
    -- Category attributes
    category_name character varying(15),
    category_description text,
    -- Supplier attributes
    supplier_name character varying(40),
    supplier_country character varying(15),
    supplier_region character varying(15),
    -- SCD Type 2 tracking columns
    valid_from timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    valid_to timestamp without time zone DEFAULT '9999-12-31',
    is_current boolean DEFAULT true,
    CONSTRAINT pk_dim_product PRIMARY KEY (product_key)
);

-- 5. DIMENSION: SHIPPER
CREATE TABLE IF NOT EXISTS public.dim_shipper (
    shipper_key serial NOT NULL, -- Surrogate Key
    shipper_id smallint NOT NULL, -- Natural Key
    company_name character varying(40) NOT NULL,
    CONSTRAINT pk_dim_shipper PRIMARY KEY (shipper_key)
);

-- 6. FACT: SALES
-- Line-item grain fact table combining orders and order_details.
-- Partitioned by order_date_key for query pruning on time-series analysis.
CREATE TABLE IF NOT EXISTS public.fact_sales (
    sales_key bigserial NOT NULL,
    order_id smallint NOT NULL, -- Degenerate Dimension
    
    -- Foreign Keys to Dimensions
    order_date_key integer NOT NULL,
    required_date_key integer,
    shipped_date_key integer,
    customer_key integer NOT NULL,
    employee_key integer NOT NULL,
    product_key integer NOT NULL,
    shipper_key integer,
    
    -- Degenerate Order Header Attributes
    ship_city character varying(15),
    ship_region character varying(15),
    ship_country character varying(15),
    
    -- Measures
    quantity smallint NOT NULL,
    unit_price numeric(10,2) NOT NULL,
    discount numeric(5,4) NOT NULL,
    
    -- Derived Additive Measures
    gross_sales_amount numeric(12,2) NOT NULL, -- quantity * unit_price
    net_sales_amount numeric(12,2) NOT NULL,   -- quantity * unit_price * (1 - discount)
    allocated_freight numeric(10,2),           -- Freight distributed proportionally by line item value
    
    CONSTRAINT pk_fact_sales PRIMARY KEY (order_date_key, sales_key)
) PARTITION BY RANGE (order_date_key);

-- Create default partition (Required for PostgreSQL partitioning)
CREATE TABLE public.fact_sales_default PARTITION OF public.fact_sales DEFAULT;

ALTER TABLE public.fact_sales ADD CONSTRAINT fk_fact_sales_shipper FOREIGN KEY (shipper_key) REFERENCES public.dim_shipper(shipper_key);

-- Performance Optimization: Indexes for Star Schema Joins
-- ==============================================================================
-- Bitmap indexes are ideal for OLAP, but PostgreSQL uses B-Trees. 
-- Creating indexes on all foreign keys in the fact table to optimize Star Joins.
CREATE INDEX idx_fact_sales_customer_key ON public.fact_sales(customer_key);

CREATE INDEX idx_fact_sales_order_date_key ON public.fact_sales(order_date_key);
