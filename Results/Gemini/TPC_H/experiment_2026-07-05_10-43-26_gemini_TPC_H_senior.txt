### Analytical Physical Model DDL (OLAP Optimized)

The following SQL DDL is designed for a modern columnar OLAP data warehouse (such as Snowflake, BigQuery, or Redshift). It implements the Star Schema design detailed in the JSON above.

```sql
-- ==============================================================================
-- DIMENSIONAL MODEL DDL (STAR SCHEMA)
-- ==============================================================================

-- ------------------------------------------------------------------------------
-- 1. DIM_DATE (Conformed Dimension)
-- ------------------------------------------------------------------------------
CREATE TABLE DIM_DATE (
    DATE_SK         INTEGER NOT NULL,       -- Format: YYYYMMDD
    FULL_DATE       DATE NOT NULL,
    DAY_OF_WEEK     VARCHAR(10) NOT NULL,
    DAY_NUM_IN_MONTH INTEGER NOT NULL,
    MONTH_NAME      VARCHAR(10) NOT NULL,
    MONTH_NUM       INTEGER NOT NULL,
    QUARTER         INTEGER NOT NULL,
    YEAR            INTEGER NOT NULL,
    IS_WEEKEND_FLAG BOOLEAN NOT NULL,
    PRIMARY KEY (DATE_SK)
);
-- Note: Date dimensions are typically small enough to be broadcasted/replicated across all compute nodes.

-- ------------------------------------------------------------------------------
-- 2. DIM_CUSTOMER (Flattened Snowflake)
-- ------------------------------------------------------------------------------
CREATE TABLE DIM_CUSTOMER (
    CUST_SK         INTEGER NOT NULL,       -- Surrogate Key
    C_CUSTKEY       INTEGER NOT NULL,       -- Natural/Business Key
    C_NAME          VARCHAR(25) NOT NULL,
    C_ADDRESS       VARCHAR(40) NOT NULL,
    C_PHONE         CHAR(15) NOT NULL,
    C_ACCTBAL       DECIMAL(15,2) NOT NULL,
    C_MKTSEGMENT    CHAR(10) NOT NULL,
    NATION_NAME     CHAR(25) NOT NULL,      -- Denormalized from NATION
    REGION_NAME     CHAR(25) NOT NULL,      -- Denormalized from REGION
    C_COMMENT       VARCHAR(117),
    PRIMARY KEY (CUST_SK)
);
-- Clustering Recommendation: Cluster by REGION_NAME, NATION_NAME, C_MKTSEGMENT

-- ------------------------------------------------------------------------------
-- 3. DIM_SUPPLIER (Flattened Snowflake)
-- ------------------------------------------------------------------------------
CREATE TABLE DIM_SUPPLIER (
    SUPP_SK         INTEGER NOT NULL,       -- Surrogate Key
    S_SUPPKEY       INTEGER NOT NULL,       -- Natural/Business Key
    S_NAME          CHAR(25) NOT NULL,
    S_ADDRESS       VARCHAR(40) NOT NULL,
    S_PHONE         CHAR(15) NOT NULL,
    S_ACCTBAL       DECIMAL(15,2) NOT NULL,
    NATION_NAME     CHAR(25) NOT NULL,      -- Denormalized from NATION
    REGION_NAME     CHAR(25) NOT NULL,      -- Denormalized from REGION
    S_COMMENT       VARCHAR(101),
    PRIMARY KEY (SUPP_SK)
);
-- Clustering Recommendation: Cluster by REGION_NAME, NATION_NAME

-- ------------------------------------------------------------------------------
-- 4. DIM_PART (Standard Dimension)
-- ------------------------------------------------------------------------------
CREATE TABLE DIM_PART (
    PART_SK         INTEGER NOT NULL,       -- Surrogate Key
    P_PARTKEY       INTEGER NOT NULL,       -- Natural/Business Key
    P_NAME          VARCHAR(55) NOT NULL,
    P_MFGR          CHAR(25) NOT NULL,
    P_BRAND         CHAR(10) NOT NULL,
    P_TYPE          VARCHAR(25) NOT NULL,
    P_SIZE          INTEGER NOT NULL,
    P_CONTAINER     CHAR(10) NOT NULL,
    P_RETAILPRICE   DECIMAL(15,2) NOT NULL,
    P_COMMENT       VARCHAR(23),
    PRIMARY KEY (PART_SK)
);
-- Clustering Recommendation: Cluster by P_BRAND, P_MFGR, P_TYPE

-- ------------------------------------------------------------------------------
-- 5. FACT_LINEITEM (Wide Fact Table)
-- ------------------------------------------------------------------------------
CREATE TABLE FACT_LINEITEM (
    -- Surrogate Keys
    LINEITEM_SK     BIGINT NOT NULL,        -- Surrogate Key for the Fact row
    CUST_SK         INTEGER NOT NULL,
    PART_SK         INTEGER NOT NULL,
    SUPP_SK         INTEGER NOT NULL,
    
    -- Date Surrogate Keys
    ORDER_DATE_SK   INTEGER NOT NULL,
    SHIP_DATE_SK    INTEGER NOT NULL,
    COMMIT_DATE_SK  INTEGER NOT NULL,
    RECEIPT_DATE_SK INTEGER NOT NULL,
    
    -- Degenerate Dimensions / Natural Keys
    ORDERKEY        BIGINT NOT NULL,
    LINENUMBER      INTEGER NOT NULL,
    
    -- Denormalized Order Header Attributes (from ORDERS)
    ORDERSTATUS     CHAR(1) NOT NULL,
    ORDERPRIORITY   CHAR(15) NOT NULL,
    SHIPPRIORITY    INTEGER NOT NULL,
    CLERK           CHAR(15) NOT NULL,
    
    -- Lineitem Attributes
    RETURNFLAG      CHAR(1) NOT NULL,
    LINESTATUS      CHAR(1) NOT NULL,
    SHIPINSTRUCT    CHAR(25) NOT NULL,
    SHIPMODE        CHAR(10) NOT NULL,
    
    -- Base Measures
    QUANTITY        DECIMAL(15,2) NOT NULL,
    EXTENDEDPRICE   DECIMAL(15,2) NOT NULL,
    DISCOUNT        DECIMAL(15,2) NOT NULL,
    TAX             DECIMAL(15,2) NOT NULL,
    
    -- Derived Analytical Measures
    NET_REVENUE     DECIMAL(15,4) NOT NULL, -- EXTENDEDPRICE * (1 - DISCOUNT)
    TOTAL_CHARGES   DECIMAL(15,4) NOT NULL, -- EXTENDEDPRICE * (1 - DISCOUNT) * (1 + TAX)
    
    PRIMARY KEY (LINEITEM_SK)
);
-- Partitioning Recommendation: Partition by ORDER_DATE_SK (or SHIP_DATE_SK depending on primary query access pattern)
-- Clustering Recommendation: Cluster by CUST_SK, PART_SK, SUPP_SK

-- ------------------------------------------------------------------------------
-- 6. FACT_PARTSUPP (Inventory/Supply Fact Table)
-- ------------------------------------------------------------------------------
CREATE TABLE FACT_PARTSUPP (
    -- Surrogate Keys
    PART_SK         INTEGER NOT NULL,
    SUPP_SK         INTEGER NOT NULL,
    
    -- Measures
    AVAILQTY        INTEGER NOT NULL,
    SUPPLYCOST      DECIMAL(15,2) NOT NULL,
    INVENTORY_VALUE DECIMAL(15,2) NOT NULL, -- Derived: AVAILQTY * SUPPLYCOST
    
    PRIMARY KEY (PART_SK, SUPP_SK)
);
-- Clustering Recommendation: Cluster by SUPP_SK, PART_SK
```
