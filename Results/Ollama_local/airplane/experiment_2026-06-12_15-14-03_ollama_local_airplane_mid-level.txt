-- Drop existing tables if they exist to ensure clean state
DROP TABLE IF EXISTS fact_bookings CASCADE;

DROP TABLE IF EXISTS dim_aircrafts CASCADE;
DROP TABLE IF EXISTS dim_passengers CASCADE;

-- 1. Dimension Tables (Simplified for Analytics)
-- These tables are kept minimal but may be joined if needed for specific historical queries.
CREATE TABLE IF NOT EXISTS dim_airports (
    airport_code character(3) COLLATE pg_catalog."default" NOT NULL,
    airport_name jsonb NOT NULL,
    city jsonb NOT NULL,
    coordinates point NOT NULL,
    timezone text COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT dim_airports_pkey PRIMARY KEY (airport_code)
);

CREATE TABLE IF NOT EXISTS dim_aircrafts (
    aircraft_code character(3) COLLATE pg_catalog."default" NOT NULL,
    model jsonb NOT NULL,
    range integer NOT NULL,
    CONSTRAINT dim_aircrafts_pkey PRIMARY KEY (aircraft_code)
);

-- 2. Fact Tables (Denormalized for Read Performance)

-- Fact_Flights: Embeds airport and aircraft dimensions directly.
CREATE TABLE IF NOT EXISTS fact_flights (
    flight_id serial NOT NULL,
    flight_no character(6) COLLATE pg_catalog."default" NOT NULL,
    scheduled_departure timestamp with time zone NOT NULL,
    scheduled_arrival timestamp with time zone NOT NULL,
    departure_airport character(3) COLLATE pg_catalog."default" NOT NULL,
    arrival_airport character(3) COLLATE pg_catalog."default" NOT NULL,
    status character varying(20) COLLATE pg_catalog."default" NOT NULL,
    aircraft_code character(3) COLLATE pg_catalog."default" NOT NULL,
    -- Embedded Dimension Data
    airport_name jsonb NOT NULL,
    airport_city jsonb NOT NULL,
    airport_coordinates point NOT NULL,
    airport_timezone text COLLATE pg_catalog."default" NOT NULL,
    aircraft_model jsonb NOT NULL,
    aircraft_range integer NOT NULL,
    -- Derived Attributes (Calculated at load time)
    flight_distance_km integer NOT NULL,
    flight_duration_hours numeric(5, 2) NOT NULL,
    -- Actual times (nullable)
    actual_departure timestamp with time zone,
    actual_arrival timestamp with time zone,
    actual_departure_diff numeric(5, 2),
    actual_arrival_diff numeric(5, 2),
    CONSTRAINT fact_flights_pkey PRIMARY KEY (flight_id),
    CONSTRAINT fact_flights_flight_no_unique UNIQUE (flight_no, scheduled_departure)
);

-- Fact_Bookings: Consolidated table merging bookings, tickets, and ticket_flights.
CREATE TABLE IF NOT EXISTS fact_bookings (
    ticket_no character(13) COLLATE pg_catalog."default" NOT NULL,
    flight_id integer NOT NULL,
    book_ref character(6) COLLATE pg_catalog."default" NOT NULL,
    book_date timestamp with time zone NOT NULL,
    total_amount numeric(10, 2) NOT NULL,
    -- Passenger Details (from tickets)
    passenger_id character varying(20) COLLATE pg_catalog."default" NOT NULL,
    passenger_name text COLLATE pg_catalog."default" NOT NULL,
    contact_data jsonb NOT NULL,
    -- Ticket Segment Details (from ticket_flights)
    fare_conditions character varying(10) COLLATE pg_catalog."default" NOT NULL,
    segment_amount numeric(10, 2) NOT NULL,
    -- Boarding Pass Details (from boarding_passes)
    boarding_no integer NOT NULL,
    seat_no character varying(4) COLLATE pg_catalog."default" NOT NULL,
    seat_assignment_jsonb jsonb NOT NULL,
    CONSTRAINT fact_bookings_pkey PRIMARY KEY (ticket_no),
    CONSTRAINT fact_bookings_flight_id_unique UNIQUE (flight_id, ticket_no)
);

-- 3. Indexes for Performance

CREATE INDEX IF NOT EXISTS idx_fact_flights_airport_code ON fact_flights (departure_airport);
