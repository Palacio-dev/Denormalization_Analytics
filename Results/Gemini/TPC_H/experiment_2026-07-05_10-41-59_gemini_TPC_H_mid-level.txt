```sql
-- ==============================================================================
-- Analytical Data Model (Star Schema)
-- Optimized for Read-Heavy BI / OLAP Workloads
-- ==============================================================================

-- ------------------------------------------------------------------------------
-- 1. DIMENSION: dim_customer
-- Combines CUSTOMER, NATION, and REGION
-- ------------------------------------------------------------------------------
CREATE TABLE dim_customer (
    customer_key       INTEGER NOT NULL,
    customer_name      VARCHAR(25) NOT NULL,
    customer_address   VARCHAR(40) NOT NULL,
    customer_phone     CHAR(15) NOT NULL,
    account_balance    DECIMAL(15,2) NOT NULL,
    market_segment     CHAR(10) NOT NULL,
    nation_name        CHAR(25) NOT NULL,
    region_name        CHAR(25) NOT NULL,
    PRIMARY KEY (customer_key)
);

-- ------------------------------------------------------------------------------
-- 2. DIMENSION: dim_supplier
-- Combines SUPPLIER, NATION, and REGION
-- ------------------------------------------------------------------------------
CREATE TABLE dim_supplier (
    supplier_key       INTEGER NOT NULL,
    supplier_name      CHAR(25) NOT NULL,
    supplier_address   VARCHAR(40) NOT NULL,
    supplier_phone     CHAR(15) NOT NULL,
    account_balance    DECIMAL(15,2) NOT NULL,
    nation_name        CHAR(25) NOT NULL,
    region_name        CHAR(25) NOT NULL,
    PRIMARY KEY (supplier_key)
);

-- ------------------------------------------------------------------------------
-- 3. DIMENSION: dim_part
-- Based directly on PART table (comments removed for analytical optimization)
-- ------------------------------------------------------------------------------
CREATE TABLE dim_part (
    part_key           INTEGER NOT NULL,
    part_name          VARCHAR(55) NOT NULL,
    manufacturer       CHAR(25) NOT NULL,
    brand              CHAR(10) NOT NULL,
    part_type          VARCHAR(25) NOT NULL,
    part_size          INTEGER NOT NULL,
    container          CHAR(10) NOT NULL,
    retail_price       DECIMAL(15,2) NOT NULL,
    PRIMARY KEY (part_key)
);

-- ------------------------------------------------------------------------------
-- 4. FACT: fact_sales
-- Combines ORDERS and LINEITEM. 
-- Grain: One row per order line item.
-- ------------------------------------------------------------------------------
CREATE TABLE fact_sales (
    -- Degenerate Dimensions & Primary Keys
    order_key          INTEGER NOT NULL,
    line_number        INTEGER NOT NULL,
    
    -- Foreign Keys to Dimensions
    customer_key       INTEGER NOT NULL,
    part_key           INTEGER NOT NULL,
    supplier_key       INTEGER NOT NULL,
    
    -- Dates (Can optionally link to a dim_date in a full Enterprise DW)
    order_date         DATE NOT NULL,
    ship_date          DATE NOT NULL,
    commit_date        DATE NOT NULL,
    receipt_date       DATE NOT NULL,
    
    -- Order Header Attributes (Denormalized)
    order_status       CHAR(1) NOT NULL,
    order_priority     CHAR(15) NOT NULL,
    ship_priority      INTEGER NOT NULL,
    clerk              CHAR(15) NOT NULL,
    
    -- Line Item Attributes
    return_flag        CHAR(1) NOT NULL,
    line_status        CHAR(1) NOT NULL,
    ship_instruct      CHAR(25) NOT NULL,
    ship_mode          CHAR(10) NOT NULL,
    
    -- Base Measures
    quantity           DECIMAL(15,2) NOT NULL,
    extended_price     DECIMAL(15,2) NOT NULL,
    discount_pct       DECIMAL(15,2) NOT NULL,
    tax_pct            DECIMAL(15,2) NOT NULL,
    
    -- Derived Measures (Pre-calculated for BI Performance)
    net_revenue        DECIMAL(15,2) NOT NULL, -- extended_price * (1 - discount_pct)
    total_amount       DECIMAL(15,2) NOT NULL, -- extended_price * (1 - discount_pct) * (1 + tax_pct)
    
    PRIMARY KEY (order_key, line_number)
);

-- ------------------------------------------------------------------------------
-- 5. FACT: fact_part_supplier
-- Based on PARTSUPP. Acts as an inventory/supply-cost snapshot fact.
-- Grain: One row per part per supplier.
-- ------------------------------------------------------------------------------
CREATE TABLE fact_part_supplier (
    part_key           INTEGER NOT NULL,
    supplier_key       INTEGER NOT NULL,
    
    -- Measures
    available_qty      INTEGER NOT NULL,
    supply_cost        DECIMAL(15,2) NOT NULL,
    
    PRIMARY KEY (part_key, supplier_key)
);
```
