-- ANALYTICAL PHYSICAL MODEL: HYBRID DIMENSIONAL SCHEMA
-- Strategy: Star Schema with Dimension Flattening

-- Drop existing tables if they exist (for clean migration)
DROP TABLE IF EXISTS fact_orders;
DROP TABLE IF EXISTS dim_customer;
DROP TABLE IF EXISTS dim_product;

-- ---------------------------------------------------------------
-- 1. DIMENSION TABLES
-- ---------------------------------------------------------------

-- Dimension: Customers
-- Stores slowly changing attributes for segmentation and historical analysis.
CREATE TABLE dim_customer (
    customer_id INT PRIMARY KEY, -- Surrogate Key for reference consistency
    customer_name VARCHAR(100),
    customer_email VARCHAR(100),
    customer_address TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    last_updated TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Dimension: Products
-- Stores stable attributes for product lifecycle analysis.
CREATE TABLE dim_product (
    product_id INT PRIMARY KEY, -- Surrogate Key
    product_name VARCHAR(100),
    product_category VARCHAR(50), -- Added for analytical categorization
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- ---------------------------------------------------------------
-- 2. FACT TABLE (DENORMALIZED)
-- ---------------------------------------------------------------

-- Fact: Orders
-- A wide table containing all transactional data to minimize joins.
-- Partitioned by order_date for efficient time-based aggregation.
CREATE TABLE fact_orders (
    order_date DATE,
    customer_id INT, -- Foreign key reference to dim_customer
    customer_name VARCHAR(100), -- Denormalized for immediate retrieval
    customer_email VARCHAR(100), -- Denormalized for immediate retrieval
    customer_address TEXT,     -- Denormalized for immediate retrieval
    
    product_id INT, -- Foreign key reference to dim_product
    product_name VARCHAR(100), -- Denormalized for immediate retrieval
    
    quantity INT,
    unit_price DECIMAL(10, 2), -- Original price from OLTP
    
    -- Derived metrics (Calculated on write or stored for speed)
    revenue DECIMAL(15, 2) GENERATED ALWAYS AS (quantity * unit_price) STORED,
    total_order_value DECIMAL(15, 2) GENERATED ALWAYS AS (SUM(quantity * unit_price)) STORED,
    
    PRIMARY KEY (order_date, customer_id, product_id), -- Composite PK for uniqueness
    CONSTRAINT fk_orders_customer FOREIGN KEY (customer_id) REFERENCES dim_customer(customer_id),
    CONSTRAINT fk_orders_product FOREIGN KEY (product_id) REFERENCES dim_product(product_id)
);

    order_date,
