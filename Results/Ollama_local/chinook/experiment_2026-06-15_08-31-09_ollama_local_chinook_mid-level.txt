-- 1. DIMENSION TABLES (Simplified)
-- Stores reference keys and minimal attributes for easy filtering.

CREATE TABLE IF NOT EXISTS public."Employee_Dim" (
    "EmployeeId" integer NOT NULL,
    "LastName" character varying(20) COLLATE pg_catalog."default" NOT NULL,
    "FirstName" character varying(20) COLLATE pg_catalog."default" NOT NULL,
    "Title" character varying(30) COLLATE pg_catalog."default",
    "ReportsTo" integer,
    "BirthDate" timestamp without time zone,
    "HireDate" timestamp without time zone,
    CONSTRAINT "PK_Employee_Dim" PRIMARY KEY ("EmployeeId")
);

CREATE TABLE IF NOT EXISTS public."Customer_Dim" (
    "CustomerId" integer NOT NULL,
    "FirstName" character varying(40) COLLATE pg_catalog."default" NOT NULL,
    "LastName" character varying(20) COLLATE pg_catalog."default" NOT NULL,
    "Company" character varying(80) COLLATE pg_catalog."default",
    "Email" character varying(60) COLLATE pg_catalog."default" NOT NULL,
    "SupportRepId" integer,
    CONSTRAINT "PK_Customer_Dim" PRIMARY KEY ("CustomerId")
);

-- Note: Billing Address is moved to Fact table for faster aggregation.

-- 2. FACT TABLES (Denormalized)
-- Contains all transactional and descriptive data to eliminate joins.

-- Sales Fact Table
-- Combines Invoice, InvoiceLine, and Track details.
-- Also includes Album, Artist, Genre, and Media Type details.
CREATE TABLE IF NOT EXISTS public."Sales_Fact" (
    "InvoiceId" integer NOT NULL,
    "CustomerId" integer NOT NULL,
    "InvoiceDate" timestamp without time zone NOT NULL,
    "BillingAddress" character varying(70) COLLATE pg_catalog."default",
    "BillingCity" character varying(40) COLLATE pg_catalog."default",
    "BillingState" character varying(40) COLLATE pg_catalog."default",
    "BillingCountry" character varying(40) COLLATE pg_catalog."default",
    "BillingPostalCode" character varying(10) COLLATE pg_catalog."default",
    "Total" numeric(10, 2) NOT NULL,
    
    -- Track Details (Flattened from Track table)
    "TrackId" integer NOT NULL,
    "TrackName" character varying(200) COLLATE pg_catalog."default" NOT NULL,
    "AlbumId" integer,
    "AlbumTitle" character varying(160) COLLATE pg_catalog."default",
    "ArtistId" integer,
    "ArtistName" character varying(120) COLLATE pg_catalog."default",
    "GenreId" integer,
    "GenreName" character varying(120) COLLATE pg_catalog."default",
    "MediaTypeId" integer NOT NULL,
    "MediaTypeName" character varying(120) COLLATE pg_catalog."default",
    "Composer" character varying(220) COLLATE pg_catalog."default",
    "Milliseconds" integer NOT NULL,
    "Bytes" integer,
    "UnitPrice" numeric(10, 2) NOT NULL,
    "Quantity" integer NOT NULL,
    
    -- Support Rep Details (Flattened from Customer/Employee link)
    "SupportRepLastName" character varying(20) COLLATE pg_catalog."default",
    "SupportRepFirstName" character varying(20) COLLATE pg_catalog."default",
    
    CONSTRAINT "PK_Sales_Fact" PRIMARY KEY ("InvoiceId", "TrackId")
);
