### Analytical Physical Model Design (SQL DDL)

The following script creates the optimized Star Schema. It includes surrogate keys, denormalized dimensions, and a central snapshot fact table. Indexing and clustering strategies are included as comments, assuming a modern columnar or PostgreSQL-based OLAP engine (like Redshift or standard Postgres tuned for analytics).

```sql
-- ==============================================================================
-- TARGET ANALYTICAL SCHEMA (OLAP STAR SCHEMA)
-- ==============================================================================

-- 1. DATE DIMENSION
-- Enables robust time-series analysis and standard BI time hierarchies.
CREATE TABLE IF NOT EXISTS public.dim_data (
    sk_data INTEGER PRIMARY KEY, -- Format: YYYYMMDD
    data_completa DATE NOT NULL,
    ano INTEGER NOT NULL,
    trimestre INTEGER NOT NULL,
    mes INTEGER NOT NULL,
    nome_mes VARCHAR(20) NOT NULL,
    dia INTEGER NOT NULL,
    dia_semana VARCHAR(20) NOT NULL,
    flag_fim_semana BOOLEAN NOT NULL
);

-- 2. GEOGRAPHY DIMENSION (Denormalized Localizacoes + Municipios + Estados)
-- Flattens the snowflake hierarchy and handles the dual-municipality complexity.
CREATE TABLE IF NOT EXISTS public.dim_geografia (
    sk_geografia SERIAL PRIMARY KEY,
    id_localizacao_nk INTEGER NOT NULL, -- Natural Key from source
    lat_localizacao DOUBLE PRECISION,
    long_localizacao DOUBLE PRECISION,
    
    -- Primary Municipality (id_municipio1)
    municipio_principal VARCHAR NOT NULL,
    uf_principal CHAR(2) NOT NULL,
    
    -- Secondary Municipality (id_municipio2) - Nullable for non-border locations
    municipio_secundario VARCHAR,
    uf_secundario CHAR(2)
);

-- 3. USINA (PLANT) DIMENSION (Denormalized Usina + Proprietarios + Rios)
-- Absorbs low-cardinality lookup tables to create a rich, flat dimension.
CREATE TABLE IF NOT EXISTS public.dim_usina (
    sk_usina SERIAL PRIMARY KEY,
    id_usina_nk INTEGER NOT NULL, -- Natural Key from source
    nome_usina VARCHAR NOT NULL,
    estagio VARCHAR,
    lat_usina DOUBLE PRECISION,
    long_usina DOUBLE PRECISION,
    
    -- Absorbed Lookups
    nome_proprietario VARCHAR,
    nome_rio VARCHAR
);

-- 4. USINA CAPACITY FACT TABLE (Snapshot Fact)
-- Isolates measures and foreign keys. 
CREATE TABLE IF NOT EXISTS public.fact_usina_capacidade (
    sk_usina INTEGER NOT NULL,
    sk_geografia INTEGER NOT NULL,
    sk_data_iniciop INTEGER,     -- FK to dim_data
    sk_data_atualizacao INTEGER, -- FK to dim_data
    
    -- Measures
    potkw FLOAT,
    potfis FLOAT,
    
    -- Constraints
    CONSTRAINT fk_fact_usina FOREIGN KEY (sk_usina) REFERENCES public.dim_usina(sk_usina),
    CONSTRAINT fk_fact_geografia FOREIGN KEY (sk_geografia) REFERENCES public.dim_geografia(sk_geografia),
    CONSTRAINT fk_fact_data_inicio FOREIGN KEY (sk_data_iniciop) REFERENCES public.dim_data(sk_data),
    CONSTRAINT fk_fact_data_atualizacao FOREIGN KEY (sk_data_atualizacao) REFERENCES public.dim_data(sk_data)
);

-- ==============================================================================
-- OPTIMIZATION & PHYSICAL DESIGN RECOMMENDATIONS
-- ==============================================================================

-- Partitioning: 
-- If the dataset grows significantly and is often queried by the start date, 
-- partitioning the fact table by year/month of 'sk_data_iniciop' is recommended.
-- Example (PostgreSQL native partitioning):
-- PARTITION BY RANGE (sk_data_iniciop);

-- Indexing / Clustering:
-- In a traditional PostgreSQL environment used for OLAP, B-Tree indexes on Surrogate Keys are vital.
CREATE INDEX idx_fact_usina_sk ON public.fact_usina_capacidade(sk_usina);
CREATE INDEX idx_fact_geografia_sk ON public.fact_usina_capacidade(sk_geografia);
CREATE INDEX idx_fact_data_inicio_sk ON public.fact_usina_capacidade(sk_data_iniciop);

-- For Cloud Data Warehouses (e.g., Snowflake, BigQuery, Redshift):
-- 1. Redshift: 
--    DISTKEY(sk_usina) 
--    SORTKEY(sk_geografia, sk_data_iniciop)
-- 2. BigQuery: 
--    CLUSTER BY sk_geografia, sk_usina
-- 3. Snowflake: 
--    CLUSTER BY (sk_geografia, sk_data_iniciop)

COMMIT;
```
