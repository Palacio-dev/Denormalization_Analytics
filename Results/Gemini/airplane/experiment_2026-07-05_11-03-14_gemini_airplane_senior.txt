-- ==============================================================================
-- DIMENSIONAL MODEL DDL (OLAP)
-- Schema: analytics
-- ==============================================================================

CREATE SCHEMA IF NOT EXISTS analytics;

-- ------------------------------------------------------------------------------
-- 1. DIMENSIONS
-- ------------------------------------------------------------------------------

-- Dim Date (Standard Date Dimension)
CREATE TABLE IF NOT EXISTS analytics.dim_date (
    date_sk INT NOT NULL,
    full_date DATE NOT NULL,
    day_of_week INT NOT NULL,
    day_name VARCHAR(10) NOT NULL,
    month_num INT NOT NULL,
    month_name VARCHAR(10) NOT NULL,
    quarter INT NOT NULL,
    year INT NOT NULL,
    is_weekend BOOLEAN NOT NULL,
    CONSTRAINT dim_date_pkey PRIMARY KEY (date_sk)
);

-- Dim Aircrafts
-- Flattened JSON, Pre-aggregated seat counts from source 'seats' table
CREATE TABLE IF NOT EXISTS analytics.dim_aircrafts (
    aircraft_sk BIGSERIAL NOT NULL,
    aircraft_code CHAR(3) NOT NULL,
    model_en VARCHAR(100) NOT NULL,
    model_ru VARCHAR(100),
    range_km INT NOT NULL,
    total_seats INT NOT NULL DEFAULT 0,
    economy_seats INT NOT NULL DEFAULT 0,
    business_seats INT NOT NULL DEFAULT 0,
    first_class_seats INT NOT NULL DEFAULT 0,
    valid_from TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    valid_to TIMESTAMP WITH TIME ZONE DEFAULT '9999-12-31 23:59:59',
    is_current BOOLEAN DEFAULT TRUE,
    CONSTRAINT dim_aircrafts_pkey PRIMARY KEY (aircraft_sk)
);
CREATE INDEX idx_dim_aircrafts_code ON analytics.dim_aircrafts(aircraft_code);

-- Dim Airports
-- Flattened JSON, split coordinates, added SK
CREATE TABLE IF NOT EXISTS analytics.dim_airports (
    airport_sk BIGSERIAL NOT NULL,
    airport_code CHAR(3) NOT NULL,
    airport_name_en VARCHAR(150) NOT NULL,
    airport_name_ru VARCHAR(150),
    city_en VARCHAR(100) NOT NULL,
    city_ru VARCHAR(100),
    longitude NUMERIC(10, 6) NOT NULL,
    latitude NUMERIC(10, 6) NOT NULL,
    timezone VARCHAR(50) NOT NULL,
    CONSTRAINT dim_airports_pkey PRIMARY KEY (airport_sk)
);
CREATE INDEX idx_dim_airports_code ON analytics.dim_airports(airport_code);

-- Dim Passengers
-- Extracted from tickets, flattened contact_data JSON
CREATE TABLE IF NOT EXISTS analytics.dim_passengers (
    passenger_sk BIGSERIAL NOT NULL,
    passenger_id VARCHAR(20) NOT NULL,
    passenger_name VARCHAR(150) NOT NULL,
    email VARCHAR(150),
    phone VARCHAR(50),
    CONSTRAINT dim_passengers_pkey PRIMARY KEY (passenger_sk)
);
CREATE INDEX idx_dim_passengers_id ON analytics.dim_passengers(passenger_id);

-- ------------------------------------------------------------------------------
-- 2. FACTS
-- ------------------------------------------------------------------------------

-- Fact Flights (Operational Grain)
-- Granularity: One row per flight execution
-- Partitioning: By scheduled departure month
CREATE TABLE IF NOT EXISTS analytics.fact_flights (
    flight_sk BIGSERIAL NOT NULL,
    flight_id INT NOT NULL,
    flight_no CHAR(6) NOT NULL,
    scheduled_departure_date_sk INT NOT NULL,
    scheduled_arrival_date_sk INT NOT NULL,
    departure_airport_sk BIGINT NOT NULL,
    arrival_airport_sk BIGINT NOT NULL,
    aircraft_sk BIGINT NOT NULL,
    scheduled_departure TIMESTAMP WITH TIME ZONE NOT NULL,
    scheduled_arrival TIMESTAMP WITH TIME ZONE NOT NULL,
    actual_departure TIMESTAMP WITH TIME ZONE,
    actual_arrival TIMESTAMP WITH TIME ZONE,
    departure_delay_mins INT, -- Derived: actual_departure - scheduled_departure
    arrival_delay_mins INT,   -- Derived: actual_arrival - scheduled_arrival
    status VARCHAR(20) NOT NULL,
    is_cancelled BOOLEAN NOT NULL DEFAULT FALSE,
    CONSTRAINT fact_flights_pkey PRIMARY KEY (flight_sk, scheduled_departure)
) PARTITION BY RANGE (scheduled_departure);

-- Foreign Keys for Fact Flights
ALTER TABLE analytics.fact_flights ADD CONSTRAINT fk_ff_dep_airport FOREIGN KEY (departure_airport_sk) REFERENCES analytics.dim_airports(airport_sk);
ALTER TABLE analytics.fact_flights ADD CONSTRAINT fk_ff_arr_airport FOREIGN KEY (arrival_airport_sk) REFERENCES analytics.dim_airports(airport_sk);
ALTER TABLE analytics.fact_flights ADD CONSTRAINT fk_ff_aircraft FOREIGN KEY (aircraft_sk) REFERENCES analytics.dim_aircrafts(aircraft_sk);
ALTER TABLE analytics.fact_flights ADD CONSTRAINT fk_ff_dep_date FOREIGN KEY (scheduled_departure_date_sk) REFERENCES analytics.dim_date(date_sk);

-- Clustering/Indexing for Fact Flights
CREATE INDEX idx_ff_dep_airport ON analytics.fact_flights(departure_airport_sk);
CREATE INDEX idx_ff_arr_airport ON analytics.fact_flights(arrival_airport_sk);
CREATE INDEX idx_ff_aircraft ON analytics.fact_flights(aircraft_sk);


-- Fact Ticket Flights (Financial / Passenger Grain)
-- Granularity: One row per passenger per flight segment
-- Consolidates: bookings, tickets, ticket_flights, boarding_passes
-- Partitioning: By booking date
CREATE TABLE IF NOT EXISTS analytics.fact_ticket_flights (
    ticket_flight_sk BIGSERIAL NOT NULL,
    ticket_no CHAR(13) NOT NULL,
    flight_sk BIGINT NOT NULL,
    passenger_sk BIGINT NOT NULL,
    book_date_sk INT NOT NULL,
    book_ref CHAR(6) NOT NULL,
    book_date TIMESTAMP WITH TIME ZONE NOT NULL,
    fare_conditions VARCHAR(10) NOT NULL,
    amount NUMERIC(10, 2) NOT NULL,
    booking_total_amount NUMERIC(10, 2) NOT NULL, -- Denormalized from bookings
    boarding_no INT,                              -- Denormalized from boarding_passes
    seat_no VARCHAR(4),                           -- Denormalized from boarding_passes
    has_boarded BOOLEAN NOT NULL DEFAULT FALSE,   -- Derived: TRUE if boarding_no is not null
    CONSTRAINT fact_ticket_flights_pkey PRIMARY KEY (ticket_flight_sk, book_date)
) PARTITION BY RANGE (book_date);

-- Foreign Keys for Fact Ticket Flights
ALTER TABLE analytics.fact_ticket_flights ADD CONSTRAINT fk_ftf_passenger FOREIGN KEY (passenger_sk) REFERENCES analytics.dim_passengers(passenger_sk);
ALTER TABLE analytics.fact_ticket_flights ADD CONSTRAINT fk_ftf_book_date FOREIGN KEY (book_date_sk) REFERENCES analytics.dim_date(date_sk);
-- Note: flight_sk FK is omitted here to avoid cross-partition FK complexity in standard Postgres, 
-- but logically joins to fact_flights(flight_sk).

-- Clustering/Indexing for Fact Ticket Flights
CREATE INDEX idx_ftf_flight_sk ON analytics.fact_ticket_flights(flight_sk);
CREATE INDEX idx_ftf_passenger_sk ON analytics.fact_ticket_flights(passenger_sk);
CREATE INDEX idx_ftf_fare_conditions ON analytics.fact_ticket_flights(fare_conditions);
