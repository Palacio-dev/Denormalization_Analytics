CREATE TABLE IF NOT EXISTS public.denorm_matricula_aluno
(
    -- Enrollment Details (from matricula_ifb)
    id_matricula serial NOT NULL,
    co_ciclo_matricula character varying(50),
    periodo character varying(30),
    situacao_matricula character varying(50),
    tipo_cota character varying(50),
    atestado_baixarenda character varying(20),
    ano_matricula character varying(20),
    unidade_ensino character varying(100),
    
    -- Student Details (from aluno_ifb)
    id_aluno integer,
    data_nascimento date,
    idade integer,
    sexo character(1),
    nacionalidade character varying(50),
    raca character varying(50),
    portador_deficiencia boolean,
    
    -- Course Details (from curso_ifb)
    co_curso character varying(50),
    no_curso character varying(200),
    co_tipo_curso character varying(10),
    tipo_curso character varying(100),
    co_tipo_nivel character varying(10),
    ds_tipo_nivel character varying(100),
    ds_eixo_tecnologico character varying(200),
    modalidade_ensino character varying(100),
    carga_horaria integer,
    
    -- Origin School Details (from escola & localizacao)
    co_escola_educacenso character varying(50),
    no_escola_educacenso character varying(200),
    tp_dependencia_adm_escola smallint,
    tp_localizacao_escola smallint,
    inse character varying(10),
    escola_uf character varying(2),
    escola_municipio character varying(100),
    
    CONSTRAINT denorm_matricula_aluno_pkey PRIMARY KEY (id_matricula)
);

CREATE INDEX idx_denorm_mat_aluno ON public.denorm_matricula_aluno(id_aluno);

    vl_salario_contratual numeric(10, 2),
    
    -- Job Location Details (from localizacao)
    emprego_uf character varying(2),
    emprego_municipio character varying(100),
    
    CONSTRAINT denorm_aluno_emprego_pkey PRIMARY KEY (id_vinculo)
