-- Dimension Table: DimUser

    user_id INT NOT NULL,
    created_date DATE DEFAULT GETDATE()
);

-- Dimension Table: DimEntitlement
CREATE TABLE DimEntitlement (
    entitlement_sk INT IDENTITY(1,1) PRIMARY KEY,
    entitlement_id INT NOT NULL,
    activeDate DATE,
    inactiveDate DATE,
    useLimit INT,
    useCount INT,
    FeatureTypeCode VARCHAR(4),
    PackageTypeCode VARCHAR(4),
    IsActive BIT,
    created_date DATE DEFAULT GETDATE()
);

-- Fact Table: FactUserEntitlement
CREATE TABLE FactUserEntitlement (
    user_sk INT,
    entitlement_sk INT,
    user_id INT NOT NULL,
    entitlement_id INT NOT NULL,
    activeDate DATE,
    inactiveDate DATE,
    useLimit INT,
    useCount INT,
    FeatureTypeCode VARCHAR(4),
    PackageTypeCode VARCHAR(4),
    IsActive BIT,
    PRIMARY KEY (user_sk, entitlement_sk),
    FOREIGN KEY (user_sk) REFERENCES DimUser(user_sk),
    FOREIGN KEY (entitlement_sk) REFERENCES DimEntitlement(entitlement_sk)
);

-- Index Recommendations
CREATE INDEX IX_FactUserEntitlement_IsActive ON FactUserEntitlement(IsActive);
CREATE INDEX IX_FactUserEntitlement_FeatureType ON FactUserEntitlement(FeatureTypeCode);

### Notes:
- **Surrogate Keys**: Introduced in dimension tables to support SCD and improve join performance.
- **Embedded Dimensions**: Dimensional attributes are copied into the fact table to reduce joins.
- **Indexing**: Supports common analytical filters such as active status and feature/package types.
- **Partitioning**: Recommended on date-based columns for large-scale query optimization.
- **ETL Strategy**: The fact table should be populated via batch ETL that reads from the OLTP source and applies business logic to derive IsActive and other computed fields.
