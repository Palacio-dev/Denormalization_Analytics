### SQL DDL Script for the Denormalized Schema

```sql
-- ==============================================================================
-- DENORMALIZED ANALYTICAL SCHEMA (STAR SCHEMA)
-- Optimized for OLAP / BI Workloads
-- ==============================================================================

BEGIN;

-- ------------------------------------------------------------------------------
-- 1. DIMENSION: dim_tempo (Date Dimension)
-- Supports time-series analysis, drill-downs by year, quarter, month, etc.
-- ------------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.dim_tempo
(
    id_tempo integer NOT NULL, -- Format: YYYYMMDD
    data_completa date NOT NULL,
    ano integer NOT NULL,
    mes integer NOT NULL,
    trimestre integer NOT NULL,
    dia_semana text COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT dim_tempo_pkey PRIMARY KEY (id_tempo)
);

-- ------------------------------------------------------------------------------
-- 2. DIMENSION: dim_geografia (Geography Dimension)
-- Replaces the 'regiao' table.
-- ------------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.dim_geografia
(
    id_municipio integer NOT NULL,
    municipio_nome text COLLATE pg_catalog."default" NOT NULL,
    estado text COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT dim_geografia_pkey PRIMARY KEY (id_municipio)
);

-- ------------------------------------------------------------------------------
-- 3. DIMENSION: dim_instituicao (Organizer Dimension)
-- Replaces 'instituicao_organizadora'.
-- ------------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.dim_instituicao
(
    id_instituicao integer NOT NULL,
    nome text COLLATE pg_catalog."default" NOT NULL,
    carater text COLLATE pg_catalog."default" NOT NULL,
    CONSTRAINT dim_instituicao_pkey PRIMARY KEY (id_instituicao)
);

-- ------------------------------------------------------------------------------
-- 4. DIMENSION / FACTLESS FACT: dim_unidade_conservacao
-- Denormalizes 'unidade_de_conservacao', 'bioma', 'categoria_manejo', 
-- and 'esfera_administrativa' into a single wide table.
-- Contains area measures, allowing it to be queried as a secondary fact table.
-- ------------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.dim_unidade_conservacao
(
    id_uc integer NOT NULL,
    nome_uc text COLLATE pg_catalog."default" NOT NULL,
    ano_de_criacao integer,
    
    -- Measures (Can be aggregated)
    area_soma_biomas double precision,
    area_soma_biomas_continentais double precision,
    area_marinha double precision,
    
    -- Denormalized Lookups
    bioma_nome text COLLATE pg_catalog."default",
    manejo_categoria text COLLATE pg_catalog."default",
    manejo_categoria_iucn text COLLATE pg_catalog."default",
    manejo_plano text COLLATE pg_catalog."default",
    esfera_tipo text COLLATE pg_catalog."default",
    
    -- Foreign Keys
    id_municipio integer,
    
    CONSTRAINT dim_unidade_conservacao_pkey PRIMARY KEY (id_uc),
    CONSTRAINT dim_uc_id_municipio_fkey FOREIGN KEY (id_municipio)
        REFERENCES public.dim_geografia (id_municipio) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
);

-- ------------------------------------------------------------------------------
-- 5. FACT: fact_mutirao_limpeza (Cleaning Efforts Fact Table)
-- Consolidates 'mutirao_de_limpeza', 'coleta_de_lixo', and 'local_de_coleta'.
-- Contains all metrics related to a cleaning event.
-- ------------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS public.fact_mutirao_limpeza
(
    id_mutirao integer NOT NULL,
    
    -- Foreign Keys to Dimensions
    id_tempo integer NOT NULL,
    id_municipio integer,
    id_instituicao integer,
    
    -- Event Attributes (Degenerate Dimensions / Details)
    local_coleta_nome text COLLATE pg_catalog."default",
    horario_de_inicio time without time zone,
    duracao_total time without time zone,
    
    -- Measures
    numero_aproximado_de_participantes integer,
    qtd_plastico integer DEFAULT 0,
    qtd_borracha integer DEFAULT 0,
    qtd_madeira integer DEFAULT 0,
    qtd_produtos_texteis integer DEFAULT 0,
    qtd_papel integer DEFAULT 0,
    qtd_metal integer DEFAULT 0,
    qtd_vidro integer DEFAULT 0,
    qtd_materiais_de_pesca integer DEFAULT 0,
    qtd_outros_residuos integer DEFAULT 0,
    
    CONSTRAINT fact_mutirao_limpeza_pkey PRIMARY KEY (id_mutirao),
    
    CONSTRAINT fact_mutirao_id_tempo_fkey FOREIGN KEY (id_tempo)
        REFERENCES public.dim_tempo (id_tempo) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
        
    CONSTRAINT fact_mutirao_id_municipio_fkey FOREIGN KEY (id_municipio)
        REFERENCES public.dim_geografia (id_municipio) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
        
    CONSTRAINT fact_mutirao_id_instituicao_fkey FOREIGN KEY (id_instituicao)
        REFERENCES public.dim_instituicao (id_instituicao) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
);

-- ------------------------------------------------------------------------------
-- 6. INDEXES FOR ANALYTICAL QUERIES
-- Creating indexes on Foreign Keys to speed up Star Schema Joins.
-- ------------------------------------------------------------------------------
CREATE INDEX IF NOT EXISTS idx_fact_mutirao_tempo 
    ON public.fact_mutirao_limpeza(id_tempo);

CREATE INDEX IF NOT EXISTS idx_fact_mutirao_municipio 
    ON public.fact_mutirao_limpeza(id_municipio);

CREATE INDEX IF NOT EXISTS idx_fact_mutirao_instituicao 
    ON public.fact_mutirao_limpeza(id_instituicao);

CREATE INDEX IF NOT EXISTS idx_dim_uc_municipio 
    ON public.dim_unidade_conservacao(id_municipio);

END;
```
