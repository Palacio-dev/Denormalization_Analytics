-- ==============================================================================
-- Denormalized Schema for Analytics and Reporting

-- Table 1: Yearly Country Metrics
-- Grain: Country + Year
-- Combines: paises, continente, populacao, emissao_g_efeito_estufa
CREATE TABLE IF NOT EXISTS public.denorm_pais_ano_metricas
(
    id_paises integer NOT NULL,
    pais_nome character varying(50) COLLATE pg_catalog."default" NOT NULL,
    pais_fundacao integer NOT NULL,
    id_continente integer NOT NULL,
    continente_nome character varying(50) COLLATE pg_catalog."default" NOT NULL,
    ano integer NOT NULL,
    quantidade_hab integer,
    quantidade_emitida bigint,
    CONSTRAINT denorm_pais_ano_metricas_pkey PRIMARY KEY (id_paises, ano)
);

-- Index to optimize time-series filtering and continent-level aggregations
CREATE INDEX idx_denorm_metricas_ano ON public.denorm_pais_ano_metricas(ano);
CREATE INDEX idx_denorm_metricas_continente ON public.denorm_pais_ano_metricas(continente_nome);


-- Table 2: Yearly Energy Consumption
-- Grain: Country + Year + Energy Source
-- Combines: paises, continente, consumo_anual, fonte_energia
CREATE TABLE IF NOT EXISTS public.denorm_consumo_energia
(
    id_paises integer NOT NULL,
    pais_nome character varying(50) COLLATE pg_catalog."default" NOT NULL,
    id_continente integer NOT NULL,
    continente_nome character varying(50) COLLATE pg_catalog."default" NOT NULL,
    ano integer NOT NULL,
    id_fonte_energia integer NOT NULL,
    fonte_energia_nome character varying(50)[] COLLATE pg_catalog."default" NOT NULL,
    porcentagem integer NOT NULL,
    CONSTRAINT denorm_consumo_energia_pkey PRIMARY KEY (id_paises, ano, id_fonte_energia)
);

-- Index to optimize filtering by energy source and time-series analysis
CREATE INDEX idx_denorm_consumo_ano ON public.denorm_consumo_energia(ano);
CREATE INDEX idx_denorm_consumo_continente ON public.denorm_consumo_energia(continente_nome);
