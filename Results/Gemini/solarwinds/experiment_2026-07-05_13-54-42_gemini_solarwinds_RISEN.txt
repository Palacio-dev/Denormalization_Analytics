-- Dimension Table: Customers
-- Preserves all customer records, including those without any associated orders.
CREATE TABLE dim_customers (
  customer_id SERIAL PRIMARY KEY,
  customer_name VARCHAR(100),
  customer_email VARCHAR(100),
  customer_address TEXT
);

-- Dimension Table: Products
-- Preserves all product records, including those that have never been sold.
CREATE TABLE dim_products (
  product_id SERIAL PRIMARY KEY,
  product_name VARCHAR(100),
  price DECIMAL(10,2)
);

-- Fact Table: Sales
-- Collapses 'orders' and 'order_details', adds redundant attributes for zero-join querying, 
-- and includes derived metrics for fast aggregations.
CREATE TABLE fact_sales (
  order_detail_id SERIAL PRIMARY KEY,
  order_id INT NOT NULL,
  order_date DATE,
  
  -- Foreign keys maintained for reconstructibility and optional dimension joins
  customer_id INT REFERENCES dim_customers(customer_id),
  product_id INT REFERENCES dim_products(product_id),
  
  -- Redundant Attributes (Technique 1)
  customer_name VARCHAR(100),
  product_name VARCHAR(100),
  product_price DECIMAL(10,2),
  
  -- Original Attributes
  quantity INT,
  
  -- Derived Attributes (Technique 2)
  line_total_amount DECIMAL(12,2)
);
