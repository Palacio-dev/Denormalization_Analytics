BEGIN;

-- Create a schema for the analytical models
CREATE SCHEMA IF NOT EXISTS analytics;

-- ==============================================================================
-- Table 1: Current Employee Snapshot
-- Purpose: 0-join table for all current state reporting (active employees only)
-- ==============================================================================
CREATE TABLE IF NOT EXISTS analytics.denorm_employee_current
(
    employee_id bigint NOT NULL,
    first_name character varying(14) COLLATE pg_catalog."default" NOT NULL,
    last_name character varying(16) COLLATE pg_catalog."default" NOT NULL,
    birth_date date NOT NULL,
    gender employees.employee_gender NOT NULL,
    hire_date date NOT NULL,
    
    -- Current Department Info
    department_id character(4) COLLATE pg_catalog."default",
    dept_name character varying(40) COLLATE pg_catalog."default",
    
    -- Current Role & Compensation
    title character varying(50) COLLATE pg_catalog."default",
    salary bigint,
    is_manager boolean NOT NULL DEFAULT false,
    
    CONSTRAINT pk_denorm_employee_current PRIMARY KEY (employee_id)
);

-- Indexes for common current-state analytical slices
CREATE INDEX IF NOT EXISTS idx_curr_dept ON analytics.denorm_employee_current(dept_name);
CREATE INDEX IF NOT EXISTS idx_curr_title ON analytics.denorm_employee_current(title);
CREATE INDEX IF NOT EXISTS idx_curr_manager ON analytics.denorm_employee_current(is_manager);


-- ==============================================================================
-- Table 2: Employee History (One Big Table)
-- Purpose: 0-join table for historical reporting, point-in-time analysis, 
--          and tracking changes over time (SCD Type 2).
-- ==============================================================================
CREATE TABLE IF NOT EXISTS analytics.denorm_employee_history
(
    history_id bigserial NOT NULL,
    employee_id bigint NOT NULL,
    first_name character varying(14) COLLATE pg_catalog."default" NOT NULL,
    last_name character varying(16) COLLATE pg_catalog."default" NOT NULL,
    birth_date date NOT NULL,
    gender employees.employee_gender NOT NULL,
    hire_date date NOT NULL,
    
    -- Historical Department Info
    department_id character(4) COLLATE pg_catalog."default",
    dept_name character varying(40) COLLATE pg_catalog."default",
    
    -- Historical Role & Compensation
    title character varying(50) COLLATE pg_catalog."default",
    salary bigint,
    is_manager boolean NOT NULL DEFAULT false,
    
    -- Unified Contiguous Time Slices
    effective_from date NOT NULL,
    effective_to date NOT NULL,
    
    CONSTRAINT pk_denorm_employee_history PRIMARY KEY (history_id)
);

-- Indexes for historical lookups and time-series aggregations
CREATE INDEX IF NOT EXISTS idx_hist_emp_id ON analytics.denorm_employee_history(employee_id);
CREATE INDEX IF NOT EXISTS idx_hist_dept ON analytics.denorm_employee_history(dept_name);
CREATE INDEX IF NOT EXISTS idx_hist_title ON analytics.denorm_employee_history(title);
CREATE INDEX IF NOT EXISTS idx_hist_dates ON analytics.denorm_employee_history(effective_from, effective_to);

END;
