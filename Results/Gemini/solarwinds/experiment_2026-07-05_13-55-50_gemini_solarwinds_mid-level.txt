-- ==============================================================================
-- Analytical Data Model (Star Schema)
-- Optimized for BI Dashboards, Reporting, and Aggregations
-- ==============================================================================

-- 1. Date Dimension
-- Enables robust time-series analysis, filtering, and grouping.
CREATE TABLE dim_date (
    date_id INT PRIMARY KEY, -- Format: YYYYMMDD
    full_date DATE NOT NULL,
    calendar_year INT NOT NULL,
    calendar_quarter INT NOT NULL,
    calendar_month INT NOT NULL,
    month_name VARCHAR(20) NOT NULL,
    day_of_month INT NOT NULL,
    day_of_week INT NOT NULL,
    day_name VARCHAR(20) NOT NULL,
    is_weekend BOOLEAN NOT NULL
);

-- 2. Customer Dimension (SCD Type 2)
-- Tracks customer information and history (e.g., if they move to a new address).
CREATE TABLE dim_customers (
    customer_sk SERIAL PRIMARY KEY,       -- Surrogate Key
    customer_id INT NOT NULL,             -- Natural Key from source
    customer_name VARCHAR(100),
    customer_email VARCHAR(100),
    customer_address TEXT,
    valid_from DATE NOT NULL,             -- SCD2 Start Date
    valid_to DATE,                        -- SCD2 End Date
    is_current BOOLEAN NOT NULL           -- SCD2 Current Flag
);

-- Create index on natural key for faster ETL lookups
CREATE INDEX idx_dim_customers_nk ON dim_customers(customer_id);

-- 3. Product Dimension (SCD Type 2)
-- Tracks product attributes and current pricing.
CREATE TABLE dim_products (
    product_sk SERIAL PRIMARY KEY,        -- Surrogate Key
    product_id INT NOT NULL,              -- Natural Key from source
    product_name VARCHAR(100),
    current_price DECIMAL(10,2),
    valid_from DATE NOT NULL,             -- SCD2 Start Date
    valid_to DATE,                        -- SCD2 End Date
    is_current BOOLEAN NOT NULL           -- SCD2 Current Flag
);

-- Create index on natural key for faster ETL lookups
CREATE INDEX idx_dim_products_nk ON dim_products(product_id);

-- 4. Sales Fact Table
-- Grain: One row per order line item (order_detail).
-- Consolidates 'orders' and 'order_details' to reduce joins.
CREATE TABLE fact_sales (
    sales_sk SERIAL PRIMARY KEY,          -- Surrogate Key for the fact record
    
    -- Foreign Keys to Dimensions
    date_id INT NOT NULL REFERENCES dim_date(date_id),
    customer_sk INT NOT NULL REFERENCES dim_customers(customer_sk),
    product_sk INT NOT NULL REFERENCES dim_products(product_sk),
    
    -- Degenerate Dimensions (kept for audit/drill-through to source)
    order_id INT NOT NULL,
    order_detail_id INT NOT NULL,
    
    -- Measures
    quantity INT NOT NULL,
    unit_price DECIMAL(10,2) NOT NULL,    -- Historical price at the time of purchase
    total_amount DECIMAL(10,2) NOT NULL   -- Pre-calculated: quantity * unit_price
);

-- Create indexes on foreign keys to optimize Star Schema joins
CREATE INDEX idx_fact_sales_date ON fact_sales(date_id);
CREATE INDEX idx_fact_sales_customer ON fact_sales(customer_sk);
CREATE INDEX idx_fact_sales_product ON fact_sales(product_sk);

-- Optional: Create an OBT (One Big Table) View for BI tools that prefer flat structures
CREATE VIEW vw_obt_sales AS
SELECT 
    f.order_id,
    f.order_detail_id,
    d.full_date AS order_date,
    d.calendar_year,
    d.calendar_quarter,
    d.calendar_month,
    c.customer_id,
    c.customer_name,
    c.customer_email,
    c.customer_address,
    p.product_id,
    p.product_name,
    f.quantity,
    f.unit_price,
    f.total_amount
FROM fact_sales f
JOIN dim_date d ON f.date_id = d.date_id
JOIN dim_customers c ON f.customer_sk = c.customer_sk
JOIN dim_products p ON f.product_sk = p.product_sk;
