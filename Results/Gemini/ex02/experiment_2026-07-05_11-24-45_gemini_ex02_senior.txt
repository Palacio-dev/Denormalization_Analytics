-- ==============================================================================
-- ANALYTICAL PHYSICAL MODEL (OLAP STAR SCHEMA)

-- 1. DIMENSION: DATE (dim_data)
-- Supports time-series analysis, seasonality, and historical tracking.
CREATE TABLE IF NOT EXISTS public.dim_data
(
    sk_data integer NOT NULL, -- Format: YYYYMMDD
    data_completa date NOT NULL,
    ano integer NOT NULL,
    mes integer NOT NULL,
    nome_mes text NOT NULL,
    trimestre integer NOT NULL,
    dia_da_semana integer NOT NULL,
    nome_dia_semana text NOT NULL,
    fim_de_semana boolean NOT NULL,
    CONSTRAINT dim_data_pkey PRIMARY KEY (sk_data)
);

-- 2. DIMENSION: GEOGRAPHY (dim_geografia)
-- Converted from 'regiao'.
CREATE TABLE IF NOT EXISTS public.dim_geografia
(
    sk_geografia integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    id_municipio integer NOT NULL, -- Natural Key
    nome_municipio text NOT NULL,
    estado text NOT NULL,
    CONSTRAINT dim_geografia_pkey PRIMARY KEY (sk_geografia)
);

-- 3. DIMENSION: INSTITUTION (dim_instituicao)
-- Converted from 'instituicao_organizadora'.
CREATE TABLE IF NOT EXISTS public.dim_instituicao
(
    sk_instituicao integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    id_instituicao_organizadora integer NOT NULL, -- Natural Key
    nome_instituicao text NOT NULL,
    carater_instituicao text NOT NULL,
    CONSTRAINT dim_instituicao_pkey PRIMARY KEY (sk_instituicao)
);

-- 4. DIMENSION: CONSERVATION UNIT (dim_unidade_conservacao)
-- Flattened dimension absorbing bioma, categoria_manejo, and esfera_administrativa.
-- Note: Contains area metrics. In some models this could be a fact, but here it acts as a rich entity profile.
CREATE TABLE IF NOT EXISTS public.dim_unidade_conservacao
(
    sk_uc integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    id_uc integer NOT NULL, -- Natural Key
    nome_uc text NOT NULL,
    ano_de_criacao integer,
    area_soma_biomas double precision,
    area_soma_biomas_continentais double precision,
    area_marinha double precision,
    -- Absorbed from bioma
    nome_bioma text,
    -- Absorbed from categoria_manejo
    categoria_manejo text,
    categoria_iucn text,
    plano_manejo text,
    -- Absorbed from esfera_administrativa
    esfera_administrativa text,
    -- Foreign Key to Geography
    sk_geografia integer,
    CONSTRAINT dim_unidade_conservacao_pkey PRIMARY KEY (sk_uc),
    CONSTRAINT fk_dim_uc_geografia FOREIGN KEY (sk_geografia) REFERENCES public.dim_geografia (sk_geografia)
);

-- 5. FACT: CLEANUP EVENTS & WASTE COLLECTION (fact_mutirao_coleta)
-- Merges mutirao_de_limpeza, coleta_de_lixo, and local_de_coleta.
CREATE TABLE IF NOT EXISTS public.fact_mutirao_coleta
(
    sk_mutirao integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    id_mutirao integer NOT NULL, -- Natural Key / Degenerate Dimension
    
    -- Surrogate Foreign Keys
    sk_data integer NOT NULL,
    sk_geografia integer NOT NULL,
    sk_instituicao integer NOT NULL,
    
    -- Degenerate Dimensions (Event specific details)
    nome_local_coleta text,
    horario_de_inicio time without time zone,
    duracao_total time without time zone,
    
    -- Metrics (Facts)
    numero_aproximado_de_participantes integer,
    plastico integer DEFAULT 0,
    borracha integer DEFAULT 0,
    madeira integer DEFAULT 0,
    produtos_texteis integer DEFAULT 0,
    papel integer DEFAULT 0,
    metal integer DEFAULT 0,
    vidro integer DEFAULT 0,
    materiais_de_pesca integer DEFAULT 0,
    outros_residuos integer DEFAULT 0,
    
    -- Derived Metric (Calculated during ETL)
    total_residuos_coletados integer DEFAULT 0,
    
    CONSTRAINT fact_mutirao_coleta_pkey PRIMARY KEY (sk_mutirao),
    CONSTRAINT fk_fact_data FOREIGN KEY (sk_data) REFERENCES public.dim_data (sk_data),
    CONSTRAINT fk_fact_geografia FOREIGN KEY (sk_geografia) REFERENCES public.dim_geografia (sk_geografia),
    CONSTRAINT fk_fact_instituicao FOREIGN KEY (sk_instituicao) REFERENCES public.dim_instituicao (sk_instituicao)
) PARTITION BY RANGE (sk_data); 
-- Partitioning recommendation: Partitioning by date (sk_data) improves query performance 
-- for time-based filtering and makes archiving/dropping old data easier.
