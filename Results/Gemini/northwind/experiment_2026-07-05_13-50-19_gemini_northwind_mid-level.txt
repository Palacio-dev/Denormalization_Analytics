    CONSTRAINT pk_dim_date PRIMARY KEY (date_id)
);

CREATE TABLE IF NOT EXISTS public.dim_customers
(
    customer_id character varying(5) NOT NULL,
    company_name character varying(40) NOT NULL,
    contact_name character varying(30),
    contact_title character varying(30),
    address character varying(60),
    city character varying(15),
    region character varying(15),
    postal_code character varying(10),
    country character varying(15),
    phone character varying(24),
    fax character varying(24),
    customer_demographics text, -- Flattened comma-separated list of demographics
    CONSTRAINT pk_dim_customers PRIMARY KEY (customer_id)
);

CREATE TABLE IF NOT EXISTS public.dim_employees
(
    employee_id smallint NOT NULL,
    full_name character varying(35) NOT NULL,
    title character varying(30),
    title_of_courtesy character varying(25),
    birth_date date,
    hire_date date,
    address character varying(60),
    city character varying(15),
    region character varying(15),
    postal_code character varying(10),
    country character varying(15),
    manager_name character varying(35), -- Flattened from reports_to
    territories text, -- Flattened comma-separated list of territories
    regions text, -- Flattened comma-separated list of regions
    CONSTRAINT pk_dim_employees PRIMARY KEY (employee_id)
);

CREATE TABLE IF NOT EXISTS public.dim_products
(
    product_id smallint NOT NULL,
    product_name character varying(40) NOT NULL,
    quantity_per_unit character varying(20),
    unit_price real,
    discontinued integer NOT NULL,
    -- Category attributes
    category_name character varying(15),
    category_description text,
    -- Supplier attributes
    supplier_name character varying(40),
    supplier_country character varying(15),
    supplier_region character varying(15),
    CONSTRAINT pk_dim_products PRIMARY KEY (product_id)
);

CREATE TABLE IF NOT EXISTS public.dim_shippers
(
    shipper_id smallint NOT NULL,
    company_name character varying(40) NOT NULL,
    CONSTRAINT pk_dim_shippers PRIMARY KEY (shipper_id)
);

-- ==========================================
-- FACT TABLE
-- ==========================================

CREATE TABLE IF NOT EXISTS public.fact_sales
(
    -- Degenerate Dimensions / Identifiers
    order_id smallint NOT NULL,
    product_id smallint NOT NULL,
    
    -- Foreign Keys to Dimensions
    customer_id character varying(5),
    employee_id smallint,
    shipper_id smallint,
    
    -- Role-playing Date Dimensions (Foreign Keys to dim_date)
    order_date_id integer,
    required_date_id integer,
    shipped_date_id integer,
    
    -- Order Header Attributes (Shipment info specific to the order)
    ship_name character varying(40),
    ship_address character varying(60),
    ship_city character varying(15),
    ship_region character varying(15),
    ship_postal_code character varying(10),
    ship_country character varying(15),
    
    -- Measures (Line Item Grain)
    quantity smallint NOT NULL,
    unit_price real NOT NULL,
    discount real NOT NULL,
    sales_amount real NOT NULL, -- Derived: quantity * unit_price * (1 - discount)
    
    -- Measures (Header Grain - Allocated or Repeated)
    order_freight real, 
    
    CONSTRAINT pk_fact_sales PRIMARY KEY (order_id, product_id),
    CONSTRAINT fk_fact_sales_customers FOREIGN KEY (customer_id) REFERENCES public.dim_customers (customer_id),
    CONSTRAINT fk_fact_sales_employees FOREIGN KEY (employee_id) REFERENCES public.dim_employees (employee_id),
    CONSTRAINT fk_fact_sales_products FOREIGN KEY (product_id) REFERENCES public.dim_products (product_id),
    CONSTRAINT fk_fact_sales_shippers FOREIGN KEY (shipper_id) REFERENCES public.dim_shippers (shipper_id),
    CONSTRAINT fk_fact_sales_order_date FOREIGN KEY (order_date_id) REFERENCES public.dim_date (date_id),
    CONSTRAINT fk_fact_sales_required_date FOREIGN KEY (required_date_id) REFERENCES public.dim_date (date_id),
    CONSTRAINT fk_fact_sales_shipped_date FOREIGN KEY (shipped_date_id) REFERENCES public.dim_date (date_id)
);
