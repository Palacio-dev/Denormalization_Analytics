-- Denormalized Analytical Schema DDL

-- Dimension: Time
CREATE TABLE IF NOT EXISTS public.dim_time (
    sk_time SERIAL PRIMARY KEY,
    year INTEGER NOT NULL UNIQUE
);

-- Dimension: Continent
CREATE TABLE IF NOT EXISTS public.dim_continent (
    sk_continent SERIAL PRIMARY KEY,
    continent_id INTEGER NOT NULL UNIQUE,
    continent_name VARCHAR(50) NOT NULL
);

-- Dimension: Country
CREATE TABLE IF NOT EXISTS public.dim_country (
    sk_country SERIAL PRIMARY KEY,
    country_id INTEGER NOT NULL UNIQUE,
    country_name VARCHAR(50) NOT NULL,
    foundation_year INTEGER,
    continent_id INTEGER,
    continent_name VARCHAR(50),
    FOREIGN KEY (continent_id) REFERENCES public.dim_continent(continent_id)
);

-- Dimension: Energy Source
CREATE TABLE IF NOT EXISTS public.dim_energy_source (
    sk_energy_source SERIAL PRIMARY KEY,
    energy_source_id INTEGER NOT NULL UNIQUE,
    energy_source_name VARCHAR(50) NOT NULL
);

-- Fact Table: Energy Consumption and Emission
CREATE TABLE IF NOT EXISTS public.fact_energy_consumption_emission (
    sk_fact SERIAL PRIMARY KEY,
    sk_country INTEGER NOT NULL,
    sk_time INTEGER NOT NULL,
    sk_energy_source INTEGER,
    energy_percentage INTEGER,
    emission_quantity BIGINT,
    population_quantity INTEGER,
    per_capita_emission DECIMAL(18, 6),
    FOREIGN KEY (sk_country) REFERENCES public.dim_country(sk_country),
    FOREIGN KEY (sk_time) REFERENCES public.dim_time(sk_time),
    FOREIGN KEY (sk_energy_source) REFERENCES public.dim_energy_source(sk_energy_source)
);

-- Indexes for performance
CREATE INDEX idx_fact_sk_country ON public.fact_energy_consumption_emission(sk_country);
CREATE INDEX idx_fact_sk_time ON public.fact_energy_consumption_emission(sk_time);
CREATE INDEX idx_fact_sk_energy_source ON public.fact_energy_consumption_emission(sk_energy_source);
