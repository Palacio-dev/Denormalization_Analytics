### Complete SQL DDL Creation Script for the Denormalized Schema

```sql
-- Drop existing denormalized tables if they exist (for idempotency)
DROP TABLE IF EXISTS public.unidade_de_conservacao_analytics;
DROP TABLE IF EXISTS public.coleta_de_lixo_analytics;
DROP TABLE IF EXISTS public.conservation_event_master;

-- ============================================================================
-- Table 1: Conservation Event Master
-- Purpose: Consolidates UCI details with Event details to avoid JOINs for basic reporting.
-- ============================================================================
CREATE TABLE IF NOT EXISTS public.unidade_de_conservacao_analytics
(
    id_uc integer NOT NULL,
    -- Event Details (Denormalized from mutirao_de_limpeza)
    data_evento date NOT NULL,
    horario_inicio time without time zone NOT NULL,
    duracao_total time without time zone NOT NULL,
    nome_bioma text NOT NULL,
    categoria_texto text NOT NULL,
    plano_texto text NOT NULL,
    nome_instituicao text NOT NULL,
    carater_instituicao text NOT NULL,
    nome_regiao text NOT NULL,
    estado_regiao text NOT NULL,
    tipo_esfera text NOT NULL,
    nome_local text NOT NULL,
    CONSTRAINT unidade_de_conservacao_analytics_pkey PRIMARY KEY (id_uc),
    CONSTRAINT unidade_de_conservacao_analytics_data_check CHECK (data_evento >= '1900-01-01'::date)
);

-- ============================================================================
-- Table 2: Waste Collection Analytics
-- Purpose: Stores waste volumes per event to avoid JOINs between mutirao and coleta tables.
-- ============================================================================
CREATE TABLE IF NOT EXISTS public.coleta_de_lixo_analytics
(
    id_evento integer NOT NULL, -- Primary key to link to conservation_event_master
    volume_total integer DEFAULT 0,
    volume_plastico integer DEFAULT 0,
    volume_borracha integer DEFAULT 0,
    volume_madeira integer DEFAULT 0,
    volume_produtos_texteis integer DEFAULT 0,
    volume_papel integer DEFAULT 0,
    volume_metal integer DEFAULT 0,
    volume_vidro integer DEFAULT 0,
    volume_materiais_de_pesca integer DEFAULT 0,
    volume_outros_residuos integer DEFAULT 0,
    CONSTRAINT coleta_de_lixo_analytics_pkey PRIMARY KEY (id_evento)
);

-- ============================================================================
-- Table 3: Conservation Event Master (Linking Table)
-- Purpose: Links the UCI Master with the Waste Analytics using the event ID.
-- ============================================================================
CREATE TABLE IF NOT EXISTS public.conservation_event_master
(
    id_uc integer NOT NULL,
    id_evento integer NOT NULL,
    -- Redundant columns for quick filtering without joining
    nome_bioma text,
    categoria_texto text,
    CONSTRAINT conservation_event_master_pkey PRIMARY KEY (id_uc, id_evento),
    CONSTRAINT conservation_event_master_uc_fkey FOREIGN KEY (id_uc)
        REFERENCES public.unidade_de_conservacao_analytics (id_uc) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE,
    CONSTRAINT conservation_event_master_event_fkey FOREIGN KEY (id_evento)
        REFERENCES public.coleta_de_lixo_analytics (id_evento) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE CASCADE
);

-- ============================================================================
-- Populate Data (Example Logic using INSERT ... SELECT)
-- Note: In a real environment, this would be run as part of an ETL process
-- to keep the denormalized tables synchronized with the source.
-- ============================================================================

-- Populate Conservation Event Master
INSERT INTO public.unidade_de_conservacao_analytics (
    id_uc,
    data_evento,
    horario_inicio,
    duracao_total,
    nome_bioma,
    categoria_texto,
    plano_texto,
    nome_instituicao,
    carater_instituicao,
    nome_regiao,
    estado_regiao,
    tipo_esfera,
    nome_local
)
SELECT 
    u.id_uc,
    m.data_do_ocorrido,
    m.horario_de_inicio,
    m.duracao_total,
    b.nome AS nome_bioma,
    c.categoria AS categoria_texto,
    c.plano,
    i.nome AS nome_instituicao,
    i.carater,
    r.nome AS nome_regiao,
    r.estado,
    e.tipo,
    l.nome AS nome_local
FROM public.unidade_de_conservacao u
JOIN public.bioma b ON u.id_bioma = b.id_bioma
JOIN public.categoria_manejo c ON u.id_manejo = c.id_manejo
JOIN public.esfera_administrativa e ON u.id_esfera = e.id_esfera
JOIN public.instituicao_organizadora i ON u.id_instituicao_organizadora = i.id_instituicao_organizadora
JOIN public.regiao r ON u.id_municipio = r.id_municipio
JOIN public.local_de_coleta l ON u.id_municipio = l.id_mutirao
JOIN public.mutirao_de_limpeza m ON u.id_municipio = m.id_municipio;

-- Populate Waste Collection Analytics
INSERT INTO public.coleta_de_lixo_analytics (
    id_evento,
    volume_total,
    volume_plastico,
    volume_borracha,
    volume_madeira,
    volume_produtos_texteis,
    volume_papel,
    volume_metal,
    volume_vidro,
    volume_materiais_de_pesca,
    volume_outros_residuos
)
SELECT 
    m.id_mutirao, -- Using mutirao ID as the event ID for simplicity in this schema
    COALESCE(SUM(c.volume_total), 0) AS volume_total,
    COALESCE(SUM(c.volume_plastico), 0) AS volume_plastico,
    COALESCE(SUM(c.volume_borracha), 0) AS volume_borracha,
    COALESCE(SUM(c.volume_madeira), 0) AS volume_madeira,
    COALESCE(SUM(c.volume_produtos_texteis), 0) AS volume_produtos_texteis,
    COALESCE(SUM(c.volume_papel), 0) AS volume_papel,
    COALESCE(SUM(c.volume_metal), 0) AS volume_metal,
    COALESCE(SUM(c.volume_vidro), 0) AS volume_vidro,
    COALESCE(SUM(c.volume_materiais_de_pesca), 0) AS volume_materiais_de_pesca,
    COALESCE(SUM(c.volume_outros_residuos), 0) AS volume_outros_residuos
FROM public.mutirao_de_limpeza m
LEFT JOIN public.coleta_de_lixo c ON m.id_mutirao = c.id_mutirao
GROUP BY m.id_mutirao;

-- Create indexes for faster analytics queries
CREATE INDEX IF NOT EXISTS idx_uc_bioma ON public.unidade_de_conservacao_analytics(nome_bioma);
CREATE INDEX IF NOT EXISTS idx_evento_date ON public.coleta_de_lixo_analytics(data_evento);
CREATE INDEX IF NOT EXISTS idx_evento_uc ON public.conservation_event_master(id_uc);
```
