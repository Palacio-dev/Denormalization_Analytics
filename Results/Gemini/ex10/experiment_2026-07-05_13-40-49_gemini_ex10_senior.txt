### Analytical Physical Model Design (SQL DDL)

The following PostgreSQL script creates the optimized Star Schema. It includes surrogate keys, standardizes data types for BI compatibility, and implements table partitioning for the fact tables to support large-scale analytical queries.

```sql
-- ============================================================================
-- DIMENSIONAL MODEL: STAR SCHEMA
-- ============================================================================

-- 1. DIMENSION: Geography (Flattened from paises and continente)
-- Includes Surrogate Key (geo_sk) and preserves Natural Keys for traceability.
CREATE TABLE IF NOT EXISTS public.dim_geography
(
    geo_sk SERIAL NOT NULL,
    country_id integer NOT NULL,         -- Source system ID (Natural Key)
    country_name character varying(50) NOT NULL,
    foundation_year integer,
    continent_id integer NOT NULL,       -- Source system ID
    continent_name character varying(50) NOT NULL,
    CONSTRAINT dim_geography_pkey PRIMARY KEY (geo_sk)
);

-- Index for ETL lookups
CREATE UNIQUE INDEX idx_dim_geography_nk ON public.dim_geography(country_id);
-- Index for BI filtering
CREATE INDEX idx_dim_geography_continent ON public.dim_geography(continent_name);


-- 2. DIMENSION: Energy Source (Flattened from fonte_energia)
-- Note: Array type removed in favor of standard scalar text for BI GROUP BY compatibility.
CREATE TABLE IF NOT EXISTS public.dim_energy
(
    energy_sk SERIAL NOT NULL,
    energy_source_id integer NOT NULL,   -- Source system ID (Natural Key)
    energy_source_name character varying(100) NOT NULL,
    CONSTRAINT dim_energy_pkey PRIMARY KEY (energy_sk)
);

-- Index for ETL lookups
CREATE UNIQUE INDEX idx_dim_energy_nk ON public.dim_energy(energy_source_id);


-- ============================================================================
-- FACT TABLES
-- ============================================================================

-- 3. FACT: Country Yearly Metrics (Consolidated from populacao & emissao_g_efeito_estufa)
-- Grain: 1 row per Country per Year
-- Partitioned by Year to optimize historical trend analysis and data lifecycle management.
CREATE TABLE IF NOT EXISTS public.fact_country_yearly_metrics
(
    metric_sk bigserial NOT NULL,
    geo_sk integer NOT NULL,
    year integer NOT NULL,
    population_count bigint,             -- Upgraded to bigint for future-proofing
    emission_amount bigint,
    emissions_per_capita numeric(15,4),  -- Derived metric: emission_amount / population_count
    CONSTRAINT fact_country_yearly_metrics_pkey PRIMARY KEY (metric_sk, year)
) PARTITION BY RANGE (year);

-- Foreign Key Constraints (Often disabled or managed logically in massive OLAP environments, 
-- but included here for strict relational integrity).
ALTER TABLE public.fact_country_yearly_metrics
    ADD CONSTRAINT fk_fact_metrics_geo FOREIGN KEY (geo_sk)
    REFERENCES public.dim_geography (geo_sk);

-- Clustering / Indexing recommendations
CREATE INDEX idx_fact_metrics_geo ON public.fact_country_yearly_metrics(geo_sk);
CREATE INDEX idx_fact_metrics_year ON public.fact_country_yearly_metrics(year);

-- Example Partitions for fact_country_yearly_metrics
CREATE TABLE fact_country_yearly_metrics_pre_2000 PARTITION OF public.fact_country_yearly_metrics FOR VALUES FROM (MINVALUE) TO (2000);
CREATE TABLE fact_country_yearly_metrics_2000_2010 PARTITION OF public.fact_country_yearly_metrics FOR VALUES FROM (2000) TO (2010);
CREATE TABLE fact_country_yearly_metrics_2010_2020 PARTITION OF public.fact_country_yearly_metrics FOR VALUES FROM (2010) TO (2020);
CREATE TABLE fact_country_yearly_metrics_2020_2030 PARTITION OF public.fact_country_yearly_metrics FOR VALUES FROM (2020) TO (2030);


-- 4. FACT: Energy Consumption (Transformed from consumo_anual)
-- Grain: 1 row per Country per Year per Energy Source
-- Partitioned by Year.
CREATE TABLE IF NOT EXISTS public.fact_energy_consumption
(
    consumption_sk bigserial NOT NULL,
    geo_sk integer NOT NULL,
    energy_sk integer NOT NULL,
    year integer NOT NULL,
    consumption_percentage numeric(5,2) NOT NULL, -- Changed from integer to numeric for precision
    CONSTRAINT fact_energy_consumption_pkey PRIMARY KEY (consumption_sk, year)
) PARTITION BY RANGE (year);

-- Foreign Key Constraints
ALTER TABLE public.fact_energy_consumption
    ADD CONSTRAINT fk_fact_energy_geo FOREIGN KEY (geo_sk)
    REFERENCES public.dim_geography (geo_sk);

ALTER TABLE public.fact_energy_consumption
    ADD CONSTRAINT fk_fact_energy_source FOREIGN KEY (energy_sk)
    REFERENCES public.dim_energy (energy_sk);

-- Clustering / Indexing recommendations
CREATE INDEX idx_fact_energy_geo ON public.fact_energy_consumption(geo_sk);
CREATE INDEX idx_fact_energy_energy ON public.fact_energy_consumption(energy_sk);
CREATE INDEX idx_fact_energy_year ON public.fact_energy_consumption(year);

-- Example Partitions for fact_energy_consumption
CREATE TABLE fact_energy_consumption_pre_2000 PARTITION OF public.fact_energy_consumption FOR VALUES FROM (MINVALUE) TO (2000);
CREATE TABLE fact_energy_consumption_2000_2010 PARTITION OF public.fact_energy_consumption FOR VALUES FROM (2000) TO (2010);
CREATE TABLE fact_energy_consumption_2010_2020 PARTITION OF public.fact_energy_consumption FOR VALUES FROM (2010) TO (2020);
CREATE TABLE fact_energy_consumption_2020_2030 PARTITION OF public.fact_energy_consumption FOR VALUES FROM (2020) TO (2030);
```
