### Analytical Schema DDL (PostgreSQL)

```sql
-- ==============================================================================
-- ANALYTICAL SCHEMA (STAR SCHEMA) FOR HERBARIUM BI & REPORTING
-- ==============================================================================

CREATE SCHEMA IF NOT EXISTS analytical;

-- ------------------------------------------------------------------------------
-- 1. DIMENSION TABLES
-- ------------------------------------------------------------------------------

-- Date Dimension for time-series analysis
CREATE TABLE IF NOT EXISTS analytical.dim_tempo (
    date_key INT PRIMARY KEY, -- Format YYYYMMDD
    data_completa DATE NOT NULL,
    ano INT NOT NULL,
    mes INT NOT NULL,
    dia INT NOT NULL,
    trimestre INT NOT NULL,
    semestre INT NOT NULL,
    nome_mes VARCHAR(20) NOT NULL,
    dia_semana VARCHAR(20) NOT NULL,
    is_fim_semana BOOLEAN NOT NULL
);

-- Flattened Taxonomy Dimension
CREATE TABLE IF NOT EXISTS analytical.dim_taxonomia (
    taxonomia_key BIGSERIAL PRIMARY KEY,
    reino_nome VARCHAR(200),
    familia_nome VARCHAR(200),
    sub_familia_nome VARCHAR(300),
    sub_familia_autor VARCHAR(200),
    genero_nome VARCHAR(200),
    especie_nome VARCHAR(200),
    especie_autor VARCHAR(200),
    sub_especie_nome VARCHAR(255),
    sub_especie_autor VARCHAR(200),
    variedade_nome VARCHAR(200),
    variedade_autor VARCHAR(200),
    nome_cientifico_completo TEXT, -- Pre-concatenated full scientific name
    tipo_taxon VARCHAR(250) -- From public.tipos
);

-- Flattened Geography and Environment Dimension
CREATE TABLE IF NOT EXISTS analytical.dim_geografia_local (
    geografia_key BIGSERIAL PRIMARY KEY,
    pais_nome VARCHAR(255),
    pais_sigla CHAR(4),
    estado_nome VARCHAR(255),
    estado_sigla CHAR(4),
    cidade_nome VARCHAR(255), -- Denormalized from implied cities table
    local_descricao TEXT,
    fase_sucessional_nome VARCHAR(200),
    solo_nome VARCHAR(300),
    relevo_nome VARCHAR(300),
    vegetacao_nome VARCHAR(300)
);

-- Flattened Collector Dimension
CREATE TABLE IF NOT EXISTS analytical.dim_coletor (
    coletor_key BIGSERIAL PRIMARY KEY,
    coletor_principal_nome VARCHAR(255),
    coletor_principal_email VARCHAR(200),
    coletores_complementares TEXT -- Aggregated from public.coletores_complementares
);

-- Flattened Herbarium/Institution Dimension
CREATE TABLE IF NOT EXISTS analytical.dim_herbario (
    herbario_key BIGSERIAL PRIMARY KEY,
    herbario_nome VARCHAR(200),
    herbario_sigla VARCHAR(80),
    herbario_email VARCHAR(200),
    logradouro VARCHAR(200),
    numero VARCHAR(10),
    cidade_nome VARCHAR(255),
    telefone_principal VARCHAR(200) -- Aggregated/Selected from public.telefones
);

-- Flattened User Dimension (for audit and system usage)
CREATE TABLE IF NOT EXISTS analytical.dim_usuario (
    usuario_key BIGSERIAL PRIMARY KEY,
    usuario_nome VARCHAR(200),
    usuario_email VARCHAR(200),
    tipo_usuario VARCHAR(100),
    herbario_vinculo_nome VARCHAR(200)
);

-- ------------------------------------------------------------------------------
-- 2. FACT TABLES
-- ------------------------------------------------------------------------------

-- Core Fact Table: Specimens (Tombos)
CREATE TABLE IF NOT EXISTS analytical.fact_tombos (
    tombo_hcf BIGINT PRIMARY KEY, -- Degenerate dimension, natural key
    
    -- Foreign Keys to Dimensions
    data_tombo_key INT REFERENCES analytical.dim_tempo(date_key),
    data_coleta_key INT REFERENCES analytical.dim_tempo(date_key),
    data_identificacao_key INT REFERENCES analytical.dim_tempo(date_key),
    taxonomia_key BIGINT REFERENCES analytical.dim_taxonomia(taxonomia_key),
    geografia_key BIGINT REFERENCES analytical.dim_geografia_local(geografia_key),
    coletor_key BIGINT REFERENCES analytical.dim_coletor(coletor_key),
    herbario_key BIGINT REFERENCES analytical.dim_herbario(herbario_key),
    
    -- Degenerate Dimensions / Attributes
    numero_coleta BIGINT,
    situacao VARCHAR(50), -- Cast from ENUM tombos_situacao
    ativo BOOLEAN,
    rascunho BOOLEAN,
    unicata BOOLEAN,
    nomes_populares TEXT,
    identificadores_nomes TEXT, -- Aggregated array/comma-separated string of identifiers
    
    -- Measures (Facts)
    latitude DOUBLE PRECISION,
    longitude DOUBLE PRECISION,
    altitude DOUBLE PRECISION,
    qtd_fotos INT DEFAULT 0, -- Pre-calculated count from public.tombos_fotos
    qtd_alteracoes INT DEFAULT 0 -- Pre-calculated count of edits
);

-- Fact Table: Specimen Loans and Transfers (Remessas)
CREATE TABLE IF NOT EXISTS analytical.fact_remessas (
    remessa_tombo_key BIGSERIAL PRIMARY KEY,
    
    -- Foreign Keys
    tombo_hcf BIGINT REFERENCES analytical.fact_tombos(tombo_hcf),
    herbario_origem_key BIGINT REFERENCES analytical.dim_herbario(herbario_key),
    herbario_destino_key BIGINT REFERENCES analytical.dim_herbario(herbario_key),
    data_envio_key INT REFERENCES analytical.dim_tempo(date_key),
    data_vencimento_key INT REFERENCES analytical.dim_tempo(date_key),
    
    -- Attributes
    tipo_retirada VARCHAR(50), -- Cast from ENUM retirada_exsiccata_tombos_tipo
    devolvido BOOLEAN,
    observacao_remessa TEXT,
    
    -- Measures
    dias_em_emprestimo INT, -- Calculated field: current_date - data_envio (or data_devolucao)
    dias_atraso INT -- Calculated field: current_date - data_vencimento (if devolvido = false)
);

-- Fact Table: Specimen Audit/Alterations
CREATE TABLE IF NOT EXISTS analytical.fact_alteracoes (
    alteracao_key BIGSERIAL PRIMARY KEY,
    
    -- Foreign Keys
    tombo_hcf BIGINT REFERENCES analytical.fact_tombos(tombo_hcf),
    usuario_key BIGINT REFERENCES analytical.dim_usuario(usuario_key),
    data_alteracao_key INT REFERENCES analytical.dim_tempo(date_key),
    
    -- Attributes
    status_alteracao VARCHAR(50), -- Cast from ENUM alteracoes_status
    observacao TEXT,
    is_alteracao_legado BOOLEAN DEFAULT FALSE -- Flag to identify records from tombo_alteracoes_antigas
);

-- ------------------------------------------------------------------------------
-- 3. INDEXES FOR PERFORMANCE
-- ------------------------------------------------------------------------------

-- Indexes on Fact Tombos
CREATE INDEX idx_fact_tombos_taxonomia ON analytical.fact_tombos(taxonomia_key);
CREATE INDEX idx_fact_tombos_geografia ON analytical.fact_tombos(geografia_key);
CREATE INDEX idx_fact_tombos_data_coleta ON analytical.fact_tombos(data_coleta_key);
CREATE INDEX idx_fact_tombos_herbario ON analytical.fact_tombos(herbario_key);

-- Indexes on Fact Remessas
CREATE INDEX idx_fact_remessas_tombo ON analytical.fact_remessas(tombo_hcf);
CREATE INDEX idx_fact_remessas_destino ON analytical.fact_remessas(herbario_destino_key);
CREATE INDEX idx_fact_remessas_vencimento ON analytical.fact_remessas(data_vencimento_key);

-- Indexes on Fact Alteracoes
CREATE INDEX idx_fact_alteracoes_tombo ON analytical.fact_alteracoes(tombo_hcf);
CREATE INDEX idx_fact_alteracoes_usuario ON analytical.fact_alteracoes(usuario_key);
```
