-- ==========================================
-- NEW DENORMALIZED TABLE
-- Represents a snapshot of an educational entity (Course) 
-- with all its associated indicators for a specific year.
-- Primary Key: (id_curso, id_ano)
-- ==========================================

CREATE TABLE INDICADORES_ESCOLA_ANO (
    id_curso INT NOT NULL,
    id_ano INT NOT NULL,
    sigla_uf VARCHAR(2) NOT NULL,
    nome_municipio VARCHAR(100) NOT NULL,
    nome_ies VARCHAR(100) NOT NULL,
    sigla_ies VARCHAR(20),
    codigo_ies INT UNIQUE NOT NULL,
    rede_ies VARCHAR(100) NOT NULL,
    nome_escola VARCHAR(100) NOT NULL,
    cod_escola INTEGER UNIQUE NOT NULL,
    nome_curso VARCHAR(100) NOT NULL,
    
    -- IDEB Indicators
    rend_1 REAL,
    rend_2 REAL,
    rend_3 REAL,
    rend_4 REAL,
    ind_rend REAL,
    nota_ideb REAL,
    
    -- ENADE Indicators
    total_inscritos INT CHECK (total_inscritos >= 0),
    total_concluintes INT CHECK (total_concluintes >= 0),
    nota_bruta_ce REAL,
    nota_padronizada_ce REAL,
    nota_bruta_fg REAL,
    nota_padronizada_fg REAL,
    nota_enade_continua REAL,
    nota_enade_faixa REAL,
    
    -- SAEB Indicators
    nota_mat REAL,
    nota_port REAL,
    nota_padrao REAL,
    
    -- Business Logic Constraints
    CONSTRAINT pk_indicadores_escola_ano PRIMARY KEY (id_curso, id_ano),
    CONSTRAINT chk_ano_positive CHECK (id_ano > 0),
    CONSTRAINT chk_qtd_positive CHECK (
        total_inscritos >= 0 AND 
        total_concluintes >= 0 AND
        total_inscritos >= total_concluintes
    )
);
