-- ==============================================================================
-- Denormalized Schema for Analytics and Reporting
-- ==============================================================================

-- 1. Denormalized table for Cleanup Efforts (Mutirões de Limpeza)
-- Combines: mutirao_de_limpeza, coleta_de_lixo, local_de_coleta, instituicao_organizadora, regiao
CREATE TABLE IF NOT EXISTS public.denorm_mutirao_limpeza
(
    id_mutirao integer NOT NULL,
    
    -- Cleanup Fact Details (from mutirao_de_limpeza)
    data_do_ocorrido date NOT NULL,
    horario_de_inicio time without time zone NOT NULL,
    numero_aproximado_de_participantes integer,
    duracao_total time without time zone NOT NULL,
    
    -- Location Details (from local_de_coleta)
    local_coleta_nome text COLLATE pg_catalog."default",
    
    -- Waste Collection Metrics (from coleta_de_lixo)
    plastico integer,
    borracha integer,
    madeira integer,
    produtos_texteis integer,
    papel integer,
    metal integer,
    vidro integer,
    materiais_de_pesca integer,
    outros_residuos integer,
    
    -- Organizer Dimension (from instituicao_organizadora)
    instituicao_nome text COLLATE pg_catalog."default",
    instituicao_carater text COLLATE pg_catalog."default",
    
    -- Geography Dimension (from regiao)
    municipio_id integer,
    municipio_nome text COLLATE pg_catalog."default",
    estado text COLLATE pg_catalog."default",
    
    CONSTRAINT denorm_mutirao_limpeza_pkey PRIMARY KEY (id_mutirao)
);

-- Create indexes to support common analytical filtering and grouping
CREATE INDEX IF NOT EXISTS idx_denorm_mutirao_data ON public.denorm_mutirao_limpeza(data_do_ocorrido);
CREATE INDEX IF NOT EXISTS idx_denorm_mutirao_estado ON public.denorm_mutirao_limpeza(estado);
CREATE INDEX IF NOT EXISTS idx_denorm_mutirao_instituicao ON public.denorm_mutirao_limpeza(instituicao_carater);


-- 2. Denormalized table for Conservation Units (Unidades de Conservação)
-- Combines: unidade_de_conservacao, bioma, categoria_manejo, esfera_administrativa, regiao
CREATE TABLE IF NOT EXISTS public.denorm_unidade_conservacao
(
    id_uc integer NOT NULL,
    
    -- Conservation Unit Details (from unidade_de_conservacao)
    uc_nome text COLLATE pg_catalog."default" NOT NULL,
    ano_de_criacao integer,
    area_soma_biomas double precision,
    area_soma_biomas_continentais double precision,
    area_marinha double precision,
    
    -- Biome Dimension (from bioma)
    bioma_nome text COLLATE pg_catalog."default",
    
    -- Management Category Dimension (from categoria_manejo)
    manejo_categoria text COLLATE pg_catalog."default",
    manejo_categoria_iucn text COLLATE pg_catalog."default",
    manejo_plano text COLLATE pg_catalog."default",
    
    -- Administrative Sphere Dimension (from esfera_administrativa)
    esfera_tipo text COLLATE pg_catalog."default",
    
    -- Geography Dimension (from regiao)
    municipio_id integer,
    municipio_nome text COLLATE pg_catalog."default",
    estado text COLLATE pg_catalog."default",
    
    CONSTRAINT denorm_unidade_conservacao_pkey PRIMARY KEY (id_uc)
);

-- Create indexes to support common analytical filtering and grouping
CREATE INDEX IF NOT EXISTS idx_denorm_uc_bioma ON public.denorm_unidade_conservacao(bioma_nome);
CREATE INDEX IF NOT EXISTS idx_denorm_uc_estado ON public.denorm_unidade_conservacao(estado);
CREATE INDEX IF NOT EXISTS idx_denorm_uc_esfera ON public.denorm_unidade_conservacao(esfera_tipo);
CREATE INDEX IF NOT EXISTS idx_denorm_uc_iucn ON public.denorm_unidade_conservacao(manejo_categoria_iucn);
