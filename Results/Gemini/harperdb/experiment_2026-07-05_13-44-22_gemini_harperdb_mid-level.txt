-- ==============================================================================
-- Denormalized Analytical Schema: denorm_user_entitlements
-- Grain: One row per User per Entitlement
-- Purpose: Optimized for read-heavy BI workloads, reporting, and aggregations.
-- ==============================================================================

CREATE TABLE denorm_user_entitlements (
    -- Identifiers
    user_id INT NOT NULL,
    entitlement_id INT NOT NULL,
    
    -- Dimensional Attributes
    feature_type VARCHAR(4),
    package_type VARCHAR(4),
    active_date DATE,
    inactive_date DATE,
    active_flag INT,
    
    -- Derived Dimensional Attributes
    is_expired BOOLEAN,               -- e.g., inactive_date < CURRENT_DATE
    validity_duration_days INT,       -- e.g., DATEDIFF(day, active_date, inactive_date)
    
    -- Measures
    use_limit INT,
    use_count INT,
    
    -- Derived Measures
    utilization_percentage DECIMAL(10, 4), -- e.g., (use_count * 1.0) / NULLIF(use_limit, 0)
    
    -- Composite Primary Key based on the grain of the table
    PRIMARY KEY (user_id, entitlement_id)
);

-- ==============================================================================
-- Analytical Indexes
-- Designed to support common filtering and grouping operations in BI tools
-- ==============================================================================

-- Index for user-centric queries (e.g., "Show all entitlements for User X")
CREATE INDEX idx_denorm_ue_user

-- Index for product/feature-centric aggregations (e.g., "Total users per package")
CREATE INDEX idx_denorm_ue_package_feature

-- Index for status-centric queries (e.g., "Show all active, non-expired entitlements")
CREATE INDEX idx_denorm_ue_status
