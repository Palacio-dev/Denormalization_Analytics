-- ==============================================================================
-- DENORMALIZED ANALYTICAL SCHEMA (STAR SCHEMA)
-- DIALECT: PostgreSQL

DROP TABLE IF EXISTS FATO_AVALIACAO CASCADE;
DROP TABLE IF EXISTS DIM_CURSO CASCADE;
DROP TABLE IF EXISTS DIM_ESCOLA CASCADE;
DROP TABLE IF EXISTS DIM_LOCALIDADE CASCADE;

-- ------------------------------------------------------------------------------
-- 1. DIM_LOCALIDADE (Collapses UF and MUNICIPIO)
-- ------------------------------------------------------------------------------
CREATE TABLE DIM_LOCALIDADE (
    id_municipio INT PRIMARY KEY,
    id_uf INT NOT NULL,
    sigla_uf VARCHAR(2) NOT NULL,
    nome_uf VARCHAR(100) NOT NULL,
    qtd_habitantes INT CHECK(qtd_habitantes >= 0),
    nome_municipio VARCHAR(100) NOT NULL,
    qtd_IES INT CHECK (qtd_IES >= 0),
    qtd_ESCOLAS INT CHECK (qtd_ESCOLAS >= 0),
    media_ideb REAL,
    media_saeb REAL,
    media_enade REAL,
    total_participantes INT CHECK (total_participantes >= 0)
);

-- ------------------------------------------------------------------------------
-- 2. DIM_ESCOLA (Standalone Dimension with Redundant Geographic Attributes)
-- ------------------------------------------------------------------------------
CREATE TABLE DIM_ESCOLA (
    id_escola INT PRIMARY KEY,
    id_municipio INT NOT NULL REFERENCES DIM_LOCALIDADE(id_municipio) ON DELETE CASCADE,
    nome_municipio VARCHAR(100) NOT NULL,
    sigla_uf VARCHAR(2) NOT NULL,
    nome_escola VARCHAR(100) NOT NULL,
    cod_escola INTEGER UNIQUE NOT NULL,
    rede_escola VARCHAR(100) NOT NULL
);

-- ------------------------------------------------------------------------------
-- 3. DIM_CURSO (Collapses IES and CURSO, adds Redundant Geographic Attributes)
-- ------------------------------------------------------------------------------
CREATE TABLE DIM_CURSO (
    id_curso INT PRIMARY KEY,
    id_ies INT NOT NULL,
    id_municipio INT NOT NULL REFERENCES DIM_LOCALIDADE(id_municipio) ON DELETE CASCADE,
    nome_municipio VARCHAR(100) NOT NULL,
    sigla_uf VARCHAR(2) NOT NULL,
    nome_ies VARCHAR(100) NOT NULL,
    sigla_ies VARCHAR(20),
    codigo_ies INT NOT NULL,
    rede_ies VARCHAR(100) NOT NULL,
    nome_curso VARCHAR(100) NOT NULL
);

-- ------------------------------------------------------------------------------
-- 4. FATO_AVALIACAO (Collapses ANO, ENADE, SAEB, and IDEB into a single Fact)
-- ------------------------------------------------------------------------------
CREATE TABLE FATO_AVALIACAO (
    id_avaliacao INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_curso INT NOT NULL REFERENCES DIM_CURSO(id_curso) ON DELETE CASCADE,
    ano INT NOT NULL,
    
    -- ENADE Attributes

    enade_nota_padronizada_ce REAL,
    enade_nota_bruta_fg REAL,
    enade_nota_padronizada_fg REAL,
    enade_nota_enade_continua REAL,
    enade_nota_enade_faixa REAL,
    
    -- SAEB Attributes
    saeb_nota_mat REAL,
    saeb_nota_port REAL,
    saeb_nota_padrao REAL,
    
    -- IDEB Attributes
    ideb_rend_1 REAL,
