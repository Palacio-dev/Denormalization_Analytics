-- Create a dedicated schema for the analytical model
CREATE SCHEMA IF NOT EXISTS analytical;

-- ==========================================
-- DIMENSION TABLES
-- ==========================================

-- 1. Dimension: Country (Denormalized with Continent)
CREATE TABLE IF NOT EXISTS analytical.dim_pais
(
    id_pais integer NOT NULL,
    nome_pais character varying(50) NOT NULL,
    ano_fundacao integer NOT NULL,
    nome_continente character varying(50) NOT NULL,
    CONSTRAINT dim_pais_pkey PRIMARY KEY (id_pais)
);

-- 2. Dimension: Energy Source
CREATE TABLE IF NOT EXISTS analytical.dim_fonte_energia
(
    id_fonte_energia integer NOT NULL,
    -- Converted from varchar[] to standard varchar for BI compatibility
    nome_fonte_energia character varying(100) NOT NULL, 
    CONSTRAINT dim_fonte_energia_pkey PRIMARY KEY (id_fonte_energia)
);

-- ==========================================
-- FACT TABLES
-- ==========================================

-- 3. Fact: Country Annual Indicators (Population & Emissions)
-- Grain: 1 row per Country per Year
CREATE TABLE IF NOT EXISTS analytical.fato_indicadores_pais_ano
(
    id_pais integer NOT NULL,
    ano integer NOT NULL,
    populacao_total bigint,
    emissoes_gee_total bigint,
    CONSTRAINT fato_indicadores_pais_ano_pkey PRIMARY KEY (id_pais, ano),
    CONSTRAINT fk_fato_indicadores_pais FOREIGN KEY (id_pais)
        REFERENCES analytical.dim_pais (id_pais) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
);

-- 4. Fact: Annual Energy Consumption
-- Grain: 1 row per Country per Year per Energy Source
CREATE TABLE IF NOT EXISTS analytical.fato_consumo_energia
(
    id_pais integer NOT NULL,
    id_fonte_energia integer NOT NULL,
    ano integer NOT NULL,
    -- Converted to numeric for better BI precision on percentages
    porcentagem_consumo numeric(5,2) NOT NULL, 
    CONSTRAINT fato_consumo_energia_pkey PRIMARY KEY (id_pais, id_fonte_energia, ano),
    CONSTRAINT fk_fato_consumo_pais FOREIGN KEY (id_pais)
        REFERENCES analytical.dim_pais (id_pais) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT fk_fato_consumo_fonte FOREIGN KEY (id_fonte_energia)
        REFERENCES analytical.dim_fonte_energia (id_fonte_energia) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
);

-- ==========================================
-- INDEXES FOR READ-HEAVY WORKLOADS
-- ==========================================

-- Indexes on foreign keys and commonly filtered/grouped columns
CREATE INDEX idx_fato_indicadores_ano ON analytical.fato_indicadores_pais_ano(ano);
CREATE INDEX idx_fato_consumo_ano ON analytical.fato_consumo_energia(ano);
CREATE INDEX idx_dim_pais_continente ON analytical.dim_pais(nome_continente);
