-- 2. Create Fact Table (Denormalized Star Schema Core)
-- This table consolidates consumption, emissions, and population data.
-- It includes denormalized geographic and energy attributes for query performance.

CREATE TABLE IF NOT EXISTS public.facto_prod_consumo_paises (
    id_consumo integer NOT NULL,
    id_emissao integer NOT NULL,
    id_pop integer NOT NULL,
    
    -- Primary Key: Unique combination of entity and year
    PRIMARY KEY (id_consumo, ano),
    
    -- Temporal Dimension
    ano integer NOT NULL,
    
    -- Denormalized Geographic Attributes (Eliminates JOINs with paises and continente)
    nome_pais character varying(50) COLLATE pg_catalog."default" NOT NULL,
    nome_continente character varying(50) COLLATE pg_catalog."default" NOT NULL,
    
    -- Denormalized Energy Attribute (Eliminates JOIN with fonte_energia)
    nome_tipo_energia character varying(50) COLLATE pg_catalog."default" NOT NULL,
    
    -- Measures
    porcentagem integer NOT NULL,
    quantidade_emitida bigint NOT NULL,
    quantidade_hab integer NOT NULL
);

-- 3. Create Dimension Table for Time (Standard Practice for BI)
CREATE TABLE IF NOT EXISTS public.dimenso_anos (
    ano integer NOT NULL,
    CONSTRAINT dimenso_anos_pkey PRIMARY KEY (ano)
);
