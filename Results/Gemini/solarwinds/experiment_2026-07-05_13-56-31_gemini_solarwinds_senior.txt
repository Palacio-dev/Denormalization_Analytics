-- ==============================================================================
-- DIMENSIONAL MODEL DDL (OLAP)
-- Target Architecture: Star Schema
-- ==============================================================================

-- 1. Date Dimension
-- Enables robust time-series analysis, holiday tracking, and fiscal calendars.
CREATE TABLE dim_date (
    date_sk INT PRIMARY KEY, -- Format: YYYYMMDD (e.g., 20231025)
    full_date DATE NOT NULL,
    calendar_year INT NOT NULL,
    calendar_quarter INT NOT NULL,
    calendar_month INT NOT NULL,
    month_name VARCHAR(20) NOT NULL,
    day_of_month INT NOT NULL,
    day_of_week INT NOT NULL,
    day_name VARCHAR(20) NOT NULL,
    is_weekend BOOLEAN NOT NULL
);

-- 2. Customer Dimension (SCD Type 2)
-- Tracks historical changes to customer profiles (e.g., address changes).
CREATE TABLE dim_customer (
    customer_sk SERIAL PRIMARY KEY,      -- Surrogate Key
    customer_id INT NOT NULL,            -- Natural Key (Source OLTP)
    customer_name VARCHAR(100),
    customer_email VARCHAR(100),
    customer_address TEXT,
    -- SCD Type 2 Metadata
    valid_from TIMESTAMP NOT NULL,
    valid_to TIMESTAMP NOT NULL DEFAULT '9999-12-31 23:59:59',
    is_current BOOLEAN NOT NULL DEFAULT TRUE
);

-- Index for ETL lookups on Natural Key
CREATE INDEX idx_dim_customer_nk ON dim_customer(customer_id);

-- 3. Product Dimension (SCD Type 2)
-- Tracks historical changes to product names and baseline prices.
CREATE TABLE dim_product (
    product_sk SERIAL PRIMARY KEY,       -- Surrogate Key
    product_id INT NOT NULL,             -- Natural Key (Source OLTP)
    product_name VARCHAR(100),
    current_price DECIMAL(10,2),
    -- SCD Type 2 Metadata
    valid_from TIMESTAMP NOT NULL,
    valid_to TIMESTAMP NOT NULL DEFAULT '9999-12-31 23:59:59',
    is_current BOOLEAN NOT NULL DEFAULT TRUE
);

-- Index for ETL lookups on Natural Key
CREATE INDEX idx_dim_product_nk ON dim_product(product_id);

-- 4. Sales Fact Table (Transaction Grain)
-- Merges 'orders' and 'order_details'. 
-- Partitioned by date to optimize time-based queries and data lifecycle management.
CREATE TABLE fact_sales (
    sales_sk BIGSERIAL,                  -- Optional Fact Surrogate Key (Useful for some BI tools)
    date_sk INT NOT NULL REFERENCES dim_date(date_sk),
    customer_sk INT NOT NULL REFERENCES dim_customer(customer_sk),
    product_sk INT NOT NULL REFERENCES dim_product(product_sk),
    
    -- Degenerate Dimensions (Traceability to OLTP)
    order_id INT NOT NULL,
    order_detail_id INT NOT NULL,
    
    -- Measures
    quantity INT NOT NULL,
    unit_price DECIMAL(10,2) NOT NULL,   -- Captured at time of transaction
    total_amount DECIMAL(12,2) NOT NULL, -- Pre-calculated: quantity * unit_price
    
    PRIMARY KEY (date_sk, sales_sk)      -- Composite PK required for table partitioning in Postgres
) PARTITION BY RANGE (date_sk);

-- Partitioning Example (Creating a partition for Q1 2024)
CREATE TABLE fact_sales_2024_q1 PARTITION OF fact_sales
    FOR VALUES FROM (20240101) TO (20240401);

-- Clustering / Indexing Recommendations for OLAP
-- Creating indexes on foreign keys to speed up Star Joins and Group By aggregations.
CREATE INDEX idx_fact_sales_customer ON fact_sales(customer_sk);
CREATE INDEX idx_fact_sales_product ON fact_sales(product_sk);

-- Note on Columnar Databases (e.g., Snowflake, Redshift, BigQuery):
-- If deploying to a modern cloud data warehouse, traditional B-Tree indexes are usually omitted.
-- Instead, you would use Cluster Keys / Sort Keys. 
-- Example for Snowflake: ALTER TABLE fact_sales CLUSTER BY (date_sk, customer_sk);
-- Example for Redshift: COMPOUND SORTKEY(date_sk, customer_sk, product_sk);
