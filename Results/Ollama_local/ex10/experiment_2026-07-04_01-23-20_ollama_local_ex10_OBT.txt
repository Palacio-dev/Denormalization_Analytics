-- Denormalized Schema: One Big Table Model
-- Strategy: Eliminate joins by flattening all entities into a single table.
-- Primary Key: Composite (Country, Year, Energy Source) to ensure uniqueness of the analytical row.

CREATE TABLE IF NOT EXISTS public.denormalized_energy_data (
    -- Primary Key: Represents the unique business entity for analysis
    -- (Country, Year, Energy Source) combination
    nome_paises character varying(50) NOT NULL,
    ano integer NOT NULL,
    nome_fonte_energia character varying(50) NOT NULL,
    PRIMARY KEY (nome_paises, ano, nome_fonte_energia),

    -- Core Metrics (Directly from consumo_anual and emissao_g_efeito_estufa)
    consumo_porcentagem integer NOT NULL,
    quantidade_emitida bigint NOT NULL,

    -- Demographic Data (Flattened from populacao)
    -- Note: We assume a unique population record per country/year exists in the source.
    -- If multiple populations exist for the same country/year, this row represents one of them.
    -- To handle this in a "Big Table" without complex grouping logic in the DDL,
    -- we include the population value directly.
    quantidade_hab integer NOT NULL,

    -- Contextual Data (Flattened from paises and continente)
    -- Derived: Continente Name (from paises.continente -> continente.nome)
    continente_nome character varying(50) NOT NULL,
    -- Original: paises.fundacao
    fundacao integer NOT NULL,

    -- Energy Context (Flattened from fonte_energia)
    -- Original: fonte_energia.nome
    nome_fonte_energia_original character varying(50) NOT NULL,

    -- Additional Metadata
    -- Note: 'ano_populacao' is redundant if 'ano' is the primary key dimension,
    -- but included for semantic clarity if the source data had a specific column distinction.
    -- Given the constraints, we treat 'ano' as the universal year.
    
    -- Constraints: No Foreign Keys allowed per ONE-BIG-TABLE MODEL
);
