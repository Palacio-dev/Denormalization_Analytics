-- ==============================================================================
-- DENORMALIZED ANALYTICS SCHEMA (STAR SCHEMA)
-- ==============================================================================

-- 1. DIMENSION: CUSTOMERS
-- Combines customers, customer_customer_demo, and customer_demographics
CREATE TABLE IF NOT EXISTS public.dim_customers
(
    customer_id character varying(5) COLLATE pg_catalog."default" NOT NULL,
    company_name character varying(40) COLLATE pg_catalog."default" NOT NULL,
    contact_name character varying(30) COLLATE pg_catalog."default",
    contact_title character varying(30) COLLATE pg_catalog."default",
    address character varying(60) COLLATE pg_catalog."default",
    city character varying(15) COLLATE pg_catalog."default",
    region character varying(15) COLLATE pg_catalog."default",
    postal_code character varying(10) COLLATE pg_catalog."default",
    country character varying(15) COLLATE pg_catalog."default",
    demographics_summary text COLLATE pg_catalog."default", -- Aggregated from customer_demographics
    CONSTRAINT pk_dim_customers PRIMARY KEY (customer_id)
);

-- 2. DIMENSION: PRODUCTS
-- Combines products, categories, and suppliers (Drops the picture bytea column)
CREATE TABLE IF NOT EXISTS public.dim_products
(
    product_id smallint NOT NULL,
    product_name character varying(40) COLLATE pg_catalog."default" NOT NULL,
    category_name character varying(15) COLLATE pg_catalog."default",
    category_description text COLLATE pg_catalog."default",
    supplier_name character varying(40) COLLATE pg_catalog."default",
    supplier_country character varying(15) COLLATE pg_catalog."default",
    quantity_per_unit character varying(20) COLLATE pg_catalog."default",
    unit_price real,
    units_in_stock smallint,
    discontinued integer NOT NULL,
    CONSTRAINT pk_dim_products PRIMARY KEY (product_id)
);

-- 3. DIMENSION: EMPLOYEES
-- Combines employees, employee_territories, territories, and region. 
-- Resolves reports_to self-join into manager_name. (Drops the photo bytea column)
CREATE TABLE IF NOT EXISTS public.dim_employees
(
    employee_id smallint NOT NULL,
    full_name character varying(31) COLLATE pg_catalog."default" NOT NULL, -- Derived: first_name + ' ' + last_name
    title character varying(30) COLLATE pg_catalog."default",
    hire_date date,
    city character varying(15) COLLATE pg_catalog."default",
    region character varying(15) COLLATE pg_catalog."default",
    country character varying(15) COLLATE pg_catalog."default",
    manager_name character varying(31) COLLATE pg_catalog."default", -- Derived from self-join on reports_to
    territories_summary text COLLATE pg_catalog."default", -- Comma-separated list of territories
    regions_summary text COLLATE pg_catalog."default",     -- Comma-separated list of regions
    CONSTRAINT pk_dim_employees PRIMARY KEY (employee_id)
);

-- 4. FACT: SALES
-- Combines orders, order_details, and shippers. 
-- Represents the lowest grain of a transaction (a line item on an order).
CREATE TABLE IF NOT EXISTS public.fact_sales
(
    order_id smallint NOT NULL,
    product_id smallint NOT NULL,
    customer_id character varying(5) COLLATE pg_catalog."default",
    employee_id smallint,
    
    -- Date Dimensions (Inline for simplicity, though a dim_date is also common)
    order_date date,
    required_date date,
    shipped_date date,
    
    -- Shipper Dimension (Denormalized directly into the fact table)
    shipper_name character varying(40) COLLATE pg_catalog."default",
    
    -- Shipping Geography
    ship_city character varying(15) COLLATE pg_catalog."default",
    ship_region character varying(15) COLLATE pg_catalog."default",
    ship_country character varying(15) COLLATE pg_catalog."default",
    
    -- Metrics / Measures
    unit_price real NOT NULL,
    quantity smallint NOT NULL,
    discount real NOT NULL,
    sales_amount real NOT NULL, -- Derived: quantity * unit_price * (1 - discount)
    freight real,               -- Note: Freight is at the order level, may need allocation logic in ETL
    
    CONSTRAINT pk_fact_sales PRIMARY KEY (order_id, product_id),
    
    -- Foreign Keys to Dimensions
    CONSTRAINT fk_fact_sales_customers FOREIGN KEY (customer_id)
        REFERENCES public.dim_customers (customer_id) MATCH SIMPLE,
    CONSTRAINT fk_fact_sales_products FOREIGN KEY (product_id)
        REFERENCES public.dim_products (product_id) MATCH SIMPLE,
    CONSTRAINT fk_fact_sales_employees FOREIGN KEY (employee_id)
        REFERENCES public.dim_employees (employee_id) MATCH SIMPLE
);

-- Note: The `us_states` table was omitted from the dimensional model as it acts as an isolated 
-- lookup table in the source schema. If state names are required for reporting, they should be 
-- joined during the ETL process and stored directly within the `ship_region` or `region` columns 
-- of the fact and dimension tables.
