### Complete SQL DDL Creation Script for the Denormalized Schema

```sql
-- Create a new schema for the analytical data warehouse
CREATE SCHEMA IF NOT EXISTS analytics;

-- ==========================================
-- DIMENSION TABLES
-- ==========================================

-- 1. Product Dimension
-- Combines Product, ProductSubcategory, ProductCategory, and ProductModel
CREATE TABLE IF NOT EXISTS analytics.dim_product
(
    product_key serial PRIMARY KEY,
    product_id integer NOT NULL,
    product_name character varying(50) NOT NULL,
    product_number character varying(25) NOT NULL,
    color character varying(15),
    standard_cost numeric NOT NULL,
    list_price numeric NOT NULL,
    size character varying(5),
    weight numeric(8, 2),
    product_line character(2),
    class character(2),
    style character(2),
    subcategory_name character varying(50),
    category_name character varying(50),
    model_name character varying(50),
    sell_start_date timestamp without time zone,
    sell_end_date timestamp without time zone,
    is_current boolean DEFAULT true
);

COMMENT ON TABLE analytics.dim_product IS 'Flattened product dimension including category and model details.';

-- 2. Customer Dimension
-- Combines Customer, Person, Store, Address, StateProvince, and CountryRegion
CREATE TABLE IF NOT EXISTS analytics.dim_customer
(
    customer_key serial PRIMARY KEY,
    customer_id integer NOT NULL,
    person_type character(2),
    first_name character varying(50),
    last_name character varying(50),
    full_name character varying(100),
    store_name character varying(50),
    address_line1 character varying(60),
    city character varying(30),
    state_province_code character(3),
    state_province_name character varying(50),
    country_region_code character varying(3),
    country_region_name character varying(50),
    postal_code character varying(15)
);

COMMENT ON TABLE analytics.dim_customer IS 'Flattened customer dimension containing both individual and store customers with geographical data.';

-- 3. Employee Dimension
-- Combines Employee, SalesPerson, Person, and Department
CREATE TABLE IF NOT EXISTS analytics.dim_employee
(
    employee_key serial PRIMARY KEY,
    employee_id integer NOT NULL, -- Maps to BusinessEntityID
    national_id_number character varying(15),
    job_title character varying(50),
    first_name character varying(50),
    last_name character varying(50),
    full_name character varying(100),
    birth_date date,
    hire_date date,
    gender character(1),
    department_name character varying(50),
    department_group character varying(50),
    is_salesperson boolean DEFAULT false,
    sales_quota numeric,
    commission_pct numeric,
    current_flag boolean DEFAULT true
);

COMMENT ON TABLE analytics.dim_employee IS 'Flattened employee dimension including sales quota for sales representatives.';

-- 4. Vendor Dimension
-- Combines Vendor and BusinessEntity
CREATE TABLE IF NOT EXISTS analytics.dim_vendor
(
    vendor_key serial PRIMARY KEY,
    vendor_id integer NOT NULL, -- Maps to BusinessEntityID
    account_number character varying(15) NOT NULL,
    vendor_name character varying(50) NOT NULL,
    credit_rating smallint,
    preferred_vendor_status boolean,
    active_flag boolean
);

COMMENT ON TABLE analytics.dim_vendor IS 'Flattened vendor dimension for purchasing analytics.';

-- 5. Territory Dimension
-- Combines SalesTerritory and CountryRegion
CREATE TABLE IF NOT EXISTS analytics.dim_territory
(
    territory_key serial PRIMARY KEY,
    territory_id integer NOT NULL,
    territory_name character varying(50) NOT NULL,
    country_region_code character varying(3) NOT NULL,
    territory_group character varying(50) NOT NULL
);

COMMENT ON TABLE analytics.dim_territory IS 'Sales territory dimension.';

-- ==========================================
-- FACT TABLES
-- ==========================================

-- 6. Fact Sales
-- Combines SalesOrderHeader and SalesOrderDetail
CREATE TABLE IF NOT EXISTS analytics.fact_sales
(
    sales_key serial PRIMARY KEY,
    sales_order_id integer NOT NULL,
    sales_order_detail_id integer NOT NULL,
    order_date timestamp without time zone NOT NULL,
    due_date timestamp without time zone NOT NULL,
    ship_date timestamp without time zone,
    customer_key integer NOT NULL REFERENCES analytics.dim_customer(customer_key),
    employee_key integer REFERENCES analytics.dim_employee(employee_key),
    product_key integer NOT NULL REFERENCES analytics.dim_product(product_key),
    territory_key integer REFERENCES analytics.dim_territory(territory_key),
    online_order_flag boolean NOT NULL,
    order_qty smallint NOT NULL,
    unit_price numeric NOT NULL,
    unit_price_discount numeric NOT NULL DEFAULT 0.0,
    line_total numeric NOT NULL, -- Derived: order_qty * unit_price * (1 - unit_price_discount)
    header_freight numeric, -- Note: Header level metric, use caution when aggregating
    header_tax_amt numeric  -- Note: Header level metric, use caution when aggregating
);

COMMENT ON TABLE analytics.fact_sales IS 'Granular fact table for sales at the line-item level.';

-- 7. Fact Purchases
-- Combines PurchaseOrderHeader and PurchaseOrderDetail
CREATE TABLE IF NOT EXISTS analytics.fact_purchases
(
    purchase_key serial PRIMARY KEY,
    purchase_order_id integer NOT NULL,
    purchase_order_detail_id integer NOT NULL,
    order_date timestamp without time zone NOT NULL,
    ship_date timestamp without time zone,
    vendor_key integer NOT NULL REFERENCES analytics.dim_vendor(vendor_key),
    employee_key integer NOT NULL REFERENCES analytics.dim_employee(employee_key),
    product_key integer NOT NULL REFERENCES analytics.dim_product(product_key),
    order_qty smallint NOT NULL,
    unit_price numeric NOT NULL,
    received_qty numeric(8, 2) NOT NULL,
    rejected_qty numeric(8, 2) NOT NULL,
    line_total numeric NOT NULL -- Derived: order_qty * unit_price
);

COMMENT ON TABLE analytics.fact_purchases IS 'Granular fact table for procurement at the line-item level.';

-- 8. Fact Inventory
-- Derived from ProductInventory and Location
CREATE TABLE IF NOT EXISTS analytics.fact_inventory
(
    inventory_key serial PRIMARY KEY,
    product_key integer NOT NULL REFERENCES analytics.dim_product(product_key),
    location_id smallint NOT NULL,
    location_name character varying(50) NOT NULL,
    shelf character varying(10) NOT NULL,
    bin smallint NOT NULL,
    quantity smallint NOT NULL DEFAULT 0,
    snapshot_date timestamp without time zone DEFAULT now()
);

COMMENT ON TABLE analytics.fact_inventory IS 'Fact table representing current or snapshotted inventory levels.';

-- ==========================================
-- INDEXES FOR PERFORMANCE
-- ==========================================

CREATE INDEX idx_fact_sales_date ON analytics.fact_sales(order_date);
CREATE INDEX idx_fact_sales_customer ON analytics.fact_sales(customer_key);
CREATE INDEX idx_fact_sales_product ON analytics.fact_sales(product_key);

CREATE INDEX idx_fact_purchases_date ON analytics.fact_purchases(order_date);
CREATE INDEX idx_fact_purchases_vendor ON analytics.fact_purchases(vendor_key);
CREATE INDEX idx_fact_purchases_product ON analytics.fact_purchases(product_key);
```
