-- ==========================================
-- DENORMALIZED SCHEMA: EMPLOYEES_ANALYTICS
-- ==========================================
-- This schema is designed for high-read analytical workloads.
-- It eliminates JOINs by pre-materializing common attributes.
-- Note: Foreign Key constraints are removed to reflect the denormalized nature 
-- and potential for out-of-order data in analytical queries.

-- 1. Employee Core (Denormalized lookup for basic info)
CREATE TABLE IF NOT EXISTS employees.employee_core (
    id bigint NOT NULL,
    first_name character varying(14) COLLATE pg_catalog."default" NOT NULL,
    last_name character varying(16) COLLATE pg_catalog."default" NOT NULL,
    gender character varying(1) COLLATE pg_catalog."default" NOT NULL,
    birth_date date NOT NULL,
    hire_date date NOT NULL,
    PRIMARY KEY (id)
);

-- 2. Department Core (Denormalized lookup for department name)
CREATE TABLE IF NOT EXISTS employees.department_core (
    id character(4) COLLATE pg_catalog."default" NOT NULL,
    dept_name character varying(40) COLLATE pg_catalog."default" NOT NULL,
    PRIMARY KEY (id)
);

-- 3. Department Assignments (Merged department_employee & department_manager)
-- Combines relationship data with department name for easy lookup
CREATE TABLE IF NOT EXISTS employees.department_assignments (
    employee_id bigint NOT NULL,
    department_id character(4) COLLATE pg_catalog."default" NOT NULL,
    dept_name character varying(40) COLLATE pg_catalog."default" NOT NULL, -- Redundant attribute
    from_date date NOT NULL,
    to_date date NOT NULL,
    PRIMARY KEY (employee_id, department_id, from_date) -- Composite PK to handle multiple roles over time
);

-- 4. Employee Compensation & Roles (Merged salary & title)
-- Combines salary and title data to avoid joins
CREATE TABLE IF NOT EXISTS employees.employee_compensation (
    employee_id bigint NOT NULL,
    amount bigint NOT NULL,
    title character varying(50) COLLATE pg_catalog."default" NOT NULL,
    from_date date NOT NULL,
    to_date date,
    PRIMARY KEY (employee_id, from_date) -- Composite PK based on salary history
);
