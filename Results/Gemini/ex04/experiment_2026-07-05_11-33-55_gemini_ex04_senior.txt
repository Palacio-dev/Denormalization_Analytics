-- 1. CONFORMED DIMENSIONS

CREATE TABLE IF NOT EXISTS public.dim_location
(
    location_key serial NOT NULL,
    country_code character varying NOT NULL,
    country_name character varying,
    region_code character varying NOT NULL,
    region_name character varying NOT NULL,
    PRIMARY KEY (location_key)
);

-- Index for lookup performance
CREATE UNIQUE INDEX idx_dim_location_country_code ON public.dim_location(country_code);
CREATE INDEX idx_dim_location_region ON public.dim_location(region_code);


-- 2. FACT TABLES
-- ------------------------------------------------------------------------------

-- FACT: COUNTRY INDICATORS
-- Combines demography, employment, education, economy, and health.
-- Grain: 1 row per Country per Year
CREATE TABLE IF NOT EXISTS public.fact_country_indicators
(
    indicator_key serial NOT NULL,
    location_key integer NOT NULL,
    year integer NOT NULL,
    
    -- Demography metrics
    pop_density double precision,
    urban_pop double precision,
    rural_pop double precision,
    net_migration double precision,
    death_rate double precision,
    birth_rate double precision,
    
    -- Employment metrics
    child_emp double precision,
    unemp double precision,
    vulnerable_emp double precision,
    part_time double precision,
    employers double precision,
    labor_force_total double precision,
    labor_force_fem double precision,
    
    -- Education metrics
    child_out_of_school double precision,
    progression_to_sec double precision,
    edu_expenditure double precision,
    preprim_enroll double precision,
    prim_enrol double precision,
    sec_enrol double precision,
    terti_enrol double precision,
    
    -- Economy metrics
    gdp double precision,
    inflation double precision,
    tax_revenue double precision,
    
    -- Health metrics
    hospital_beds double precision,
    physicians double precision,
    health_expenditure double precision,
    
    PRIMARY KEY (indicator_key, year)
) PARTITION BY RANGE (year);

-- FACT: WELFARE SURVEY
-- Grain: 1 row per Survey (typically Country + Year + Survey Type)
CREATE TABLE IF NOT EXISTS public.fact_welfare_survey
(
    survey_key serial NOT NULL,
    location_key integer NOT NULL,
    year integer NOT NULL,
    
    -- Survey Metadata (Degenerate dimensions/attributes)
    welfare_type character varying,
    survey_acronym character varying,
    survey_comparability integer,
    comparable_spell character varying,
    distribution_type character varying,
    survey_coverage character varying,
    reporting_level character varying,
    
    -- Survey Metrics
    poverty_line double precision,
    headcount double precision,
    poverty_gap double precision,
    poverty_severity double precision,
    gini double precision,
    reporting_pop integer,
    reporting_pce double precision,
    spl double precision,
    
    PRIMARY KEY (survey_key, year)
) PARTITION BY RANGE (year);

-- FACT: WELFARE DECILE
-- Grain: 1 row per Survey per Decile
CREATE TABLE IF NOT EXISTS public.fact_welfare_decile
(
    decile_key serial NOT NULL,
    survey_key integer NOT NULL,
    location_key integer NOT NULL, -- Denormalized for direct slicing
    year integer NOT NULL,         -- Denormalized for direct slicing
    decile_name character varying,
    decile_value double precision,
    
    PRIMARY KEY (decile_key, year)
) PARTITION BY RANGE (year);

-- FACT: POPULATION
-- Grain: 1 row per Country per Year per Age Group per Gender
CREATE TABLE IF NOT EXISTS public.fact_population
(
    population_key serial NOT NULL,
    location_key integer NOT NULL,
    year integer NOT NULL,
    pop_ages character varying,
    gender character varying,
    population_count double precision,
    
    PRIMARY KEY (population_key, year)
) PARTITION BY RANGE (year);

-- FACT: LIFE EXPECTANCY
-- Grain: 1 row per Country per Year per Gender
CREATE TABLE IF NOT EXISTS public.fact_life_expectancy
(
    life_exp_key serial NOT NULL,
    location_key integer NOT NULL,
    year integer NOT NULL,
    gender character varying,
    life_expectancy_value double precision,
    
    PRIMARY KEY (life_exp_key, year)
