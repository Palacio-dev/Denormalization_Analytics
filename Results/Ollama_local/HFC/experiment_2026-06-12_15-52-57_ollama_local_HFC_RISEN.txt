LEFT JOIN public.solos s ON t.solo_id = s.id
LEFT JOIN public.relevos r ON t.relevo_id = r.id
LEFT JOIN public.vegetacoes v ON t.vegetacao_id = v.id
LEFT JOIN public.locais_coleta lc ON t.local_coleta_id = lc.id
LEFT JOIN public.fase_sucessional fs ON t.fase_sucessional_id = fs.numero
ORDER BY t.hcf;

-- -----------------------------------------------------------------------------
-- 4. MAINTENANCE TRIGGER (Optional but Recommended)
-- -----------------------------------------------------------------------------
-- To keep the denormalized table consistent with the original tombos table
-- automatically, we can create a trigger that updates tombos_analytics whenever
-- a tombos record is updated or inserted.
-- Note: This assumes the dimension tables are also updated or have default values.
-- For a production environment, dimension tables should also be denormalized if queried frequently.

CREATE OR REPLACE FUNCTION update_tombos_analytics()
RETURNS TRIGGER AS $$
BEGIN
    IF (TG_OP = 'INSERT') THEN
        INSERT INTO public.tombos_analytics (
            hcf, data_tombo, data_coleta_dia, observacao, nomes_populares, numero_coleta,
            latitude, longitude, altitude, entidade_id, local_coleta_id,
            variedade_id, tipo_id, data_identificacao_dia, data_identificacao_mes, data_identificacao_ano,
            situacao, especie_id, genero_id, familia_id, sub_familia_id, sub_especie_id,
            nome_cientifico, colecao_anexa_id, data_coleta_mes, data_coleta_ano,
            solo_id, relevo_id, vegetacao_id, created_at, updated_at, ativo, taxon,
            rascunho, coletor_id, descricao, unicata, cidade_id, fase_sucessional_id,
            especie_nome, especie_autor_id, especie_genero_id, especie_familia_id,
            especie_nome_popular, genero_nome, familia_nome, familia_autor_id,
            sub_familia_nome, sub_familia_autor_id, sub_especie_nome,
            solo_nome, relevo_nome, vegetacao_nome, local_coleta_descricao, local_coleta_cidade_id, local_coleta_fase_sucessional_id,
            local_coleta_fase_numero, fase_sucessional_nome
        )
        SELECT 
            NEW.hcf, NEW.data_tombo, NEW.data_coleta_dia, NEW.observacao, NEW.nomes_populares, NEW.numero_coleta,
            NEW.latitude, NEW.longitude, NEW.altitude, NEW.entidade_id, NEW.local_coleta_id,
            NEW.variedade_id, NEW.tipo_id, NEW.data_identificacao_dia, NEW.data_identificacao_mes, NEW.data_identificacao_ano,
            NEW.situacao, NEW.especie_id, NEW.genero_id, NEW.familia_id, NEW.sub_familia_id, NEW.sub_especie_id,
            NEW.nome_cientifico, NEW.colecao_anexa_id, NEW.data_coleta_mes, NEW.data_coleta_ano,
            NEW.solo_id, NEW.relevo_id, NEW.vegetacao_id, NEW.created_at, NEW.updated_at, NEW.ativo, NEW.taxon,
            NEW.rascunho, NEW.coletor_id, NEW.descricao, NEW.unicata, NEW.cidade_id, NEW.fase_sucessional_id,
            e.nome, e.autor_id, e.genero_id, e.familia_id, COALESCE(e.nomes_populares, ''),
            g.nome, f.nome, f.autor_id, sf.nome, sf.autor_id, se.nome,
            s.nome, r.nome, v.nome, lc.descricao, lc.cidade_id, lc.fase_sucessional_id, lc.fase_numero,
            fs.nome
        FROM public.especies e
        LEFT JOIN public.generos g ON NEW.genero_id = g.id
        LEFT JOIN public.familias f ON NEW.familia_id = f.id
        LEFT JOIN public.sub_familias sf ON NEW.sub_familia_id = sf.id
        LEFT JOIN public.sub_especies se ON NEW.sub_especie_id = se.id
        LEFT JOIN public.solos s ON NEW.solo_id = s.id
        LEFT JOIN public.relevos r ON NEW.relevo_id = r.id
        LEFT JOIN public.vegetacoes v ON NEW.vegetacao_id = v.id
        LEFT JOIN public.locais_coleta lc ON NEW.local_coleta_id = lc.id
        LEFT JOIN public.fase_sucessional fs ON NEW.fase_sucessional_id = fs.numero
        WHERE e.id = NEW.especie_id;
        
    ELSIF (TG_OP = 'UPDATE') THEN
        UPDATE public.tombos_analytics
        SET 
            data_tombo = NEW.data_tombo,
            data_coleta_dia = NEW.data_coleta_dia,
            observacao = NEW.observacao,
            nomes_populares = NEW.nomes_populares,
            numero_coleta = NEW.numero_coleta,
            latitude = NEW.latitude,
            longitude = NEW.longitude,
            altitude = NEW.altitude,
            entidade_id = NEW.entidade_id,
            local_coleta_id = NEW.local_coleta_id,
            variedade_id = NEW.variedade_id,
            tipo_id = NEW.tipo_id,
            data_identificacao_dia = NEW.data_identificacao_dia,
            data_identificacao_mes = NEW.data_identificacao_mes,
            data_identificacao_ano = NEW.data_identificacao_ano,
            situacao = NEW.situacao,
            especie_id = NEW.especie_id,
            genero_id = NEW.genero_id,
            familia_id = NEW.familia_id,
            sub_familia_id = NEW.sub_familia_id,
            sub_especie_id = NEW.sub_especie_id,
            nome_cientifico = NEW.nome_cientifico,
            colecao_anexa_id = NEW.colecao_anexa_id,
            data_coleta_mes = NEW.data_coleta_mes,
            data_coleta_ano = NEW.data_coleta_ano,
            solo_id = NEW.solo_id,
            relevo_id = NEW.relevo_id,
            vegetacao_id = NEW.vegetacao_id,
            created_at = NEW.created_at,
            updated_at = NEW.updated_at,
            ativo = NEW.ativo,
            taxon = NEW.taxon,
            rascunho = NEW.rascunho,
            coletor_id = NEW.coletor_id,
            descricao = NEW.descricao,
            unicata = NEW.unicata,
            cidade_id = NEW.cidade_id,
            fase_sucessional_id = NEW.fase_sucessional_id,
            especie_nome = COALESCE(e.nome, OLD.especie_nome),
            especie_autor_id = COALESCE(e.autor_id, OLD.especie_autor_id),
            especie_genero_id = COALESCE(e.genero_id, OLD.especie_genero_id),
            especie_familia_id = COALESCE(e.familia_id, OLD.especie_familia_id),
            especie_nome_popular = COALESCE(e.nomes_populares, OLD.especie_nome_popular),
            genero_nome = COALESCE(g.nome, OLD.genero_nome),
            familia_nome = COALESCE(f.nome, OLD.familia_nome),
            familia_autor_id = COALESCE(f.autor_id, OLD.familia_autor_id),
            sub_familia_nome = COALESCE(sf.nome, OLD.sub_familia_nome),
            sub_familia_autor_id = COALESCE(sf.autor_id, OLD.sub_familia_autor_id),
            sub_especie_nome = COALESCE(se.nome, OLD.sub_especie_nome),
            solo_nome = COALESCE(s.nome, OLD.solo_nome),
            relevo_nome = COALESCE(r.nome, OLD.relevo_nome),
            vegetacao_nome = COALESCE(v.nome, OLD.vegetacao_nome),
            local_coleta_descricao = COALESCE(lc.descricao, OLD.local_coleta_descricao),
            local_coleta_cidade_id = COALESCE(lc.cidade_id, OLD.local_coleta_cidade_id),
            local_coleta_fase_sucessional_id = COALESCE(lc.fase_sucessional_id, OLD.local_coleta_fase_sucessional_id),
            local_coleta_fase_numero = COALESCE(lc.fase_numero, OLD.local_coleta_fase_numero),
            fase_sucessional_nome = COALESCE(fs.nome, OLD.fase_sucessional_nome)
        FROM public.tombos t
        LEFT JOIN public.especies e ON t.especie_id = e.id
        LEFT JOIN public.generos g ON t.genero_id = g.id
        LEFT JOIN public.familias f ON t.familia_id = f.id
        LEFT JOIN public.sub_familias sf ON t.sub_familia_id = sf.id
        LEFT JOIN public.sub_especies se ON t.sub_especie_id = se.id
        LEFT JOIN public.solos s ON t.solo_id = s.id
        LEFT JOIN public.relevos r ON t.relevo_id = r.id
        LEFT JOIN public.vegetacoes v ON t.vegetacao_id = v.id
        LEFT JOIN public.locais_coleta lc ON t.local_coleta_id = lc.id
        LEFT JOIN public.fase_sucessional fs ON t.fase_sucessional_id = fs.numero
        WHERE t.hcf = NEW.hcf;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_tombos_analytics
AFTER INSERT OR UPDATE ON public.tombos
FOR EACH ROW
EXECUTE FUNCTION update_tombos_analytics();
