-- ==============================================================================
-- Target Architecture: Kimball Star Schema
-- Schema Name: analytics
-- Objective: Optimized for OLAP workloads, BI integrations, and fast aggregations
-- ==============================================================================

CREATE SCHEMA IF NOT EXISTS analytics;

-- ==============================================================================
-- DIMENSIONS
-- ==============================================================================

-- 1. Date Dimension (Conformed)
CREATE TABLE analytics.dim_date (
    date_key INT NOT NULL, -- Format: YYYYMMDD
    full_date DATE NOT NULL,
    day_of_week SMALLINT NOT NULL,
    day_name VARCHAR(10) NOT NULL,
    day_of_month SMALLINT NOT NULL,
    day_of_year SMALLINT NOT NULL,
    week_of_year SMALLINT NOT NULL,
    month_number SMALLINT NOT NULL,
    month_name VARCHAR(10) NOT NULL,
    calendar_quarter SMALLINT NOT NULL,
    calendar_year SMALLINT NOT NULL,
    is_weekend BOOLEAN NOT NULL,
    is_holiday BOOLEAN NOT NULL DEFAULT FALSE,
    CONSTRAINT pk_dim_date PRIMARY KEY (date_key)
);
COMMENT ON TABLE analytics.dim_date IS 'Conformed date dimension for all time-series analysis.';

-- 2. Product Dimension (Flattened Hierarchy)
CREATE TABLE analytics.dim_product (
    product_key SERIAL NOT NULL, -- Surrogate Key
    product_id INT NOT NULL,     -- Business Key
    product_name VARCHAR(50) NOT NULL,
    product_number VARCHAR(25) NOT NULL,
    color VARCHAR(15) DEFAULT 'N/A',
    standard_cost NUMERIC(19,4),
    list_price NUMERIC(19,4),
    size VARCHAR(5) DEFAULT 'N/A',
    weight NUMERIC(8,2),
    product_line VARCHAR(20),
    product_class VARCHAR(20),
    product_style VARCHAR(20),
    subcategory_name VARCHAR(50) DEFAULT 'N/A',
    category_name VARCHAR(50) DEFAULT 'N/A',
    model_name VARCHAR(50) DEFAULT 'N/A',
    is_salable BOOLEAN NOT NULL,
    is_manufactured BOOLEAN NOT NULL,
    valid_from TIMESTAMP NOT NULL DEFAULT NOW(),
    valid_to TIMESTAMP DEFAULT '9999-12-31',
    is_current BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT pk_dim_product PRIMARY KEY (product_key)
);
COMMENT ON TABLE analytics.dim_product IS 'SCD Type 2 dimension containing flattened product, subcategory, category, and model data.';

-- 3. Customer Dimension (Flattened Person/Store & Geography)
CREATE TABLE analytics.dim_customer (
    customer_key SERIAL NOT NULL, -- Surrogate Key
    customer_id INT NOT NULL,     -- Business Key
    customer_type VARCHAR(15) NOT NULL, -- 'Store' or 'Individual'
    customer_name VARCHAR(150) NOT NULL, -- Store Name or Person Full Name
    email_address VARCHAR(50),
    phone_number VARCHAR(25),
    address_line1 VARCHAR(60),
    city VARCHAR(30),
    state_province_code CHAR(3),
    state_province_name VARCHAR(50),
    country_region_code VARCHAR(3),
    country_region_name VARCHAR(50),
    postal_code VARCHAR(15),
    valid_from TIMESTAMP NOT NULL DEFAULT NOW(),
    valid_to TIMESTAMP DEFAULT '9999-12-31',
    is_current BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT pk_dim_customer PRIMARY KEY (customer_key)
);
COMMENT ON TABLE analytics.dim_customer IS 'SCD Type 2 dimension containing customer demographics and flattened geography.';

-- 4. Employee Dimension (Flattened HR Data)
CREATE TABLE analytics.dim_employee (
    employee_key SERIAL NOT NULL, -- Surrogate Key
    business_entity_id INT NOT NULL, -- Business Key
    employee_name VARCHAR(150) NOT NULL,
    job_title VARCHAR(50) NOT NULL,
    hire_date DATE NOT NULL,
    gender CHAR(1),
    marital_status CHAR(1),
    is_salaried BOOLEAN,
    department_name VARCHAR(50),
    department_group VARCHAR(50),
    shift_name VARCHAR(50),
    valid_from TIMESTAMP NOT NULL DEFAULT NOW(),
    valid_to TIMESTAMP DEFAULT '9999-12-31',
    is_current BOOLEAN NOT NULL DEFAULT TRUE,
    CONSTRAINT pk_dim_employee PRIMARY KEY (employee_key)
);
COMMENT ON TABLE analytics.dim_employee IS 'SCD Type 2 dimension containing employee and current department/shift data.';

-- 5. Territory Dimension
CREATE TABLE analytics.dim_territory (
    territory_key SERIAL NOT NULL,
    territory_id INT NOT NULL,
    territory_name VARCHAR(50) NOT NULL,
    territory_group VARCHAR(50) NOT NULL,
    country_region_code VARCHAR(3) NOT NULL,
    CONSTRAINT pk_dim_territory PRIMARY KEY (territory_key)
);

-- 6. Vendor Dimension
CREATE TABLE analytics.dim_vendor (
    vendor_key SERIAL NOT NULL,
    business_entity_id INT NOT NULL,
    vendor_name VARCHAR(50) NOT NULL,
    account_number VARCHAR(15) NOT NULL,
    credit_rating SMALLINT NOT NULL,
    is_preferred BOOLEAN NOT NULL,
    is_active BOOLEAN NOT NULL,
    CONSTRAINT pk_dim_vendor PRIMARY KEY (vendor_key)
);

-- ==============================================================================
-- FACT TABLES
-- ==============================================================================

-- 1. Fact Sales (Grain: Sales Order Line Item)
-- Partitioning Strategy: Range partitioned by order_date_key for fast time-based querying
CREATE TABLE analytics.fact_sales (
    sales_key BIGSERIAL NOT NULL,
    order_date_key INT NOT NULL,
    due_date_key INT NOT NULL,
    ship_date_key INT NOT NULL,
    customer_key INT NOT NULL,
    employee_key INT NOT NULL, -- Salesperson
    territory_key INT NOT NULL,
    product_key INT NOT NULL,
    sales_order_id INT NOT NULL,
    sales_order_detail_id INT NOT NULL,
    is_online_order BOOLEAN NOT NULL,
    order_qty SMALLINT NOT NULL,
    unit_price NUMERIC(19,4) NOT NULL,
    unit_price_discount NUMERIC(19,4) NOT NULL DEFAULT 0.0,
    sales_amount NUMERIC(19,4) NOT NULL, -- Derived: order_qty * unit_price * (1 - unit_price_discount)
    tax_amount NUMERIC(19,4) NOT NULL,   -- Allocated from header
    freight_amount NUMERIC(19,4) NOT NULL, -- Allocated from header
    total_due NUMERIC(19,4) NOT NULL,    -- Derived: sales_amount + tax + freight
    CONSTRAINT pk_fact_sales PRIMARY KEY (sales_key, order_date_key)
) PARTITION BY RANGE (order_date_key);

COMMENT ON TABLE analytics.fact_sales IS 'Fact table for sales transactions at the line-item grain.';

-- 2. Fact Purchases (Grain: Purchase Order Line Item)
CREATE TABLE analytics.fact_purchases (
    purchase_key BIGSERIAL NOT NULL,
    order_date_key INT NOT NULL,
    ship_date_key INT NOT NULL,
    vendor_key INT NOT NULL,
    employee_key INT NOT NULL, -- Buyer
    product_key INT NOT NULL,
    purchase_order_id INT NOT NULL,
    purchase_order_detail_id INT NOT NULL,
    order_qty SMALLINT NOT NULL,
    received_qty NUMERIC(8,2) NOT NULL,
    rejected_qty NUMERIC(8,2) NOT NULL,
    unit_price NUMERIC(19,4) NOT NULL,
    line_total NUMERIC(19,4) NOT NULL, -- Derived: order_qty * unit_price
    CONSTRAINT pk_fact_purchases PRIMARY KEY (purchase_key, order_date_key)
) PARTITION BY RANGE (order_date_key);

COMMENT ON TABLE analytics.fact_purchases IS 'Fact table for procurement transactions at the line-item grain.';
