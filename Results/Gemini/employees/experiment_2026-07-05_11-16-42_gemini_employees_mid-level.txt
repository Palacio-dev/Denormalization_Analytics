```sql
-- Create a dedicated schema for the analytical model
CREATE SCHEMA IF NOT EXISTS analytics;

-- ==========================================
-- DIMENSIONS
-- ==========================================

-- 1. Employee Dimension
-- Contains static and slowly changing (Type 1) employee details.
CREATE TABLE IF NOT EXISTS analytics.dim_employee
(
    employee_sk BIGSERIAL NOT NULL,
    employee_id BIGINT NOT NULL,
    first_name VARCHAR(14) NOT NULL,
    last_name VARCHAR(16) NOT NULL,
    full_name VARCHAR(31) NOT NULL, -- Derived attribute for BI convenience
    birth_date DATE NOT NULL,
    gender VARCHAR(10) NOT NULL,    -- Converted from ENUM for BI compatibility
    hire_date DATE NOT NULL,
    CONSTRAINT pk_dim_employee PRIMARY KEY (employee_sk),
    CONSTRAINT uk_dim_employee_id UNIQUE (employee_id)
);

CREATE INDEX idx_dim_employee_hire_date ON analytics.dim_employee(hire_date);

-- 2. Department Dimension
-- Contains department attributes.
CREATE TABLE IF NOT EXISTS analytics.dim_department
(
    department_sk BIGSERIAL NOT NULL,
    department_id CHAR(4) NOT NULL,
    department_name VARCHAR(40) NOT NULL,
    CONSTRAINT pk_dim_department PRIMARY KEY (department_sk),
    CONSTRAINT uk_dim_department_id UNIQUE (department_id)
);

-- ==========================================
-- FACTS
-- ==========================================

-- 3. Employee History Fact (Unified State/Periodic Snapshot Fact)
-- Consolidates salary, title, and department assignments into a single timeline.
-- A new row is generated in this table whenever an employee's salary, title, or department changes.
CREATE TABLE IF NOT EXISTS analytics.fact_employee_history
(
    employee_history_sk BIGSERIAL NOT NULL,
    employee_sk BIGINT NOT NULL,
    department_sk BIGINT NOT NULL,
    job_title VARCHAR(50) NOT NULL,     -- Denormalized from title table
    salary_amount BIGINT NOT NULL,      -- Measure
    effective_from_date DATE NOT NULL,  -- Start of this specific state
    effective_to_date DATE NOT NULL,    -- End of this specific state (use '9999-12-31' for active)
    is_current BOOLEAN NOT NULL,        -- True if this is the employee's current state
    record_duration_days INT,           -- Derived measure: effective_to_date - effective_from_date
    
    CONSTRAINT pk_fact_employee_history PRIMARY KEY (employee_history_sk),
    CONSTRAINT fk_fact_emp_hist_employee FOREIGN KEY (employee_sk)
        REFERENCES analytics.dim_employee (employee_sk) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT fk_fact_emp_hist_department FOREIGN KEY (department_sk)
        REFERENCES analytics.dim_department (department_sk) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
);

-- Indexes optimized for BI query patterns (filtering by current status, department, and date ranges)
CREATE INDEX idx_fact_emp_hist_current ON analytics.fact_employee_history(is_current) WHERE is_current = TRUE;
CREATE INDEX idx_fact_emp_hist_emp_sk ON analytics.fact_employee_history(employee_sk);
CREATE INDEX idx_fact_emp_hist_dept_sk ON analytics.fact_employee_history(department_sk);
CREATE INDEX idx_fact_emp_hist_dates ON analytics.fact_employee_history(effective_from_date, effective_to_date);


-- 4. Department Manager Fact
-- Tracks the historical and current managers of departments.
CREATE TABLE IF NOT EXISTS analytics.fact_department_manager
(
    department_manager_sk BIGSERIAL NOT NULL,
    department_sk BIGINT NOT NULL,
    manager_employee_sk BIGINT NOT NULL,
    effective_from_date DATE NOT NULL,
    effective_to_date DATE NOT NULL,
    is_current BOOLEAN NOT NULL,
    
    CONSTRAINT pk_fact_department_manager PRIMARY KEY (department_manager_sk),
    CONSTRAINT fk_fact_dept_mgr_department FOREIGN KEY (department_sk)
        REFERENCES analytics.dim_department (department_sk) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION,
    CONSTRAINT fk_fact_dept_mgr_employee FOREIGN KEY (manager_employee_sk)
        REFERENCES analytics.dim_employee (employee_sk) MATCH SIMPLE
        ON UPDATE NO ACTION
        ON DELETE NO ACTION
);

CREATE INDEX idx_fact_dept_mgr_current ON analytics.fact_department_manager(is_current) WHERE is_current = TRUE;
CREATE INDEX idx_fact_dept_mgr_dept_sk ON analytics.fact_department_manager(department_sk);
```
