-- Trigger to maintain consistency of redundant dept_name in employee table
-- This ensures that when a department is updated, the employee table is also updated.

RETURNS TRIGGER AS $$
BEGIN
    IF OLD.id IS NOT NULL AND NEW.id IS NOT NULL THEN
        UPDATE employees.employee
        SET dept_name = NEW.dept_name
        WHERE id = NEW.id;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;
