-- 1. DIMENSION: LOCATION
-- Consolidates UF and MUNICIPIO to avoid geographic joins.
DROP TABLE IF EXISTS dim_location CASCADE;
CREATE TABLE dim_location (
    id_location INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_municipio INT NOT NULL,
    nome_municipio VARCHAR(100) NOT NULL,
    id_uf INT NOT NULL,
    sigla_uf VARCHAR(2) NOT NULL,
    nome_uf VARCHAR(100) NOT NULL,
    uf_qtd_habitantes INT,
    municipio_qtd_ies INT,
    municipio_qtd_escolas INT,
    municipio_media_ideb REAL,
    municipio_media_saeb REAL,
    municipio_media_enade REAL,
    municipio_total_participantes INT
);

-- 2. DIMENSION: COURSE (Higher Education)
-- Consolidates IES and CURSO to allow fast filtering by institution attributes.
DROP TABLE IF EXISTS dim_course CASCADE;
CREATE TABLE dim_course (
    id_course_dim INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_curso INT NOT NULL,
    nome_curso VARCHAR(100) NOT NULL,
    id_ies INT NOT NULL,
    nome_ies VARCHAR(100) NOT NULL,
    sigla_ies VARCHAR(20),
    codigo_ies INT NOT NULL,
    rede_ies VARCHAR(100) NOT NULL
);

-- 3. DIMENSION: SCHOOL (Basic Education)
-- Preserves ESCOLA as a dimension for basic education metrics.
DROP TABLE IF EXISTS dim_school CASCADE;
CREATE TABLE dim_school (
    id_school_dim INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    id_escola INT NOT NULL,
    nome_escola VARCHAR(100) NOT NULL,
    cod_escola INT NOT NULL,
    rede_escola VARCHAR(100) NOT NULL
);

-- 4. FACT: EVALUATION
-- Flattens ANO, ENADE, SAEB, and IDEB into a single unified fact table.
-- Links directly to location, course, and school to avoid snowflake joins.
DROP TABLE IF EXISTS fact_evaluation CASCADE;
CREATE TABLE fact_evaluation (
    id_fact INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    ano INT NOT NULL,
    
    -- Foreign Keys to Dimensions
    id_location INT NOT NULL REFERENCES dim_location(id_location),
    id_course_dim INT REFERENCES dim_course(id_course_dim),
