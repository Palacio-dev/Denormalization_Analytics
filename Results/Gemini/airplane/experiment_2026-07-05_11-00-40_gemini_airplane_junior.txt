CREATE SCHEMA IF NOT EXISTS analytics;

-- ==============================================================================
-- 1. DIMENSION: Flights (Denormalized)
-- Merges: flights, aircrafts_data, airports_data (Departure & Arrival)
-- ==============================================================================
CREATE TABLE IF NOT EXISTS analytics.dim_flights
(
    flight_id integer NOT NULL,
    flight_no character(6) COLLATE pg_catalog."default" NOT NULL,
    scheduled_departure timestamp with time zone NOT NULL,
    scheduled_arrival timestamp with time zone NOT NULL,
    actual_departure timestamp with time zone,
    actual_arrival timestamp with time zone,
    status character varying(20) COLLATE pg_catalog."default" NOT NULL,
    
    -- Aircraft Data
    aircraft_code character(3) COLLATE pg_catalog."default" NOT NULL,
    aircraft_model jsonb NOT NULL,
    aircraft_range integer NOT NULL,
    
    -- Departure Airport Data
    dep_airport_code character(3) COLLATE pg_catalog."default" NOT NULL,
    dep_airport_name jsonb NOT NULL,
    dep_city jsonb NOT NULL,
    dep_coordinates point NOT NULL,
    dep_timezone text COLLATE pg_catalog."default" NOT NULL,
    
    -- Arrival Airport Data
    arr_airport_code character(3) COLLATE pg_catalog."default" NOT NULL,
    arr_airport_name jsonb NOT NULL,
    arr_city jsonb NOT NULL,
    arr_coordinates point NOT NULL,
    arr_timezone text COLLATE pg_catalog."default" NOT NULL,
    
    CONSTRAINT dim_flights_pkey PRIMARY KEY (flight_id)
);

COMMENT ON TABLE analytics.dim_flights IS 'Denormalized dimension table for flights, including aircraft and airport details';

-- ==============================================================================
-- 2. FACT: Passenger Flights (Denormalized)
-- Merges: ticket_flights, tickets, bookings, boarding_passes
-- ==============================================================================
CREATE TABLE IF NOT EXISTS analytics.fact_passenger_flights
(
    ticket_no character(13) COLLATE pg_catalog."default" NOT NULL,
    flight_id integer NOT NULL,
    
    -- Booking Data
    book_ref character(6) COLLATE pg_catalog."default" NOT NULL,
    book_date timestamp with time zone NOT NULL,
    total_booking_amount numeric(10, 2) NOT NULL,
    
    -- Passenger & Ticket Data
    passenger_id character varying(20) COLLATE pg_catalog."default" NOT NULL,
    passenger_name text COLLATE pg_catalog."default" NOT NULL,
    contact_data jsonb,
    
    -- Flight Segment Data
    fare_conditions character varying(10) COLLATE pg_catalog."default" NOT NULL,
    flight_amount numeric(10, 2) NOT NULL,
    
    -- Boarding Pass Data (Nullable because a passenger might not have checked in yet)
    boarding_no integer,
    seat_no character varying(4) COLLATE pg_catalog."default",
    
    CONSTRAINT fact_passenger_flights_pkey PRIMARY KEY (ticket_no, flight_id)
);

COMMENT ON TABLE analytics.fact_passenger_flights IS 'Denormalized fact table for passenger flight segments, including booking, ticket, and boarding pass details';

-- ==============================================================================
-- 3. DIMENSION: Aircraft Seats (Denormalized)
-- Merges: seats, aircrafts_data
-- ==============================================================================
CREATE TABLE IF NOT EXISTS analytics.dim_aircraft_seats
(
    aircraft_code character(3) COLLATE pg_catalog."default" NOT NULL,
    aircraft_model jsonb NOT NULL,
    seat_no character varying(4) COLLATE pg_catalog."default" NOT NULL,
    fare_conditions character varying(10) COLLATE pg_catalog."default" NOT NULL,
    
    CONSTRAINT dim_aircraft_seats_pkey PRIMARY KEY (aircraft_code, seat_no)
);

COMMENT ON TABLE analytics.dim_aircraft_seats IS 'Denormalized dimension table for aircraft seats and capacity';

-- ==============================================================================
-- Foreign Keys & Indexes for Analytical Performance
-- ==============================================================================

-- Link Fact table to Flight Dimension

    ADD CONSTRAINT fact_passenger_flights_flight_id_fkey FOREIGN KEY (flight_id)
    REFERENCES analytics.dim_flights (flight_id) MATCH SIMPLE
    ON UPDATE NO ACTION
    ON DELETE NO ACTION;

-- Indexes for common analytical filtering and grouping

    ON analytics.fact_passenger_flights(book_date);
