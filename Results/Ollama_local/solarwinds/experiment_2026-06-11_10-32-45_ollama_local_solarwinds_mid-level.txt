-- Products Dimension
CREATE TABLE dim_products (
    product_sk INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    product_id INT NOT NULL UNIQUE, -- Physical ID for reference
    product_name VARCHAR(100),
    price DECIMAL(10,2),
    -- Product category could be added here

);

-- Order Items Dimension (Intermediate dimension to link Order and Product)
CREATE TABLE dim_order_items (
    item_sk INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    order_item_id INT NOT NULL UNIQUE, -- Unique ID for the specific line item
    product_sk INT NOT NULL REFERENCES dim_products(product_sk),
    -- Reference back to order for context if needed
    -- order_sk INT NOT NULL REFERENCES dim_orders(order_sk)
);

-- --------------------------------------------------------
-- FACT TABLES
-- --------------------------------------------------------

-- Fact Orders (Transactional Core)
-- Contains the primary key of the order and the foreign keys to dimensions
CREATE TABLE fact_orders (
    order_fact_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    -- Surrogate Keys to Dimensions
    customer_sk INT NOT NULL REFERENCES dim_customers(customer_sk),
    order_sk INT NOT NULL REFERENCES dim_orders(order_sk),
    
    -- Measures
    order_quantity INT, -- If quantity is per order (e.g., 1 box of 10)
    order_total DECIMAL(10,2), -- Sum of all line items for this order
    
    -- Timestamps for time intelligence
    event_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Fact Order Items (Detailed Transactions)
-- This is the core transactional fact
CREATE TABLE fact_order_items (
    item_fact_id INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    -- Surrogate Keys to Dimensions
    order_sk INT NOT NULL REFERENCES dim_orders(order_sk),
    item_sk INT NOT NULL REFERENCES dim_order_items(item_sk),
    
    -- Measures
    quantity INT NOT NULL,
    unit_price DECIMAL(10,2),
    -- Pre-calculated measure to avoid runtime multiplication
    line_total DECIMAL(10,2) GENERATED ALWAYS AS (quantity * unit_price)
);
