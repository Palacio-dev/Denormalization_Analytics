### SQL DDL Script for Denormalized Schema

```sql
-- ==============================================================================
-- Analytical Schema (OLAP) - Optimized for Read-Heavy Workloads
-- ==============================================================================

-- 1. Create the denormalized Location Dimension
-- This table consolidates Localizacoes, Municipios, and Estados.
-- It flattens the relationship to two municipalities into single row attributes.
CREATE TABLE IF NOT EXISTS public.dim_localizacao (
    id_localizacao INTEGER PRIMARY KEY,
    latitude DOUBLE PRECISION,
    longitude DOUBLE PRECISION,
    municipio_principal VARCHAR,
    uf_principal CHAR(2),
    municipio_secundario VARCHAR,
    uf_secundario CHAR(2)
);

-- 2. Create the central Fact Table
-- This table acts as a Factless Fact / Profile Fact. 
-- It contains measures (potkw, potfis) and has absorbed simple dimensions (Rios, Proprietarios).
CREATE TABLE IF NOT EXISTS public.fact_usina (
    id_usina INTEGER PRIMARY KEY,
    usina_nome VARCHAR NOT NULL,
    potkw FLOAT,
    potfis FLOAT,
    iniciop DATE,
    atualizacao DATE,
    lat DOUBLE PRECISION,
    long DOUBLE PRECISION,
    estagio VARCHAR,
    
    -- Denormalized attributes (previously required joins)
    proprietario_nome VARCHAR,
    rio_nome VARCHAR,
    
    -- Foreign key to the consolidated location dimension
    id_localizacao INTEGER NOT NULL,
    
    CONSTRAINT fk_fact_usina_localizacao 
        FOREIGN KEY (id_localizacao) 
        REFERENCES public.dim_localizacao(id_localizacao)
);

-- ==============================================================================
-- Analytical Indexes for BI Performance
-- ==============================================================================

-- Index for geographic filtering (very common in BI dashboards)
CREATE INDEX IF NOT EXISTS idx_dim_loc_uf ON public.dim_localizacao(uf_principal);
CREATE INDEX IF NOT EXISTS idx_dim_loc_mun ON public.dim_localizacao(municipio_principal);

-- Index for filtering/grouping by owner and river
CREATE INDEX IF NOT EXISTS idx_fact_usina_proprietario ON public.fact_usina(proprietario_nome);
CREATE INDEX IF NOT EXISTS idx_fact_usina_rio ON public.fact_usina(rio_nome);

-- Index for timeline/date range queries
CREATE INDEX IF NOT EXISTS idx_fact_usina_iniciop ON public.fact_usina(iniciop);

COMMIT;
```
