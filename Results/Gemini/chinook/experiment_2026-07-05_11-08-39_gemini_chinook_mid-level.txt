-- ==============================================================================
-- Analytical Data Model (Star Schema)
-- Optimized for BI Dashboards, Reporting, and Aggregations
-- ==============================================================================

-- 1. Create Dimension: DimDate
CREATE TABLE IF NOT EXISTS public."DimDate"
(
    "DateKey" integer NOT NULL, -- Format: YYYYMMDD
    "FullDate" date NOT NULL,
    "Year" integer NOT NULL,
    "Quarter" integer NOT NULL,
    "Month" integer NOT NULL,
    "MonthName" character varying(20) NOT NULL,
    "DayOfMonth" integer NOT NULL,
    "DayOfWeek" integer NOT NULL,
    "DayName" character varying(20) NOT NULL,
    "IsWeekend" boolean NOT NULL,
    CONSTRAINT "PK_DimDate" PRIMARY KEY ("DateKey")
);

-- 2. Create Dimension: DimCustomer
-- Consolidates Customer and Employee (Support Rep)
CREATE TABLE IF NOT EXISTS public."DimCustomer"
(
    "CustomerKey" integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    "CustomerId" integer NOT NULL, -- Natural Key
    "CustomerName" character varying(61) NOT NULL, -- Derived: FirstName + ' ' + LastName
    "Company" character varying(80),
    "Address" character varying(70),
    "City" character varying(40),
    "State" character varying(40),
    "Country" character varying(40),
    "PostalCode" character varying(10),
    "Email" character varying(60) NOT NULL,
    "SupportRepId" integer,
    "SupportRepName" character varying(41), -- Derived: Employee FirstName + ' ' + LastName
    "SupportRepTitle" character varying(30),
    CONSTRAINT "PK_DimCustomer" PRIMARY KEY ("CustomerKey")
);

-- 3. Create Dimension: DimTrack
-- Consolidates Track, Album, Artist, Genre, and MediaType
CREATE TABLE IF NOT EXISTS public."DimTrack"
(
    "TrackKey" integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    "TrackId" integer NOT NULL, -- Natural Key
    "TrackName" character varying(200) NOT NULL,
    "AlbumName" character varying(160),
    "ArtistName" character varying(120),
    "GenreName" character varying(120),
    "MediaTypeName" character varying(120),
    "Composer" character varying(220),
    "Milliseconds" integer NOT NULL,
    "Bytes" integer,
    "UnitPrice" numeric(10, 2) NOT NULL,
    CONSTRAINT "PK_DimTrack" PRIMARY KEY ("TrackKey")
);

-- 4. Create Dimension: DimPlaylist
CREATE TABLE IF NOT EXISTS public."DimPlaylist"
(
    "PlaylistKey" integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    "PlaylistId" integer NOT NULL, -- Natural Key
    "PlaylistName" character varying(120),
    CONSTRAINT "PK_DimPlaylist" PRIMARY KEY ("PlaylistKey")
);

-- 5. Create Bridge Table: BridgePlaylistTrack
-- Resolves Many-to-Many relationship between Playlists and Tracks for OLAP
CREATE TABLE IF NOT EXISTS public."BridgePlaylistTrack"
(
    "PlaylistKey" integer NOT NULL,
    "TrackKey" integer NOT NULL,
    CONSTRAINT "PK_BridgePlaylistTrack" PRIMARY KEY ("PlaylistKey", "TrackKey")
);

-- 6. Create Fact Table: FactSales
-- Consolidates Invoice and InvoiceLine
CREATE TABLE IF NOT EXISTS public."FactSales"
(
    "SalesKey" integer GENERATED ALWAYS AS IDENTITY NOT NULL,
    "InvoiceLineId" integer NOT NULL, -- Natural Key
    "InvoiceId" integer NOT NULL, -- Degenerate Dimension
    "DateKey" integer NOT NULL, -- FK to DimDate
    "CustomerKey" integer NOT NULL, -- FK to DimCustomer
    "TrackKey" integer NOT NULL, -- FK to DimTrack
    "BillingAddress" character varying(70),
    "BillingCity" character varying(40),
    "BillingState" character varying(40),
    "BillingCountry" character varying(40),
    "BillingPostalCode" character varying(10),
    "UnitPrice" numeric(10, 2) NOT NULL,
    "Quantity" integer NOT NULL,
    "LineTotal" numeric(10, 2) NOT NULL, -- Derived: UnitPrice * Quantity
    CONSTRAINT "PK_FactSales" PRIMARY KEY ("SalesKey")
);
