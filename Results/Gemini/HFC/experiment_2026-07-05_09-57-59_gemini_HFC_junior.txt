### SQL DDL for Denormalized Schema

```sql
-- ==============================================================================
-- DENORMALIZED SCHEMA FOR ANALYTICS AND REPORTING
-- ==============================================================================

-- 1. ANALYTICS_TOMBOS (One Big Table for Specimens)
-- This table centralizes the 'tombos' (specimens) and flattens taxonomy, 
-- geography, environment, collectors, and identifiers into a single row per specimen.
CREATE TABLE IF NOT EXISTS public.analytics_tombos
(
    -- Primary Identifier
    hcf bigint NOT NULL,
    
    -- Dates
    data_tombo timestamp with time zone,
    data_coleta date, -- Derived from dia/mes/ano
    data_identificacao date, -- Derived from dia/mes/ano
    
    -- Specimen Core Data
    numero_coleta bigint,
    observacao text,
    situacao character varying(50), -- Cast from enum tombos_situacao
    ativo boolean,
    rascunho boolean,
    unicata boolean,
    tipo_nome character varying(250), -- From tipos.nome
    
    -- Taxonomy (Flattened)
    reino_nome character varying(200),
    familia_nome character varying(200),
    sub_familia_nome character varying(300),
    genero_nome character varying(200),
    especie_nome character varying(200),
    sub_especie_nome character varying(255),
    variedade_nome character varying(200),
    nome_cientifico text,
    nomes_populares text,
    taxon character varying(45),
    autor_especie_nome character varying(200), -- From autores.nome
    
    -- Geography & Environment (Flattened)
    latitude double precision,
    longitude double precision,
    altitude double precision,
    cidade_id bigint, -- Kept as ID assuming a separate geographic master table, or can be enriched via ETL
    local_coleta_descricao text,
    fase_sucessional_nome character varying(200),
    solo_nome character varying(300),
    relevo_nome character varying(300),
    vegetacao_nome character varying(300),
    
    -- People (Flattened / Aggregated)
    coletor_principal_nome character varying(255),
    coletores_complementares text, -- Aggregated from coletores_complementares
    identificadores_nomes text, -- Aggregated comma-separated string from tombos_identificadores -> identificadores
    
    -- Institution
    herbario_nome character varying(200),
    herbario_sigla character varying(80),
    
    -- Media & Attachments (Aggregated)
    quantidade_fotos integer DEFAULT 0, -- Count from tombos_fotos
    tem_foto_vivo boolean DEFAULT false, -- Boolean OR from tombos_fotos.em_vivo
    colecao_anexa_tipo character varying(50), -- Cast from enum
    colecao_anexa_observacoes text,
    
    -- External Integrations
    reflora_ja_comparou boolean DEFAULT false,
    specieslink_ja_comparou boolean DEFAULT false,
    
    -- Metadata
    created_at timestamp with time zone,
    updated_at timestamp with time zone,
    
    CONSTRAINT pk_analytics_tombos PRIMARY KEY (hcf)
);

-- Indexes for common analytical filters
CREATE INDEX idx_analytics_tombos_tax_familia ON public.analytics_tombos(familia_nome);
CREATE INDEX idx_analytics_tombos_tax_especie ON public.analytics_tombos(especie_nome);
CREATE INDEX idx_analytics_tombos_data_coleta ON public.analytics_tombos(data_coleta);
CREATE INDEX idx_analytics_tombos_herbario ON public.analytics_tombos(herbario_sigla);


-- 2. ANALYTICS_ALTERACOES (Audit and Activity Fact Table)
-- Flattens the tracking of changes made to specimens, including user and herbarium details.
CREATE TABLE IF NOT EXISTS public.analytics_alteracoes
(
    id bigint NOT NULL,
    tombo_hcf bigint NOT NULL,
    
    -- User Details (Flattened)
    usuario_nome character varying(200),
    usuario_email character varying(200),
    tipo_usuario character varying(100), -- From tipos_usuarios.tipo
    
    -- Herbarium Details
    herbario_nome character varying(200),
    herbario_sigla character varying(80),
    
    -- Alteration Details
    status character varying(50), -- Cast from enum alteracoes_status
    observacao text,
    tombo_json_snapshot text, -- State of the record at the time
    
    -- Timestamps
    created_at timestamp with time zone,
    
    CONSTRAINT pk_analytics_alteracoes PRIMARY KEY (id)
);

CREATE INDEX idx_analytics_alteracoes_tombo ON public.analytics_alteracoes(tombo_hcf);
CREATE INDEX idx_analytics_alteracoes_created ON public.analytics_alteracoes(created_at);
CREATE INDEX idx_analytics_alteracoes_usuario ON public.analytics_alteracoes(usuario_nome);


-- 3. ANALYTICS_REMESSAS (Shipments and Loans Fact Table)
-- Flattens shipments and aggregates the items (exsiccatas) inside them.
CREATE TABLE IF NOT EXISTS public.analytics_remessas
(
    id bigint NOT NULL,
    
    -- Origin and Destination
    herbario_origem_nome character varying(200),
    herbario_origem_sigla character varying(80),
    entidade_destino_nome character varying(200), -- Assuming joined with herbarios for destination
    
    -- Shipment Details
    data_envio timestamp with time zone,
    observacao text,
    
    -- Aggregated Metrics from retirada_exsiccata_tombos
    total_exsicatas_enviadas integer DEFAULT 0,
    total_exsicatas_devolvidas integer DEFAULT 0,
    total_exsicatas_pendentes integer DEFAULT 0, -- (enviadas - devolvidas)
    data_vencimento_proxima timestamp with time zone, -- MIN(data_vencimento) for pending items
    
    -- Metadata
    created_at timestamp with time zone,
    updated_at timestamp with time zone,
    
    CONSTRAINT pk_analytics_remessas PRIMARY KEY (id)
);

CREATE INDEX idx_analytics_remessas_data_envio ON public.analytics_remessas(data_envio);
CREATE INDEX idx_analytics_remessas_origem ON public.analytics_remessas(herbario_origem_sigla);
```
