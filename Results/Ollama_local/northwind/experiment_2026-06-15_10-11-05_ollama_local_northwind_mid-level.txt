### Complete SQL DDL Creation Script for the Denormalized Schema

```sql
-- Drop existing tables if they exist to ensure a clean state
DROP TABLE IF EXISTS public.orders;
DROP TABLE IF EXISTS public.products;
DROP TABLE IF EXISTS public.regions;
DROP TABLE IF EXISTS public.shippers;

-- 1. Fact Table: Orders (Denormalized with Customer, Employee, and Shipper details)
CREATE TABLE IF NOT EXISTS public.orders
(
    order_id smallint NOT NULL,
    customer_id character varying(5) COLLATE pg_catalog."default",
    employee_id smallint,
    ship_via smallint,
    order_date date,
    required_date date,
    shipped_date date,
    freight real,
    -- Customer Details (Denormalized from customers table)
    company_name character varying(40) COLLATE pg_catalog."default",
    contact_name character varying(30) COLLATE pg_catalog."default",
    contact_title character varying(30) COLLATE pg_catalog."default",
    address character varying(60) COLLATE pg_catalog."default",
    city character varying(15) COLLATE pg_catalog."default",
    region character varying(15) COLLATE pg_catalog."default",
    postal_code character varying(10) COLLATE pg_catalog."default",
    country character varying(15) COLLATE pg_catalog."default",
    phone character varying(24) COLLATE pg_catalog."default",
    fax character varying(24) COLLATE pg_catalog."default",
    -- Employee Details (Denormalized from employees table)
    last_name character varying(20) COLLATE pg_catalog."default",
    first_name character varying(10) COLLATE pg_catalog."default",
    title character varying(30) COLLATE pg_catalog."default",
    title_of_courtesy character varying(25) COLLATE pg_catalog."default",
    birth_date date,
    hire_date date,
    address character varying(60) COLLATE pg_catalog."default",
    city character varying(15) COLLATE pg_catalog."default",
    region character varying(15) COLLATE pg_catalog."default",
    postal_code character varying(10) COLLATE pg_catalog."default",
    country character varying(15) COLLATE pg_catalog."default",
    home_phone character varying(24) COLLATE pg_catalog."default",
    extension character varying(4) COLLATE pg_catalog."default",
    photo bytea,
    notes text COLLATE pg_catalog."default",
    reports_to smallint,
    photo_path character varying(255) COLLATE pg_catalog."default",
    CONSTRAINT pk_orders PRIMARY KEY (order_id)
);

-- 2. Fact Table: Products (Denormalized with Supplier, Category, and Product details)
CREATE TABLE IF NOT EXISTS public.products
(
    product_id smallint NOT NULL,
    product_name character varying(40) COLLATE pg_catalog."default" NOT NULL,
    supplier_id smallint,
    category_id smallint,
    quantity_per_unit character varying(20) COLLATE pg_catalog."default",
    unit_price real,
    units_in_stock smallint,
    units_on_order smallint,
    reorder_level smallint,
    discontinued integer NOT NULL,
    -- Supplier Details (Denormalized from suppliers table)
    company_name character varying(40) COLLATE pg_catalog."default",
    contact_name character varying(30) COLLATE pg_catalog."default",
    contact_title character varying(30) COLLATE pg_catalog."default",
    address character varying(60) COLLATE pg_catalog."default",
    city character varying(15) COLLATE pg_catalog."default",
    region character varying(15) COLLATE pg_catalog."default",
    postal_code character varying(10) COLLATE pg_catalog."default",
    country character varying(15) COLLATE pg_catalog."default",
    phone character varying(24) COLLATE pg_catalog."default",
    fax character varying(24) COLLATE pg_catalog."default",
    homepage text COLLATE pg_catalog."default",
    -- Category Details (Denormalized from categories table)
    category_name character varying(15) COLLATE pg_catalog."default",
    description text COLLATE pg_catalog."default",
    picture bytea,
    CONSTRAINT pk_products PRIMARY KEY (product_id)
);

-- 3. Dimension Table: Regions (Merged from region and territories)
CREATE TABLE IF NOT EXISTS public.regions
(
    region_id smallint NOT NULL,
    region_description character varying(60) COLLATE pg_catalog."default" NOT NULL,
    territory_id character varying(20) COLLATE pg_catalog."default",
    CONSTRAINT pk_regions PRIMARY KEY (region_id)
);

-- 4. Dimension Table: Shippers (Optional if kept separate, or denormalized into orders)
-- For this denormalized approach, we keep shippers as a small dimension or fully denormalize into orders.
-- Let's create a small shipper table for reference, assuming orders has ship_via FK.
CREATE TABLE IF NOT EXISTS public.shippers
(
    shipper_id smallint NOT NULL,
    company_name character varying(40) COLLATE pg_catalog."default" NOT NULL,
    phone character varying(24) COLLATE pg_catalog."default",
    CONSTRAINT pk_shippers PRIMARY KEY (shipper_id)
);

-- 5. Fact Table: Order Details (Kept normalized as it links products to orders)
-- Note: In a fully denormalized model, we might put product details here too, but typically order_details 
-- is kept separate to avoid duplicating product info in both products and order_details.
-- However, to maximize denormalization for "Order Line" analysis, we can keep it simple.
CREATE TABLE IF NOT EXISTS public.order_details
(
    order_id smallint NOT NULL,
    product_id smallint NOT NULL,
    unit_price real NOT NULL,
    quantity smallint NOT NULL,
    discount real NOT NULL,
    CONSTRAINT pk_order_details PRIMARY KEY (order_id, product_id)
);

-- Create Indexes for Performance
CREATE INDEX idx_orders_customer_id ON public.orders(customer_id);
CREATE INDEX idx_orders_employee_id ON public.orders(employee_id);
CREATE INDEX idx_orders_ship_via ON public.orders(ship_via);
CREATE INDEX idx_orders_region ON public.orders(region);

CREATE INDEX idx_products_supplier_id ON public.products(supplier_id);
CREATE INDEX idx_products_category_id ON public.products(category_id);
CREATE INDEX idx_products_region ON public.products(region);

CREATE INDEX idx_order_details_product_id ON public.order_details(product_id);
CREATE INDEX idx_order_details_order_id ON public.order_details(order_id);
```
