### SQL DDL Creation Script for Denormalized Schema

-- Drop existing denormalized tables if they exist
DROP TABLE IF EXISTS ENTIDADE_EDUCATIVA CASCADE;
DROP TABLE IF EXISTS INDICADOR_CURRICULAR CASCADE;

-- Table 1: ENTIDADE_EDUCATIVA
-- Merges MUNICIPIO, IES, and ESCOLA into one wide table.
-- Includes UF details embedded directly.
CREATE TABLE ENTIDADE_EDUCATIVA (
    id_entidade INT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    
    -- UF Attributes (Embedded)
    sigla_uf VARCHAR(2) UNIQUE NOT NULL,
    nome_uf VARCHAR(100) NOT NULL,
    
    -- Attributes specific to MUNICIPIO (Duplicated here)
    id_municipio_orig VARCHAR(20) GENERATED ALWAYS AS (id_municipio::TEXT) STORED, -- Kept for reference if needed
    nome_municipio VARCHAR(100) NOT NULL,
    qtd_habitantes INT CHECK(qtd_habitantes >= 0),
    qtd_ies INT CHECK(qtd_ies >= 0),
    qtd_escolas INT CHECK(qtd_escolas >= 0),
    media_ideb REAL,
    media_saeb REAL,
    media_enade REAL,
    total_participantes INT CHECK(total_participantes >= 0),
    
    -- Attributes specific to IES (Duplicated here)
    id_ies_orig VARCHAR(20) GENERATED ALWAYS AS (id_ies::TEXT) STORED,
    sigla_ies VARCHAR(20),
    codigo_ies INT UNIQUE NOT NULL,
    rede VARCHAR(100) NOT NULL,
    
    -- Attributes specific to ESCOLA (Duplicated here)
    id_escola_orig VARCHAR(20) GENERATED ALWAYS (id_escola::TEXT) STORED,
    nome_escola VARCHAR(100) NOT NULL,
    cod_escola INTEGER UNIQUE NOT NULL,
    
    -- Metadata for denormalization
    tipo_entidade VARCHAR(20) NOT NULL DEFAULT 'MUNICIPIO' -- ENUM-like string for flexibility
    
    -- Composite Primary Key for uniqueness within the merged table logic
    CONSTRAINT pk_entidade_composite PRIMARY KEY (id_entidade, tipo_entidade, sigla_uf)
);

-- Table 2: INDICADOR_CURRICULAR
-- Merges CURSO, ANO, ENADE, SAEB, and IDEB into one wide table.
-- Allows querying metrics for a specific course/year without joins.
CREATE TABLE INDICADOR_CURRICULAR (
    id_curso_orig VARCHAR(20) GENERATED ALWAYS AS (id_curso::TEXT) STORED,
    nome_curso VARCHAR(100) NOT NULL,
    
    -- ANO Attributes (Embedded)
    id_ano_orig VARCHAR(20) GENERATED ALWAYS AS (id_ano::TEXT) STORED,
    ano_referencia INT NOT NULL,
    
    -- ENADE Attributes (Duplicated per year/course)
    total_inscritos INT CHECK(total_inscritos >= 0),
    total_concluintes INT CHECK(total_concluintes >= 0),
    nota_bruta_ce REAL,
    nota_padronizada_ce REAL,
    nota_bruta_fg REAL,
    nota_padronizada_fg REAL,
    nota_enade_continua REAL,
    nota_enade_faixa REAL,
    
    -- SAEB Attributes (Duplicated per year/course)
    nota_mat REAL,
    nota_port REAL,
    nota_padrao REAL,
    
    -- IDEB Attributes (Duplicated per year/course)
    rend_1 REAL,
    rend_2 REAL,
    rend_3 REAL,
    rend_4 REAL,
    ind_rend REAL,
    nota_ideb REAL,
    
    -- Unique Constraint to prevent duplicate years for the same course
    CONSTRAINT pk_curso_ano_uniq PRIMARY KEY (id_curso_orig, ano_referencia)
);
