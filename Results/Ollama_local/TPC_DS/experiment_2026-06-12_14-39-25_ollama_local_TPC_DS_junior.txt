### SQL DDL Creation Script

Below is the complete SQL script to create the denormalized schema. This script creates a single wide table per order type (Purchase, Catalog, Web) and their corresponding return tables, removing the need for most JOINs.

-- ==========================================
-- DENORMALIZED SCHEMA FOR ANALYTICS
-- ==========================================

-- 1. PURCHASE ANALYTICS TABLE
-- Merges s_purchase and s_purchase_lineitem into a single table.
-- This allows direct aggregation of sales data without joining headers and lines.
CREATE TABLE s_purchase_analytics (
    purc_purchase_id           INTEGER NOT NULL,
    purc_store_id              CHAR(16) NOT NULL,
    purc_customer_id           CHAR(16) NOT NULL,
    purc_purchase_date         CHAR(10) NOT NULL,
    purc_purchase_time         INTEGER NOT NULL,
    purc_register_id           INTEGER,
    purc_clerk_id              INTEGER,
    purc_comment               CHAR(100),
    
    -- Denormalized Line Item Data (Flattened)
    plin_line_number           INTEGER NOT NULL,
    plin_item_id               CHAR(16) NOT NULL,
    plin_promotion_id          CHAR(16),
    plin_quantity              INTEGER NOT NULL,
    plin_sale_price            NUMERIC(7,2) NOT NULL,
    plin_coupon_amt            NUMERIC(7,2),
    plin_comment               VARCHAR(100),
    
    -- Composite Primary Key to ensure uniqueness of line items
    PRIMARY KEY (purc_purchase_id, plin_line_number, plin_item_id)
);

-- 2. CATALOG ORDER ANALYTICS TABLE
-- Merges s_catalog_order and s_catalog_order_lineitem.
CREATE TABLE s_catalog_order_analytics (
    cord_order_id              INTEGER NOT NULL,
    cord_bill_customer_id      CHAR(16) NOT NULL,
    cord_ship_customer_id      CHAR(16) NOT NULL,
    cord_order_date            CHAR(10) NOT NULL,
    cord_order_time            INTEGER NOT NULL,
    cord_ship_mode_id          CHAR(16) NOT NULL,
    cord_call_center_id        CHAR(16) NOT NULL,
    cord_order_comments        VARCHAR(100),
    
    -- Denormalized Line Item Data
    clin_line_number           INTEGER NOT NULL,
    clin_item_id               CHAR(16) NOT NULL,
    clin_promotion_id          CHAR(16),
    clin_quantity              INTEGER NOT NULL,
    clin_sales_price           NUMERIC(7,2) NOT NULL,
    clin_coupon_amt            NUMERIC(7,2),
    clin_warehouse_id          CHAR(16) NOT NULL,
    clin_ship_date             CHAR(10) NOT NULL,
    clin_catalog_number        INTEGER,
    clin_catalog_page_number   INTEGER,
    clin_ship_cost             NUMERIC(7,2),
    
    PRIMARY KEY (cord_order_id, clin_line_number, clin_item_id)
);

-- 3. WEB ORDER ANALYTICS TABLE
-- Merges s_web_order and s_web_order_lineitem.
CREATE TABLE s_web_order_analytics (

    -- Denormalized Line Item Data

    wlin_item_id               CHAR(16) NOT NULL,
    wlin_promotion_id          CHAR(16),
    wlin_quantity              INTEGER NOT NULL,
    wlin_sales_price           NUMERIC(7,2) NOT NULL,
    wlin_coupon_amt            NUMERIC(7,2),
    wlin_warehouse_id          CHAR(16) NOT NULL,
    wlin_ship_date             CHAR(10) NOT NULL,
    wlin_ship_cost             NUMERIC(7,2),
    wlin_web_page_id           CHAR(16),
    
    PRIMARY KEY (word_order_id, wlin_line_number, wlin_item_id)
);

-- 4. PURCHASE RETURNS ANALYTICS TABLE
-- Merges s_store_returns with relevant purchase context.
-- Note: We do not merge with s_purchase here because the return table already contains
-- the purchase_id reference. We keep it denormalized by adding store/customer context
-- directly if not present, or simply relying on the existing return columns.
-- To maximize denormalization, we ensure the return table has the Purchase ID as PK.
CREATE TABLE s_purchase_returns_analytics (

    sret_line_number           INTEGER NOT NULL,
    sret_item_id               CHAR(16) NOT NULL,
    sret_customer_id           CHAR(16) NOT NULL,
    sret_return_date           CHAR(10) NOT NULL,
    sret_return_time           CHAR(10) NOT NULL,
    sret_ticket_number         CHAR(20) NOT NULL,
    
    -- Financial and Quantity Details (Already in original, kept for clarity)
    sret_return_qty            INTEGER NOT NULL,
    sret_return_amt            NUMERIC(7,2) NOT NULL,
    sret_return_tax            NUMERIC(7,2),

    sret_return_ship_cost      NUMERIC(7,2),

    sret_reversed_charge       NUMERIC(7,2),
    sret_store_credit          NUMERIC(7,2),
    sret_reason_id             CHAR(16),
    
    -- Composite Primary Key
    PRIMARY KEY (sret_purchase_id, sret_line_number, sret_item_id)
);

-- 5. CATALOG RETURNS ANALYTICS TABLE
-- Merges s_catalog_returns with order context.
CREATE TABLE s_catalog_returns_analytics (
    cret_call_center_id        CHAR(16) NOT NULL,
    cret_order_id              INTEGER NOT NULL,
    cret_line_number           INTEGER NOT NULL,
    cret_item_id               CHAR(16) NOT NULL,
    cret_return_customer_id     CHAR(16) NOT NULL,

    PRIMARY KEY (cret_order_id, cret_line_number, cret_item_id)

-- 6. INVENTORY SNAPSHOT TABLE
-- Flattened to avoid joins between Warehouse, Item, and Date.
-- We assume this table represents historical snapshots.
CREATE TABLE s_inventory_snapshots (

    invn_qty_on_hand           INTEGER NOT NULL,
    
    -- Composite Primary Key for time-series tracking
    PRIMARY KEY (invn_warehouse_id, invn_item_id, invn_date)
);
