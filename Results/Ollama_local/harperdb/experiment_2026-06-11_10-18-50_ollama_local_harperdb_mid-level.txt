);

CREATE TABLE dim_entitlements (
    entitlement_id INT PRIMARY KEY,
    feature_type VARCHAR(4),
    package_type VARCHAR(4),
    use_limit INT,
    active_date DATE,
    inactive_date DATE,
    active_flag INT
);

-- Create Fact Table
-- This table is denormalized to include key attributes from the dimensions.
-- It acts as the single source of truth for the relationship and measures.
CREATE TABLE fact_user_entitlements (
    user_id INT,
    entitlement_id INT,
    feature_type VARCHAR(4),
    package_type VARCHAR(4),
    use_limit INT,
    use_count INT,
    active_date DATE,
    inactive_date DATE,
    active_flag INT,
    PRIMARY KEY (user_id, entitlement_id),
    FOREIGN KEY (user_id) REFERENCES dim_users(user_id),

);
