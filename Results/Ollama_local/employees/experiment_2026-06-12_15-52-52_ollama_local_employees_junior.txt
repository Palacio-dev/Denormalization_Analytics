-- Drop existing tables if they exist to ensure a clean state for the new schema
DROP TABLE IF EXISTS employees.employee_base;
DROP TABLE IF EXISTS employees.employee_stats;

-- 1. Create the Denormalized Employee Base Table
-- This table merges employee, department, salary, and title data to eliminate joins.
CREATE TABLE IF NOT EXISTS employees.employee_base (
    id bigint NOT NULL DEFAULT nextval('employees.id_employee_seq'::regclass),
    first_name character varying(14) COLLATE pg_catalog."default" NOT NULL,
    last_name character varying(16) COLLATE pg_catalog."default" NOT NULL,
    gender character varying(1) COLLATE pg_catalog."default" NOT NULL,
    birth_date date NOT NULL,
    hire_date date NOT NULL,
    
    -- Merged Department Data
    dept_name character varying(40) COLLATE pg_catalog."default" NOT NULL,
    dept_manager_id bigint, -- Nullable as not all employees might be managers
    dept_from_date date,
    dept_to_date date,
    
    -- Merged Salary Data
    salary_amount bigint,
    salary_from_date date,
    salary_to_date date,
    
    -- Merged Title Data
    title_name character varying(50) COLLATE pg_catalog."default",
    title_from_date date,
    title_to_date date,
    
    -- Primary Key
    CONSTRAINT employees_employee_base_pkey PRIMARY KEY (id)
);

-- 2. Create the Pre-aggregated Stats Table (Optional for heavy reporting)
CREATE TABLE IF NOT EXISTS employees.employee_stats (
    id bigint NOT NULL DEFAULT nextval('employees.id_employee_seq'::regclass),
    dept_name character varying(40) COLLATE pg_catalog."default" NOT NULL,
    avg_salary bigint,
    avg_tenure_days bigint,
    CONSTRAINT employees_employee_stats_pkey PRIMARY KEY (id)
);

-- 3. Create Triggers to Maintain Denormalized Data (Simplified Logic)
-- Note: In a real-world scenario, you would implement more robust triggers to handle
-- complex date ranges and concurrent updates. Here is a simplified example for insertion.

-- Trigger to update employee_base when an employee record is inserted
CREATE OR REPLACE FUNCTION update_employee_base_on_insert()
RETURNS TRIGGER AS $$
BEGIN
    -- Update Base Info
    UPDATE employees.employee_base
    SET first_name = NEW.first_name,
        last_name = NEW.last_name,
        gender = NEW.gender,
        birth_date = NEW.birth_date,
        hire_date = NEW.hire_date;
    
    -- Update Department Info (Assuming active employment)
    UPDATE employees.employee_base
    SET dept_name = NEW.dept_name,
        dept_manager_id = NEW.dept_manager_id,
        dept_from_date = NEW.dept_from_date,
        dept_to_date = NEW.dept_to_date;
    
    -- Update Salary Info
    UPDATE employees.employee_base
    SET salary_amount = NEW.salary_amount,
        salary_from_date = NEW.salary_from_date,
        salary_to_date = NEW.salary_to_date;
    
    -- Update Title Info
    UPDATE employees.employee_base
    SET title_name = NEW.title_name,
        title_from_date = NEW.title_from_date,
        title_to_date = NEW.title_to_date;
    
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_employee_base_on_insert
AFTER INSERT ON employees.employee
FOR EACH ROW
EXECUTE FUNCTION update_employee_base_on_insert();

-- Trigger to update employee_base when an employee record is updated
CREATE TRIGGER trg_employee_base_on_update
AFTER UPDATE ON employees.employee
FOR EACH ROW
EXECUTE FUNCTION update_employee_base_on_insert();

-- Trigger to delete from employee_base when employee is deleted
CREATE TRIGGER trg_employee_base_on_delete
AFTER DELETE ON employees.employee
FOR EACH ROW
EXECUTE FUNCTION update_employee_base_on_delete();
