-- View: Sales Performance by Territory
CREATE VIEW analytics_public.vw_sales_territory_performance AS
SELECT 
    t.territoryid,
    t.territoryname,
    COUNT(*) AS total_orders,
    SUM(s.subtotal) AS total_revenue,
    SUM(s.freight) AS total_freight,
    SUM(s.taxamt) AS total_tax,
    SUM(s.subtotal) - SUM(s.freight) - SUM(s.taxamt) AS gross_profit
FROM analytics_public.fact_sales s
JOIN analytics_public.dim_sales_territory t ON s.territoryid = t.territoryid
GROUP BY t.territoryid, t.territoryname;

-- View: Employee Department Summary
CREATE VIEW analytics_public.vw_employee_dept_summary AS
SELECT 
    e.departmentid,
    e.departmentname,
    e.groupname,
    COUNT(*) AS employee_count,
    AVG(e.vacationhours) AS avg_vacation_hours,
    AVG(e.sickleavehours) AS avg_sick_leave_hours
FROM analytics_public.fact_employee e
GROUP BY e.departmentid, e.departmentname, e.groupname;
