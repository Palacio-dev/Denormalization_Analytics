### Analytical Physical Model Design (SQL DDL)

The following PostgreSQL script creates the denormalized Star Schema. It includes surrogate keys, dimensional flattening, derived metrics, and partitioning on the fact table to support large-scale aggregations.

```sql
-- ==============================================================================
-- 1. DIMENSION TABLES
-- ==============================================================================

-- Date Dimension
CREATE TABLE IF NOT EXISTS public.dim_date
(
    date_key integer NOT NULL, -- Format: YYYYMMDD
    full_date date NOT NULL,
    year integer NOT NULL,
    quarter integer NOT NULL,
    month integer NOT NULL,
    month_name character varying(20) NOT NULL,
    day_of_month integer NOT NULL,
    day_of_week integer NOT NULL,
    day_name character varying(20) NOT NULL,
    is_weekend boolean NOT NULL,
    CONSTRAINT pk_dim_date PRIMARY KEY (date_key)
);

-- Track Dimension (Flattened Snowflake: Track + Album + Artist + Genre + MediaType)
CREATE TABLE IF NOT EXISTS public.dim_track
(
    track_key serial NOT NULL,          -- Surrogate Key
    track_id integer NOT NULL,          -- Natural Key
    track_name character varying(200) NOT NULL,
    album_title character varying(160),
    artist_name character varying(120),
    genre_name character varying(120),
    media_type_name character varying(120),
    composer character varying(220),
    duration_ms integer,
    bytes integer,
    current_unit_price numeric(10, 2),
    valid_from timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    valid_to timestamp without time zone DEFAULT '9999-12-31',
    is_current boolean DEFAULT true,
    CONSTRAINT pk_dim_track PRIMARY KEY (track_key)
);

-- Customer Dimension
CREATE TABLE IF NOT EXISTS public.dim_customer
(
    customer_key serial NOT NULL,       -- Surrogate Key
    customer_id integer NOT NULL,       -- Natural Key
    first_name character varying(40) NOT NULL,
    last_name character varying(20) NOT NULL,
    full_name character varying(65) NOT NULL,
    company character varying(80),
    address character varying(70),
    city character varying(40),
    state character varying(40),
    country character varying(40),
    postal_code character varying(10),
    email character varying(60),
    CONSTRAINT pk_dim_customer PRIMARY KEY (customer_key)
);

-- Employee Dimension (Flattened Hierarchy)
CREATE TABLE IF NOT EXISTS public.dim_employee
(
    employee_key serial NOT NULL,       -- Surrogate Key
    employee_id integer NOT NULL,       -- Natural Key
    first_name character varying(20) NOT NULL,
    last_name character varying(20) NOT NULL,
    full_name character varying(45) NOT NULL,
    title character varying(30),
    manager_employee_id integer,
    manager_name character varying(45), -- Flattened from ReportsTo
    birth_date date,
    hire_date date,
    city character varying(40),
    state character varying(40),
    country character varying(40),
    email character varying(60),
    CONSTRAINT pk_dim_employee PRIMARY KEY (employee_key)
);

-- Playlist Dimension
CREATE TABLE IF NOT EXISTS public.dim_playlist
(
    playlist_key serial NOT NULL,       -- Surrogate Key
    playlist_id integer NOT NULL,       -- Natural Key
    playlist_name character varying(120),
    CONSTRAINT pk_dim_playlist PRIMARY KEY (playlist_key)
);


-- ==============================================================================
-- 2. FACT TABLES & BRIDGES
-- ==============================================================================

-- Sales Fact Table (Partitioned by Date)
-- Combines Invoice (Header) and InvoiceLine (Line Item)
CREATE TABLE IF NOT EXISTS public.fact_sales
(
    sales_key bigserial NOT NULL,
    date_key integer NOT NULL,          -- FK to dim_date
    customer_key integer NOT NULL,      -- FK to dim_customer
    employee_key integer NOT NULL,      -- FK to dim_employee (Support Rep)
    track_key integer NOT NULL,         -- FK to dim_track
    invoice_id integer NOT NULL,        -- Degenerate Dimension (Natural Key)
    invoice_line_id integer NOT NULL,   -- Degenerate Dimension (Natural Key)
    billing_address character varying(70),
    billing_city character varying(40),
    billing_state character varying(40),
    billing_country character varying(40),
    billing_postal_code character varying(10),
    quantity integer NOT NULL,
    unit_price numeric(10, 2) NOT NULL,
    sales_amount numeric(10, 2) NOT NULL, -- Derived Measure: Quantity * UnitPrice
    CONSTRAINT pk_fact_sales PRIMARY KEY (sales_key, date_key)
) PARTITION BY RANGE (date_key);

-- Create partitions for fact_sales (Example for a few years)
CREATE TABLE fact_sales_2021 PARTITION OF public.fact_sales FOR VALUES FROM (20210101) TO (20220101);
CREATE TABLE fact_sales_2022 PARTITION OF public.fact_sales FOR VALUES FROM (20220101) TO (20230101);
CREATE TABLE fact_sales_2023 PARTITION OF public.fact_sales FOR VALUES FROM (20230101) TO (20240101);
CREATE TABLE fact_sales_default PARTITION OF public.fact_sales DEFAULT;

-- Bridge Table for Playlist-Track Many-to-Many Relationship
CREATE TABLE IF NOT EXISTS public.bridge_playlist_track
(
    playlist_key integer NOT NULL,      -- FK to dim_playlist
    track_key integer NOT NULL,         -- FK to dim_track
    CONSTRAINT pk_bridge_playlist_track PRIMARY KEY (playlist_key, track_key)
);


-- ==============================================================================
-- 3. FOREIGN KEYS & INDEXING (Optimized for OLAP)
-- ==============================================================================

-- Foreign Keys for Fact Table
ALTER TABLE public.fact_sales
    ADD CONSTRAINT fk_fact_sales_date FOREIGN KEY (date_key) REFERENCES public.dim_date (date_key),
    ADD CONSTRAINT fk_fact_sales_customer FOREIGN KEY (customer_key) REFERENCES public.dim_customer (customer_key),
    ADD CONSTRAINT fk_fact_sales_employee FOREIGN KEY (employee_key) REFERENCES public.dim_employee (employee_key),
    ADD CONSTRAINT fk_fact_sales_track FOREIGN KEY (track_key) REFERENCES public.dim_track (track_key);

-- Foreign Keys for Bridge Table
ALTER TABLE public.bridge_playlist_track
    ADD CONSTRAINT fk_bridge_playlist FOREIGN KEY (playlist_key) REFERENCES public.dim_playlist (playlist_key),
    ADD CONSTRAINT fk_bridge_track FOREIGN KEY (track_key) REFERENCES public.dim_track (track_key);

-- B-Tree Indexes on Fact Table Foreign Keys (Crucial for Star Schema Joins)
CREATE INDEX idx_fact_sales_date_key ON public.fact_sales(date_key);
CREATE INDEX idx_fact_sales_customer_key ON public.fact_sales(customer_key);
CREATE INDEX idx_fact_sales_employee_key ON public.fact_sales(employee_key);
CREATE INDEX idx_fact_sales_track_key ON public.fact_sales(track_key);

-- Indexes on frequently filtered dimensional attributes
CREATE INDEX idx_dim_track_artist ON public.dim_track(artist_name);
CREATE INDEX idx_dim_track_genre ON public.dim_track(genre_name);
CREATE INDEX idx_dim_customer_country ON public.dim_customer(country);
CREATE INDEX idx_dim_date_ym ON public.dim_date(year, month);
```
